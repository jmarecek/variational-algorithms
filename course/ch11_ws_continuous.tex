\chapter{Warm-Started Quantum Optimization: Continuous}\label{C.WSContinuous}

\ona{This \chap{} continues \citet{egger2021warm}. Instead of rounding a relaxation classically and warm-starting from a cut (\Chap~\chref{C.WSRounded}), we warm-start from the \emph{continuous} solution of a convex relaxation, and let the measurement do the rounding. The mixer is rotated so that the warm state is its ground state, which keeps the convergence theorem of \Chap~\chref{C.QAOA}, and a regularisation $\varepsilon$ connects the method continuously to standard QAOA. We picture the construction on the Bloch sphere, prove what it guarantees at every depth and in the limit of large depth, and simulate it on a worked portfolio and on $250$ random ones: its landscapes, its behaviour with the depth, and the annealing that it accelerates.}

\section{The warm state and its mixer}\label{sec:ch11:warm}

\begin{frame}\ft{Continuous warm start}
\onp{\footnotesize}
\ona{We write the problem of \Chap~\chref{C.WSRounded} as}
\begin{equation}\label{eq:ch11:qubo}
    \min_{x\in\{0,1\}^n}f(x),\qquad f(x) = x^\top Ax+b^\top x+a_0,
\end{equation}
\ona{with $A$ symmetric. If $A\succeq0$, the relaxation}
\begin{equation}\label{eq:ch11:qp}
    \min_{x\in[0,1]^n}f(x)
\end{equation}
\ona{is a convex quadratic program, with a minimiser $c^\star\in[0,1]^n$; since $\{0,1\}^n\subset[0,1]^n$, $f(c^\star)\le F^*$, the least value of \eqref{eq:ch11:qubo}. Each bit corresponds to a qubit, and the final measurement of QAOA is a randomised rounding. The continuous warm start replaces $\ket+^{\otimes n}$ by}
\begin{equation}\label{eq:ch11:init}
    \ket{\phi^\star} = \bigotimes_{i=1}^nR_Y(\theta_i)\ket0,\qquad\theta_i = 2\arcsin\sqrt{c_i^\star},
\end{equation}
\ona{whose qubit $i$ reads $1$ with probability $\sin^2(\theta_i/2) = c_i^\star$, and the mixer $-\sum_iX_i$ by}
\begin{equation}\label{eq:ch11:mixer}
    H_M^{(\rm ws)} = \sum_{i=1}^nH_{M,i},\qquad H_{M,i} = \begin{pmatrix}2c_i^\star-1 & -2\sqrt{c_i^\star(1-c_i^\star)}\\ -2\sqrt{c_i^\star(1-c_i^\star)} & 1-2c_i^\star\end{pmatrix} = -\sin\theta_i\,X_i-\cos\theta_i\,Z_i,
\end{equation}
\ona{acting on qubit $i$. A layer applies $e^{-i\gamma H_C}$, with $H_C$ diagonal with the entries $f(x)$, and then $e^{-i\beta H_M^{(\rm ws)}}$. At $c_i^\star = \frac12$ for all $i$, $\theta_i = \pi/2$, and the state and the mixer are those of standard QAOA \citep{egger2021warm}.}
\end{frame}

\begin{frame}\ft{The mixer and its ground state}
\onp{\footnotesize}
\begin{lem}\label{lem:ch11:mixer}
Let $\boldsymbol n_i = (\sin\theta_i,0,\cos\theta_i)$, the Bloch vector of $R_Y(\theta_i)\ket0$, and $\boldsymbol n_i\cdot\boldsymbol\sigma = \sin\theta_iX+\cos\theta_iZ$. Then $H_{M,i} = -\boldsymbol n_i\cdot\boldsymbol\sigma$ has the ground state $R_Y(\theta_i)\ket0$, with the eigenvalue $-1$; $\ket{\phi^\star}$ is the ground state of $H_M^{(\rm ws)}$, with the eigenvalue $-n$; and
\begin{equation}\label{eq:ch11:rot}
    e^{-i\beta H_{M,i}} = \cos\beta\,\one+i\sin\beta\,\boldsymbol n_i\cdot\boldsymbol\sigma = R_Y(\theta_i)\,R_Z(-2\beta)\,R_Y(-\theta_i),
\end{equation}
a rotation of the Bloch sphere by the angle $2\beta$ about the axis through $\boldsymbol n_i$.
\end{lem}
\ona{\emph{Proof.} Since $|\boldsymbol n_i| = 1$, $(\boldsymbol n_i\cdot\boldsymbol\sigma)^2 = \one$, and its eigenvectors are the states with the Bloch vectors $\pm\boldsymbol n_i$, with the eigenvalues $\pm1$. The rotation $R_Y(\theta_i)$ turns the north pole by $\theta_i$ in the $x$--$z$ plane, so $R_Y(\theta_i)\ket0$ has the Bloch vector $\boldsymbol n_i$, and it is the eigenvector of $H_{M,i}$ with the eigenvalue $-1$. The terms of $H_M^{(\rm ws)}$ act on different qubits and commute, so the product of their ground states is the ground state of the sum, with the eigenvalue $-n$. Since $(\boldsymbol n_i\cdot\boldsymbol\sigma)^2 = \one$, the series of the exponential gives the first equality in \eqref{eq:ch11:rot}. Conjugation by $R_Y(\theta_i)$ turns $Z$ into $\cos\theta_iZ+\sin\theta_iX = \boldsymbol n_i\cdot\boldsymbol\sigma$, so $R_Y(\theta_i)e^{i\beta Z}R_Y(-\theta_i) = e^{i\beta\,\boldsymbol n_i\cdot\boldsymbol\sigma}$, and $e^{i\beta Z} = R_Z(-2\beta)$. \qed}
\onp{Proof: $(\boldsymbol n_i\cdot\boldsymbol\sigma)^2 = \one$, $R_Y(\theta_i)\ket0$ has the Bloch vector $\boldsymbol n_i$, and $R_Y(\theta_i)ZR_Y(-\theta_i) = \boldsymbol n_i\cdot\boldsymbol\sigma$.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item The mixer leaves the warm state where it is: only the cost layer moves it, and the mixer then turns it about $\boldsymbol n_i$.
    \item On hardware, two extra rotations $R_Y(\pm\theta_i)$ for each qubit and layer.
\end{itemize}
\end{frame}

\begin{frame}\ft{Frozen qubits and the regularisation}
\onp{\footnotesize}
\begin{lem}\label{lem:ch11:frozen}
If $c_i^\star\in\{0,1\}$, every state that the circuit prepares, at every depth and for all angles, is $\ket{c_i^\star}\otimes\ket\varphi$ for a state $\ket\varphi$ of the other qubits, and its measurement returns $x_i = c_i^\star$ with probability one. If every optimal string has $x_i\ne c_i^\star$, no optimal string is ever sampled.
\end{lem}
\ona{\emph{Proof.} Then $\theta_i\in\{0,\pi\}$, qubit $i$ starts in $\ket{c_i^\star}$, and $H_{M,i} = -\cos\theta_i\,Z_i$ is diagonal, as is $H_C$. Every gate commutes with $Z_i$, and maps $\ket{c_i^\star}\otimes\ket\varphi$ to a state of the same form. \qed}
\ona{This is the obstruction of \Chap~\chref{C.QAOA}: a coordinate subspace invariant under every layer. The remedy is to keep the angles away from the poles, with a regularisation $\varepsilon\in[0,\frac12]$,}
\begin{equation}\label{eq:ch11:eps}
    \theta_i = 2\arcsin\sqrt{c_i^\varepsilon},\qquad c_i^\varepsilon = \min\{\max\{c_i^\star,\varepsilon\},1-\varepsilon\},
\end{equation}
\ona{with the mixer \eqref{eq:ch11:mixer} built from the same angles.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item $\varepsilon$ interpolates between the warm start, at $\varepsilon = 0$, and standard QAOA, at $\varepsilon = \frac12$.
    \item The price of a larger $\varepsilon$: a warm state farther from $c^\star$, and a weaker bound below.
\end{itemize}
\end{frame}

\begin{frame}\ft{A worked portfolio}
\onp{\footnotesize}
\ona{As in \citet{egger2021warm}, we choose $B = 3$ of $n = 6$ assets with equal weights, minimising $q\,x^\top\Sigma x-\mu^\top x$ with the risk aversion $q = 2$, and enforce the budget $\boldsymbol1^\top x = B$ by the penalty $\lambda(\boldsymbol1^\top x-B)^2$, $\lambda = 3$. This is \eqref{eq:ch11:qubo} with}
\begin{equation}\label{eq:ch11:portfolio}
    A = q\Sigma+\lambda\boldsymbol1\boldsymbol1^\top,\qquad b = -\mu-2\lambda B\,\boldsymbol1,\qquad a_0 = \lambda B^2,
\end{equation}
\ona{and $A\succeq0$, so that \eqref{eq:ch11:qp} is convex. The prices of the assets follow geometric Brownian motions for $250$ days, with drifts uniform in $[-5\%,5\%]$ and volatilities uniform in $[-20\%,20\%]$, and $\mu$ and $\Sigma$ are the mean and the covariance of the daily returns, multiplied by $250$; with daily values, as in the source, risk and return would be negligible beside the penalty. The first instance of the generator has
\begin{equation*}
    \mu = (-0.039,0.222,-0.049,-0.508,0.006,0.041),\qquad\big(\sqrt{\Sigma_{ii}}\big)_i = (0.041,0.087,0.018,0.173,0.121,0.199).
\end{equation*}
The relaxation, solved exactly by enumerating the active sets, has the minimiser
\begin{equation*}
    c^\star = (0.569,1,0,0,0.878,0.546),
\end{equation*}
whose coordinates add up to $2.993$, and $f(c^\star) = -0.1661$. Of the $64$ portfolios, $20$ meet the budget; the best, $d^* = (0,1,0,0,1,1)$, has $F^* = -0.1484$, $0.0177$ above $f(c^\star)$, and the second best lies only $0.0073$ above it. Every portfolio that misses the budget lies at least $2.96$ above $F^*$. The numbers of this \chap{} are computed by \texttt{code/ch11\_figures.py}.}
\onp{\begin{itemize}\setlength\itemsep{0.2em}
    \item $n = 6$ assets, $B = 3$, $q = 2$, $\lambda = 3$; returns from geometric Brownian motions, annualised.
    \item $c^\star = (0.569,1,0,0,0.878,0.546)$, $f(c^\star) = -0.1661$; $d^* = (0,1,0,0,1,1)$, $F^* = -0.1484$.
    \item $20$ portfolios meet the budget; the second best is $0.0073$ above $F^*$, every other string at least $2.96$.

\end{frame}

\section{What the warm start guarantees}\label{sec:ch11:guarantee}

\begin{frame}\ft{The warm state as a rounding}
\onp{\footnotesize}
\begin{propn}\label{propn:ch11:round}
Let $x_1,\dots,x_n$ be independent, with $\mathrm{Pr}[x_i = 1] = c_i$. Then
\begin{equation}\label{eq:ch11:round}
    \mathbb E\,f(x) = f(c)+\sum_{i=1}^nA_{ii}\,c_i(1-c_i).
\end{equation}
Hence the energy of WS-QAOA with the regularisation $\varepsilon$ and optimised angles, at every depth, is at most $f(c^\varepsilon)+\sum_iA_{ii}c_i^\varepsilon(1-c_i^\varepsilon)$; for $\varepsilon = 0$ it is at most $F^*+\sum_iA_{ii}c_i^\star(1-c_i^\star)$.
\end{propn}
\ona{\emph{Proof.} Here $\mathbb E\,x_i = c_i$, $\mathbb E\,x_i^2 = \mathbb E\,x_i = c_i$, and $\mathbb E\,x_ix_j = c_ic_j$ for $i\ne j$. Hence $\mathbb E\,x^\top Ax = \sum_{i\ne j}A_{ij}c_ic_j+\sum_iA_{ii}c_i = c^\top Ac+\sum_iA_{ii}(c_i-c_i^2)$, and $\mathbb E\,b^\top x = b^\top c$. Measuring the regularised warm state gives such bits, with $c = c^\varepsilon$. With all angles zero, every layer is the identity, so the circuit of any depth prepares the warm state, and the optimised energy is at most its energy. Finally $f(c^\star)\le F^*$. \qed}
\ona{The term $\sum_iA_{ii}c_i(1-c_i)$ is the price of rounding the coordinates independently. It vanishes when $c$ is integral, and it is large when the relaxed optimum is fractional and the diagonal of $A$ is large. A budget enforced by the penalty $\lambda(\boldsymbol1^\top x-B)^2$ contributes $\lambda$ to every $A_{ii}$: independent rounding breaks the budget, with the variance $\sum_ic_i(1-c_i)$ of $\boldsymbol1^\top x$, and the circuit must restore it.}
\onp{\begin{itemize}\setlength\itemsep{0.2em}
    \item With all angles zero, the circuit of any depth prepares the warm state: the optimised energy is never worse.
    \item The price of independent rounding, $\sum_iA_{ii}c_i(1-c_i)$, is large when $c^\star$ is fractional and the penalty $\lambda$ of a budget is large.
\end{itemize}}
\end{frame}

\begin{frame}\ft{Convergence in the depth}
\onp{\footnotesize}
\begin{cor}\label{cor:ch11:conv}
If $0<c_i^\varepsilon<1$ for every $i$, in particular if $\varepsilon>0$, then for every $\delta>0$ there are a depth and angles for which the state of WS-QAOA lies within $\delta$ of the span of the optimal strings. If $c_i^\varepsilon\in\{0,1\}$ for the qubits $i$ of a set $J$, this holds if and only if some optimal string agrees with $c^\varepsilon$ on $J$.
\end{cor}
\ona{\emph{Proof.} With the sign of \Chap~\chref{C.QAOA}, the mixer is $-H_M^{(\rm ws)} = \sum_i(\sin\theta_iX_i+\cos\theta_iZ_i)$. Its entry between two strings that differ in bit $i$ alone is $\sin\theta_i = 2\sqrt{c_i^\varepsilon(1-c_i^\varepsilon)}>0$, and its other off-diagonal entries are zero; its graph is the hypercube, so it is a mixer in the sense of \Chap~\chref{C.QAOA}, its terms form a mixing family, and its top eigenvector is $\ket{\phi^\star}$, by Lemma~\ref{lem:ch11:mixer}. The convergence theorem of \citet{binkowski2023elementary}, proved in \Chap~\chref{C.QAOA}, applies. If $\theta_i\in\{0,\pi\}$ for $i\in J$, Lemma~\ref{lem:ch11:frozen} fixes these qubits, and the same argument applies to the other qubits and to $f$ restricted to the strings that agree with $c^\varepsilon$ on $J$. Its minimisers are optimal strings if some optimal string agrees with $c^\varepsilon$ on $J$, and otherwise no optimal string is ever sampled. \qed}
\ona{The corollary gives no rate. The guarantee of Proposition~\ref{propn:ch11:round} holds at every depth, but it does not promise that depth one improves on the warm state, and it is weak when $\sum_iA_{ii}c_i(1-c_i)$ is large. For MaxCut, $A$ is not positive semidefinite, \eqref{eq:ch11:qp} is not convex, and the warm start of \Chap~\chref{C.WSRounded} is used instead.}
\onp{\begin{itemize}\setlength\itemsep{0.2em}
    \item From the convergence theorem of \Chap~\chref{C.QAOA}: the warm mixer has the hypercube as its graph if $0<\theta_i<\pi$.
    \item No rate; no promise that depth one improves on the warm state.
    \item For MaxCut, $A\not\succeq0$ and the relaxation is not convex: the rounded warm start of \Chap~\chref{C.WSRounded} instead.
\end{itemize}}
\end{frame}

\section{Comparison and outlook}

\begin{frame}\ft{Rounded versus continuous warm starts}
\begin{center}\ona{\small}\onp{\tiny}
\begin{tabular}{lcc}\toprule
 & rounded (\Chap~\chref{C.WSRounded}) & continuous (this \chap) \\\midrule
relaxation & SDP, then random-hyperplane rounding & convex QP over $[0,1]^n$ \\
initial state & a cut, regularised by $\varepsilon$ & product state with $\mathrm{Pr}[x_i=1] = c_i^\star$ \\
mixer & rotated so that $(\pi/2, 0)$ reproduces the cut & rotated so that $\ket{\phi^\star}$ is its ground state \\
guarantee & inherits $\alpha_{\rm GW}$ for any angles & \ona{$f(c^\star)+\sum_iA_{ii}c_i^\star(1-c_i^\star)$}\onp{relaxation plus a variance}, every depth \\
convergence in $p$ & not covered by the adiabatic argument & for $\varepsilon>0$ (\Chap~\chref{C.QAOA}) \\
needs & $A$ arbitrary & $A\succeq0$ (or a projection) \\
qubits & $n$ & $n$ ($\Theta(n^2)$ to encode $Y^\star$ directly) \\\bottomrule
\end{tabular}
\end{center}
\ona{Both are instances of the design space of \Chap~\chref{C.WSRounded}: what to round, when, and how. The portfolio problem of this \chap{} is developed in full in \Chap~\chref{C.Portfolio}.}
\end{frame}

\begin{frame}\ft{Other warm starts}
\begin{itemize}\setlength\itemsep{0.3em}
    \item Polynomially solvable special cases of QUBO, and low-rank approximations of $A$ or of the SDP solution.
    \item Time-limited runs of exact solvers (CPLEX, Gurobi): the incumbent is a warm start, as it is in classical branch and bound.
    \item Encoding the matrix-valued SDP solution $Y^\star$ in a $\Theta(n^2)$-qubit state, then a parametrised circuit that contains the identity, then measurement: strong guarantees, too many qubits for now.
    \item In VQE, the particle-hole representation around the Hartree--Fock state is a warm start (\Chap~\chref{C.VQE}); in general, any good classical solution is one.
    \item Warm starts also address barren plateaus: a parametrisation that starts close to the optimum does not start on the plateau.
\end{itemize}
\ona{An implementation of WS-QAOA is available in Qiskit Optimization.}
\end{frame}

\ona{\section*{Exercises}}

\begin{frame}\ft{Exercises}
\onp{\footnotesize}
\begin{exe}[The mixer as a rotation]\label{exe:ch11:mixer}
Verify \eqref{eq:ch11:rot} by multiplying the three rotations. Show that a layer with $\gamma = 0$ leaves the warm state unchanged up to a phase, for every $\beta$, and find the angle $\beta$ at which the mixer maps a state at the angular distance $\alpha$ from $\boldsymbol n_i$ to its mirror image across $\boldsymbol n_i$.
\end{exe}
\begin{exe}[Rounding a quadratic]\label{exe:ch11:round}
Show that \eqref{eq:ch11:round} holds for every symmetric $A$, not only for $A\succeq0$. For MaxCut with non-negative weights $w_{ij}$, written as the minimisation of $f(x) = -\sum_{ij}w_{ij}(x_i+x_j-2x_ix_j)$, show that \eqref{eq:ch11:qp} is not convex, and that the uniform rounding $c = \frac12\boldsymbol1$ gives half the total weight. Why does this rule out the continuous warm start of this \chap{} for MaxCut?
\end{exe}
\end{frame}

\begin{frame}\ft{Exercises, continued}
\onp{\footnotesize}
\begin{exe}[The price of the budget]\label{exe:ch11:budget}
For independent bits with $\mathrm{Pr}[x_i = 1] = c_i$, show that $\mathbb E(\boldsymbol1^\top x-B)^2 = \sum_ic_i(1-c_i)+(\sum_ic_i-B)^2$, and that the probability of meeting the budget is at most $1-\mathbb E(\boldsymbol1^\top x-B)^2/\max\{B,n-B\}^2$. Evaluate this bound for the worked example, and compare it with the probability $0.471$ in \ona{Fig.~\ref{fig:ch11:dist}}\onp{the figure of the warm distribution}.
\end{exe}
\begin{exe}[Regularisation]\label{exe:ch11:eps}
For the worked example, compute the bound $f(c^\varepsilon)+\sum_iA_{ii}c_i^\varepsilon(1-c_i^\varepsilon)$ of \ona{Proposition~\ref{propn:ch11:round}}\onp{the proposition on rounding} for $\varepsilon = 0$, $0.1$, $\dots$, $0.5$, and show that at $\varepsilon = \frac12$ it is the mean of $\ket+^{\otimes6}$. Compare it with the energies of the warm states in \ona{Fig.~\ref{fig:ch11:eps}}\onp{the figure on regularisation}.
\end{exe}
\end{frame}
