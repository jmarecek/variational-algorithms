\chapter{Applications: Variational Solvers for Markowitz Portfolio Management}\label{C.Portfolio}

\ona{This \chap{} takes the portfolio problem of \citet{Markowitz1952}, the running example of \Chap~\chref{C.WSContinuous}, and treats it as an application, following the survey \citep{abbas2024challenges}. We picture the frontier of the worked portfolio of \Chap~\chref{C.WSContinuous}, and the penalty that turns its budget into a QUBO, and we determine the least penalty that works. We picture the mixers that keep the budget instead, on the graph of the feasible portfolios, with their spectra and the preparation of their initial state. We compare six variational solvers on the worked portfolio and on $100$ random ones. Finally, we ask what the exact solution is worth when, as always in practice, the means and the covariances are estimated from data. The numbers of this \chap{} are computed by \texttt{code/ch13\_figures.py}.}

\section{Mean--variance portfolio selection}

\begin{frame}\ft{Level 1: Markowitz}
\onp{\footnotesize}
\ona{Modern portfolio theory \citep{Markowitz1952} formalises diversification as a mean--variance trade-off. A risk-averse investor in a one-period frictionless market chooses fractions $x \in \R^N$ of a budget to invest in $N$ assets with estimated expected returns $\mu \in \R^N$ and positive semidefinite covariance $\Sigma \in \R^{N\times N}$. Three equivalent formulations:}
\begin{align}
    &\min_{x}\ \tfrac12x^\top\Sigma x \quad \text{s.t.}\ \mu^\top x \ge r_{\min},\ \boldsymbol1^\top x = 1,\ x \in \mathcal S;\\
    &\max_x\ \mu^\top x - \tfrac\lambda2x^\top\Sigma x\quad \text{s.t.}\ \boldsymbol1^\top x = 1,\ x \in \mathcal S,
\end{align}
\ona{or maximise return subject to a risk bound $x^\top\Sigma x \le \gamma^2$. With $\mathcal S = \R^N$ or a polyhedron these are convex quadratic programs with closed-form or interior-point solutions in milliseconds for thousands of assets. Realism enters through $\mathcal S$: no short selling, cardinality limits, position and sector exposure limits, turnover bounds.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item \textbf{Cardinality and lot sizes make it combinatorial:} choosing $B$ of $N$ assets, or integer numbers of lots, gives a mixed-integer quadratic program, $\mathbf{NP}$-hard in general, and the entry point for QUBO formulations.
    \item Level 1 is an excellent benchmark, and of limited practical value: it ignores estimation error.
\end{itemize}
\end{frame}

\begin{frame}\ft{Levels 2 and 3: uncertainty and time}
\ona{\textbf{Level 2.} Small changes in the estimates $\mu,\Sigma$ produce large changes in the optimal $x$; with $N$ assets and $T$ observations the ratio $q = N/T$ is rarely small (e.g., $N = 600$ stocks, $T = 2500$ trading days, $q = 0.24$), so sample covariances have spurious small eigenvalues and returns are badly estimated. Remedies: Black--Litterman blending of market and expert views for $\mu$; robust optimization over an ellipsoidal uncertainty set $\mathcal U_\mu$; random-matrix cleaning and factor models for $\Sigma$; $L_1/L_2$ regularisation; transaction costs $\mathcal C(x\mid x_0)$, often $\beta\|x - x_0\|_1$, which penalise turnover; bootstrap resampling and clustering of correlated assets.}
\begin{equation}
    \max_x\ \tilde\mu^\top x - \mathcal C(x\mid x_0) - \tfrac\lambda2x^\top\tilde\Sigma x\quad\text{s.t.}\quad \boldsymbol1^\top x + \mathcal C(x\mid x_0) = 1,\ x \in \mathcal S.
\end{equation}
\ona{\textbf{Level 3.} Markets are dynamic: correlations fluctuate and flip sign; multi-period optimization over $h$ periods with a coupling term $b_\tau(x_{\tau-1},x_\tau)$ for transaction costs and time-varying constraints is where classical solver suites struggle, and an interesting domain for new methods.}
\onp{Level 2: estimation error dominates ($q = N/T$ not small); robustify $\mu$, clean $\Sigma$, penalise turnover. Level 3: multi-period, time-varying correlations; classically hard, practically needed.}
\ona{Most quantum work to date is at Level 1 with fewer than 50 variables; the survey's recommendation is to move to Levels 2 and 3, where marginal improvements have real value. We return to Level 2 at the end of the \chap.}
\end{frame}

\section{Encoding for variational solvers}

\begin{frame}\ft{From Markowitz to QUBO}
\onp{\footnotesize}
\ona{The standard encoding selects $B$ of $n$ assets with equal weights: $x \in \{0,1\}^n$,}
\begin{equation}\label{eq:ch13:qubo}
    \min_{x\in\{0,1\}^n}\ q\,x^\top\Sigma x - \mu^\top x\quad\text{s.t.}\quad \boldsymbol1^\top x = B
    \quad\longrightarrow\quad
    \min_{x\in\{0,1\}^n}\ q\,x^\top\Sigma x - \mu^\top x + \lambda(\boldsymbol1^\top x - B)^2,
\end{equation}
\ona{with a penalty weight $\lambda$ large enough, as made precise below (in \citet{egger2021warm}: $n = 6$, $B = 3$, $q = 2$, $\lambda = 3$). Then $x_i = (1 - Z_i)/2$ gives an Ising Hamiltonian $H_C$ with all-to-all $Z_iZ_j$ couplings, one qubit per asset.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item Integer lot sizes need $\lceil\log_2 K\rceil$ qubits per asset; the penalty makes $Q$ dense.
    \item Alternatives to penalties: the constraint-preserving mixers described next, Zeno projections, recursive elimination.
    \item Data: covariances and returns come from price series; in simulations, geometric Brownian motion with drifts in $[-5\%,5\%]$ and volatilities in $[-20\%,20\%]$ over 250 days.
    \item The classical competitor is CPLEX or Gurobi on the mixed-integer program, which solves $n \le 100$ instances in seconds; the convex relaxation over $[0,1]^n$ is a QP solved in milliseconds and is what warm-starts the quantum solver.
\end{itemize}
\end{frame}

\begin{frame}\ft{How large a penalty?}
\onp{\footnotesize}
\ona{Write $f_\lambda(x) = g(x)+\lambda(\boldsymbol1^\top x-B)^2$ for the penalised objective \eqref{eq:ch13:qubo}, $F^*$ for the least value of $g$ on the strings with $B$ ones, and $m_k$ for the least value of $g$ on the strings with $k$ ones.}
\begin{propn}\label{propn:ch13:penalty}
Every minimiser of $f_\lambda$ over $\{0,1\}^n$ meets the budget, and is then a minimiser of $g$ under the budget, if and only if $\lambda>\lambda^*$, where
\begin{equation}\label{eq:ch13:lstar}
    \lambda^* = \max_{k\ne B}\frac{F^*-m_k}{(k-B)^2}.
\end{equation}
Moreover $\lambda^*\le\Delta = \max_{x,i}|g(x\oplus e_i)-g(x)|$, the largest change of $g$ under the flip of one bit, and for \eqref{eq:ch13:g}, $\Delta\le\max_i\big(|\mu_i|+q\Sigma_{ii}+2q\sum_{j\ne i}|\Sigma_{ij}|\big)$.
\end{propn}
\ona{\emph{Proof.} On the strings with $B$ ones $f_\lambda = g$, with the least value $F^*$. On those with $k\ne B$ ones the least value of $f_\lambda$ is $m_k+\lambda(k-B)^2$, which exceeds $F^*$ for every $k\ne B$ if and only if $\lambda>\lambda^*$. For the bound, let $x$ have $k\ne B$ ones. Flipping $|k-B|$ suitable bits one at a time turns $x$ into a string $y$ with $B$ ones, and each flip changes $g$ by at most $\Delta$, so $F^*\le g(y)\le g(x)+|k-B|\Delta$. Hence $F^*-m_k\le|k-B|\Delta\le(k-B)^2\Delta$. Finally, setting bit $i$ of $x$ to one changes \eqref{eq:ch13:g} by $q\Sigma_{ii}+2q\sum_{j\ne i}\Sigma_{ij}x_j-\mu_i$, whose absolute value is at most $|\mu_i|+q\Sigma_{ii}+2q\sum_{j\ne i}|\Sigma_{ij}|$. \qed}
\onp{Proof: compare $m_k+\lambda(k-B)^2$ with $F^*$; reach the budget from $x$ by $|k-B|$ flips, each changing $g$ by at most $\Delta$.}

\ona{On the worked portfolio, $\lambda^* = 0.0367$, attained at $k = 2$; $\Delta = 0.573$, and the bound on $\Delta$ is $0.580$. The penalty $\lambda = 3$ of \citet{egger2021warm} is $82$ times $\lambda^*$. It is safe, but it costs: the spectrum of $H_C$ is stretched from the width $0.807$ of the feasible portfolios to $27.66$.}
\onp{Worked portfolio: $\lambda^* = 0.0367$, $\Delta = 0.573$; $\lambda = 3$ is $82\lambda^*$, and stretches the spectrum of $H_C$ from $0.807$ to $27.66$.}
\end{frame}

\section{Mixers that keep the budget}

\begin{frame}\ft{The quantum alternating operator ansatz}
\onp{\footnotesize}

\begin{frame}\ft{The spectrum of the complete $XY$ mixer}
\onp{\footnotesize}
\begin{propn}\label{propn:ch13:spin}
Let $H_{XY} = \sum_{i<j}\frac12(X_iX_j+Y_iY_j)$ on $n$ qubits. On $\mathcal F$, the span of the strings with $B$ ones, $H_{XY}$ has the eigenvalues
\begin{equation}\label{eq:ch13:spin}
    (B-j)(n-B-j)-j,\qquad j = 0,1,\dots,\min\{B,n-B\},
\end{equation}
with the multiplicities $\binom nj-\binom n{j-1}$. The largest, $B(n-B)$, is simple, with the Dicke state as its eigenvector.
\end{propn}
\ona{\emph{Proof.} Let $S_a = \frac12\sum_i\sigma_i^a$ for $a\in\{x,y,z\}$ be the components of the total spin, and $S^2 = S_x^2+S_y^2+S_z^2$. Since $X_i^2 = Y_i^2 = \one$, $S_x^2+S_y^2 = \frac n2+\frac12\sum_{i<j}(X_iX_j+Y_iY_j)$, and so $H_{XY} = S^2-S_z^2-\frac n2$. A string with $B$ ones has $S_z = \frac n2-B$. The space of $n$ qubits splits into irreducible representations of the total spin; on one of spin $s$, $S^2 = s(s+1)$, and $S_z$ takes each of the values $-s,\dots,s$ once. The eigenspace of $S_z$ for the value $M$ has the dimension $\binom n{n/2-M}$, so there are $\binom n{n/2-s}-\binom n{n/2-s-1}$ representations of spin $s$; one of them meets $\mathcal F$, in one dimension, if and only if $s\ge|\frac n2-B|$. Writing $s = \frac n2-j$, the eigenvalue $s(s+1)-(\frac n2-B)^2-\frac n2$ is \eqref{eq:ch13:spin}, for $j\le\min\{B,n-B\}$. For $j = 0$, $s = \frac n2$ is the symmetric subspace, whose only state with $B$ ones is the Dicke state. \qed}
\onp{Proof: $H_{XY} = S^2-S_z^2-\frac n2$ with the total spin $S$; $S_z = \frac n2-B$ on $\mathcal F$; spin $s = \frac n2-j$ occurs $\binom nj-\binom n{j-1}$ times.}

\ona{For the worked portfolio the eigenvalues are $9$, $3$, $-1$ and $-3$, with the multiplicities $1$, $5$, $9$ and $5$; they are integers of the same parity, so $e^{-i\beta H_{XY}}$ has the period $\pi$ in $\beta$, up to a phase. \begin{frame}\ft{Preparing the Dicke state}
\onp{\footnotesize}
\ona{Write $\ket{D^m_k}$ for the Dicke state of $m$ qubits with $k$ ones, $\binom mk^{-1/2}\sum_{\boldsymbol1^\top x = k}\ket x$.}
\begin{lem}\label{lem:ch13:dicke}
For $0<k<m$,
\begin{equation}\label{eq:ch13:dicke}
    \ket{D^m_k} = \sqrt{\tfrac km}\,\ket{D^{m-1}_{k-1}}\ket1+\sqrt{\tfrac{m-k}m}\,\ket{D^{m-1}_k}\ket0,
\end{equation}
where the last factor is qubit $m$. Measuring the qubits of $\ket{D^n_B}$ from the $n$th to the first therefore gives $1$ on qubit $m$ with the probability $k/m$, if $k$ ones remain for the $m$ qubits not yet measured, and each string with $B$ ones has the probability $1/\binom nB$.
\end{lem}
\ona{\emph{Proof.} Of the $\binom mk$ strings with $k$ ones, $\binom{m-1}{k-1}$ end in $1$ and $\binom{m-1}k$ end in $0$, and $\binom{m-1}{k-1}/\binom mk = k/m$. Along any string the numerators of the probabilities run through $B,B-1,\dots,1$ and $n-B,\dots,1$, and the denominators through $n,n-1,\dots,1$, so their product is $B!(n-B)!/n!$. \qed}
\onp{Proof: $\binom{m-1}{k-1}/\binom mk = k/m$; the product of the probabilities along any string is $B!(n-B)!/n!$.}

\ona{\citet{baertschi2019deterministic} turn \eqref{eq:ch13:dicke} into a circuit: from $\ket{0}^{\otimes(n-B)}\ket1^{\otimes B}$, each step moves amplitude $\sqrt{k/m}$ by a rotation controlled on its neighbours, with $O(nB)$ gates and depth $O(n)$ in all; \citet{baertschi2022shortdepth} shorten the depth on restricted connectivity. The Grover mixer $e^{-i\beta\ket F\!\bra F}$ is, in turn, $U_F\,e^{-i\beta\ket0\!\bra0}\,U_F^\dagger$ for the preparation $U_F$ of $\ket F = U_F\ket0$, so that the Grover mixer costs about twice the preparation, and a controlled phase \citep{baertschi2020grover}.}
\onp{\citet{baertschi2019deterministic}: a circuit with $O(nB)$ gates and depth $O(n)$. The Grover mixer is $U_Fe^{-i\beta\ket0\!\bra0}U_F^\dagger$ \citep{baertschi2020grover}.}
\end{frame}

\begin{frame}\ft{The Grover mixer}
\onp{\footnotesize}
\begin{propn}\label{propn:ch13:grover}
Let $\ket F$ be the uniform superposition of a feasible set $F$, $H_C$ diagonal with the values $f(x)$, and $\ket\psi = U_M(\beta_p)e^{-i\gamma_pH_C}\cdots U_M(\beta_1)e^{-i\gamma_1H_C}\ket F$ with the Grover mixer $U_M(\beta) = e^{-i\beta\ket F\!\bra F}$. Then $\ket\psi\in\spn\{\ket x:x\in F\}$, and $\braket{x|\psi} = a(f(x))$ for $x\in F$, for a function $a$ that depends on the angles but not on $x$.
\end{propn}
\ona{\emph{Proof.} By induction on $p$; at $p = 0$, $a\equiv|F|^{-1/2}$. The phase separator multiplies $a(v)$ by $e^{-i\gamma v}$. Since $(\ket F\!\bra F)^2 = \ket F\!\bra F$, $U_M(\beta) = \one+(e^{-i\beta}-1)\ket F\!\bra F$, which adds the same number $(e^{-i\beta}-1)\braket{F|\psi}/\sqrt{|F|}$ to the amplitude of every $x\in F$, and nothing outside $F$. \qed}
\onp{Proof: $U_M(\beta) = \one+(e^{-i\beta}-1)\ket F\!\bra F$ adds the same number to every amplitude.}

\begin{itemize}\setlength\itemsep{0.2em}
    \item Portfolios of equal objective are equally likely, whatever their assets: the circuit sees only the histogram of $f$ over $F$.
    \item With $f$ the indicator of the non-optimal strings and $\gamma = \beta = \pi$, it is Grover's search (\ona{Exercise~\ref{exe:ch13:grover}}\onp{an exercise}), and about $\frac\pi4\sqrt{|F|}$ layers find a unique optimum.
    \item The convergence theorem of \Chap~\chref{C.QAOA} applies: $\ket F\!\bra F$ restricted to $\mathcal F$ is a positive multiple of the matrix of ones, whose graph is complete.
\end{itemize}
\end{frame}

\begin{frame}\ft{Other variational approaches to portfolios}
\begin{itemize}\setlength\itemsep{0.25em}
    \item \textbf{Constrained ansätze:} the quantum alternating operator ansatz with $XY$ mixers preserves the budget exactly and compares favourably with penalty-based QAOA on trading and investment constraints \citep{Hodson2019}; Zeno-dynamics projections onto the feasible subspace \citep{herman23}.
    \item \textbf{Alternative objectives:} the Conditional Value at Risk aggregates only the best $\alpha$-fraction of samples \citep{Barkoutsos2020}, which is natural for a risk-averse investor and robust to noise.

\end{itemize}
\ona{All of these operate at Level 1. The main limitation of current approaches is the neglect of uncertainty in the inputs, of time-dependent correlations, and of the instability of the optimization; the QUBO with penalties only approximately encodes the constraints and requires post-processing. Bridging to Levels 2 and 3 is the research direction, and the next section shows why.}

\section{What the optimum is worth}

\begin{frame}\ft{Noise in the covariance}
\onp{\footnotesize}
\ona{So far $\mu$ and $\Sigma$ were given. In practice they are estimated from $T$ observations of the returns of $N$ assets, and the sample covariance is noisy even when the returns are not. The noise has a universal shape.}
\begin{thm}[\citet{marchenko1967distribution}]\label{thm:ch13:mp}
Let $R$ be a $T\times N$ matrix of independent entries with mean zero and variance one, $S = R^\top R/T$, and let $N,T\to\infty$ with $N/T\to q\in(0,1]$. Then the empirical distribution of the eigenvalues of $S$ converges almost surely to the distribution with the density
\begin{equation}\label{eq:ch13:mp}
    \rho_q(x) = \frac{\sqrt{(\lambda_+-x)(x-\lambda_-)}}{2\pi qx},\qquad\lambda_- \le x\le\lambda_+,\quad\lambda_\pm = (1\pm\sqrt q)^2.
\end{equation}
\end{thm}
\ona{The true covariance here is the identity, and all its eigenvalues are $1$; the sample spreads them over $[\lambda_-,\lambda_+]$. With a true covariance that has a few large eigenvalues, such as that of a market factor, those emerge from the bulk, and the bulk keeps the shape \eqref{eq:ch13:mp}; \citet{PhysRevLett.83.1467} found that most eigenvalues of the correlation matrix of the stocks of the S\&P~500 lie in that bulk. Cleaning the covariance, by shrinkage or by replacing the bulk, is the subject of \citet{LedoitWolf2004} and \citet{BunBouchaudPotters2017}.}
\onp{\begin{itemize}\setlength\itemsep{0.2em}
    \item True eigenvalues all $1$; the sample spreads them over $[(1-\sqrt q)^2,(1+\sqrt q)^2]$.
    \item Real data: a few factors emerge, the bulk keeps this shape \citep{PhysRevLett.83.1467}; cleaning by shrinkage \citep{LedoitWolf2004}.
\end{itemize}}
\end{frame}

\begin{frame}\ft{Optimism in sample}
\onp{\footnotesize}
\ona{The least variance of a portfolio $w$ with $\boldsymbol1^\top w = 1$ is $\sigma^2_{\min} = 1/(\boldsymbol1^\top\Sigma^{-1}\boldsymbol1)$, at $w^* = \Sigma^{-1}\boldsymbol1/(\boldsymbol1^\top\Sigma^{-1}\boldsymbol1)$. The \emph{sample minimum-variance portfolio} $\hat w = S^{-1}\boldsymbol1/(\boldsymbol1^\top S^{-1}\boldsymbol1)$ replaces $\Sigma$ by the sample covariance $S$.}
\begin{propn}\label{propn:ch13:insample}
Let the returns on $T$ days be independent and Gaussian, with the covariance $\Sigma\succ0$ on $N$ assets, $T>N$, and let $S$ be their sample covariance, with the divisor $T-1$. Then the in-sample variance of $\hat w$ satisfies
\begin{equation}\label{eq:ch13:insample}
    \mathbb E\,\hat w^\top S\hat w = \frac{T-N}{T-1}\,\sigma^2_{\min},
\end{equation}
while its true variance $\hat w^\top\Sigma\hat w$ is at least $\sigma^2_{\min}$.
\end{propn}
\ona{\emph{Proof.} $(T-1)S$ has the Wishart distribution $W_N(\Sigma,T-1)$. For $W\sim W_N(\Sigma,m)$ with $m\ge N$ and a fixed $a\ne0$, $(a^\top\Sigma^{-1}a)/(a^\top W^{-1}a)$ has the $\chi^2$ distribution with $m-N+1$ degrees of freedom \citep[Theorem~3.2.12]{muirhead1982aspects}. With $a = \boldsymbol1$ and $m = T-1$, $\hat w^\top S\hat w = 1/(\boldsymbol1^\top S^{-1}\boldsymbol1) = \sigma^2_{\min}\,\chi^2_{T-N}/(T-1)$, whose mean is \eqref{eq:ch13:insample}. The second claim holds since $\sigma^2_{\min}$ is the least true variance over all $w$ with $\boldsymbol1^\top w = 1$. \qed}
\onp{Proof: $(T-1)S$ is Wishart, and $(\boldsymbol1^\top\Sigma^{-1}\boldsymbol1)/(\boldsymbol1^\top W^{-1}\boldsymbol1)\sim\chi^2_{T-N}$ \citep{muirhead1982aspects}.}

\begin{itemize}\setlength\itemsep{0.2em}
    \item With $q = N/T$, the estimate promises about $(1-q)\sigma^2_{\min}$, and the portfolio delivers more than $\sigma^2_{\min}$: the optimiser fits the noise.
    \item The means are worse: the standard error of a mean return is the volatility over the square root of the length of the sample in years, whatever the frequency of the data.
\end{itemize}
\end{frame}

\begin{frame}\ft{What the simulations show}
\onp{\small}
\begin{itemize}\setlength\itemsep{0.35em}
    \item \textbf{Penalties:} any $\lambda>\lambda^*$ keeps the budget; on the worked portfolio $\lambda^* = 0.0367$, and the customary $\lambda = 3$ stretches the spectrum of $H_C$ from $0.807$ to $27.66$.
    \item \textbf{Mixers that keep the budget} reach low energies at depth one, but low energy is not the optimum when portfolios lie $0.0073$ apart. With depth, the complete $XY$ mixer is the best of them; the ring mixer does better from its own top eigenvector than from the Dicke state.
    \item \textbf{The warm start} finds the optimum most often, and on random instances almost always; but there the relaxation is nearly integral, and rounding it finds the optimum on $98$ of $100$.
    \item \textbf{CVaR} at the level $\alpha$ stops once the optimum carries the mass $\alpha$.
    \item \textbf{What the optimum is worth:} the optimum of the estimates is the true optimum in $14.6\%$ of the samples, and ignoring the means does better. Solving Level 1 to optimality is not where the value lies.
\end{itemize}
\end{frame}

\begin{frame}\ft{Exercises}
\onp{\footnotesize}
\begin{exe}[The least penalty]\label{exe:ch13:penalty}
Show that the bound $\lambda^*\le\Delta$ of \ona{Proposition~\ref{propn:ch13:penalty}}\onp{the proposition on the penalty} is attained: find $g$ on $\{0,1\}^2$, with $B = 1$, for which $\lambda^* = \Delta$. For which $\lambda$ does the minimiser of $f_\lambda$ meet the budget?
\end{exe}
\begin{exe}[The octahedron]\label{exe:ch13:octa}
For $n = 4$ and $B = 2$, the graph of the complete $XY$ mixer is the octahedron. Check \eqref{eq:ch13:spin} against its spectrum, and find an eigenvector for each eigenvalue.
\end{exe}
\begin{exe}[Grover's search]\label{exe:ch13:grover}
Let $x^*$ be the unique optimum in a feasible set $F$, and let $f(x) = 0$ if $x = x^*$ and $f(x) = 1$ otherwise. Show that a layer of the circuit of \ona{Proposition~\ref{propn:ch13:grover}}\onp{the proposition on the Grover mixer} with $\gamma = \beta = \pi$ is the Grover iteration, up to a sign, and that after $p$ layers $x^*$ has the probability $\sin^2((2p+1)\theta)$, where $\sin\theta = |F|^{-1/2}$.
\end{exe}
\end{frame}

\begin{frame}\ft{Exercises, continued}
\onp{\footnotesize}
\begin{exe}[CVaR saturates]\label{exe:ch13:cvar}
Show that the CVaR at the level $\alpha$ of the distribution of $f(x)$ in a state equals $F^*$ if and only if the optimal strings have the total probability at least $\alpha$. What does this imply for training on the CVaR?
\end{exe}
\begin{exe}[A small Markowitz pipeline]\label{exe:ch13:pipeline}
Using the code of \Chap~\chref{C.Motivation} as a template: (i) simulate 250 days of prices for $n = 6$ assets by geometric Brownian motion and compute $\mu$ and $\Sigma$; (ii) solve the cardinality-constrained problem \eqref{eq:ch13:qubo} exactly with Gurobi (binary variables, budget as a linear constraint) and its continuous relaxation over $[0,1]^n$ with cvxpy; (iii) build $H_C$ with the penalty encoding and run depth-one QAOA in Qiskit from a cold start and from the warm start $\theta_i = 2\arcsin\sqrt{c_i^\star}$ with the rotated mixer of \Chap~\chref{C.WSContinuous}; (iv) report the probability of sampling the optimal portfolio and the energy relative to $E_0$, and repeat over 50 random instances. Discuss how the results depend on $\lambda$ and on the overlap $d^{\star\top}c^\star/B$, and how often the optimum of the estimates is the optimum of the model that generated the prices.
\end{exe}
\end{frame}

\section{Closing the course}

\begin{frame}\ft{The course in one picture}
\onp{\footnotesize}
\begin{itemize}\setlength\itemsep{0.4em}
    \item \textbf{Variational methods} minimise an upper bound $\braket{\psi(\boldsymbol\theta)|H|\psi(\boldsymbol\theta)}$ over a circuit family (\Chaps~\chref{C.VQE}--\chref{C.QAOA}); they are heuristics whose training is \textbf{NP}-hard (\Chap~\chref{C.Inapprox}).
    \item \textbf{They converge} as the depth grows, by discretising an adiabatic path whose mixer is irreducible and non-negative (\Chaps~\chref{C.Annealing} and~\chref{C.QAOA}), with no rate.
    \item \textbf{Their optimizer converges} at the usual stochastic rate despite biased evaluations, up to a bias floor (\Chap~\chref{C.IterationComplexity}).
    \item \textbf{At utility scale} the angles must be set offline, transferred angles are cheap and good, and everything must be validated on hardware (\Chap~\chref{C.PerIteration}).
    \item \textbf{Warm starts} give the quantum algorithm the best classical solution for free and, with it, the classical guarantee; empirically, they turn a heuristic that loses to Goemans--Williamson into one that matches or beats it (\Chaps~\chref{C.Motivation}, \chref{C.WSRounded}).
\end{itemize}
\ona{Whether any of this yields a \emph{quantum advantage} in optimization is open (\Chaps~\chref{C.EQA}--\chref{C.Inapprox}); in chemistry, the variational state's job is to supply overlap for algorithms with guarantees (\Chaps~\chref{C.VQE}, \chref{C.VQLS}). What is not open is that ignoring the classical state of the art is the surest way not to get one.}
\end{frame}
