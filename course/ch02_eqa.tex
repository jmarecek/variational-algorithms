\chapter{The Exponential Quantum Advantage Hypothesis}\label{C.EQA}

\ona{This \chap{} examines the claim that motivated much of quantum computing for chemistry, and by extension the variational methods of this course: that quantum computers offer an \emph{exponential} advantage for computing the lowest energy of molecules. No background in physical chemistry is assumed. We first say what chemists compute and why it is hard, in the language of linear algebra; then explain the quantum algorithm on which the claim rests, quantum phase estimation, and what it requires as the molecule grows; and then follow the critical evaluation of \citet{lee2023evaluating}, placing it next to the complexity-theoretic view of the survey \cite{abbas2024challenges} and to the provable exponential speedup by annealing of \citet{nagaj2012quantum}, which we prove in \Chap~\chref{C.Annealing}. The lesson carries over to optimization: worst-case complexity is neither necessary nor sufficient for a practical advantage, and what decides the matter is the quality of the starting state.}

\section{The motivation}

\begin{frame}\ft{Feynman's challenge}
\ona{In May 1981, at a conference on the physics of computation at MIT, Richard Feynman asked whether a classical computer could simulate physics efficiently. His answer, for quantum systems, was no: to describe $n$ two-level systems one must store $2^n$ complex numbers, and the attempt to replace them by classical probabilities fails because quantum mechanics produces ``probabilities'' that can be negative. He proposed instead a computer built of quantum-mechanical elements, and ended the lecture, published the following year \cite{feynman1982simulating}, with the sentence that is quoted at the start of almost every talk on the subject:}
\begin{center}
\begin{minipage}{0.86\linewidth}
\textit{``\ldots\ nature isn't classical, dammit, and if you want to make a simulation of nature, you'd better make it quantum mechanical, and by golly it's a wonderful problem, because it doesn't look so easy.''}\\[0.2em]
\hspace*{\fill}--- R.~P.~Feynman, \emph{Simulating physics with computers}, 1982 \cite{feynman1982simulating}
\end{minipage}
\end{center}
\ona{Feynman's argument was about the size of the \emph{description}, which is indeed exponential. It took further work to turn it into an algorithm: \citet{lloyd1996universal} showed that a quantum computer can efficiently simulate the time evolution of any system with local interactions, and \citet{aspuru2005simulated} proposed to obtain molecular energies by combining such simulation with the phase-estimation algorithm of \citet{kitaev1995quantum}. From there it was a short step to the widespread belief that quantum computers would compute molecular energies exponentially faster than classical ones. Note what Feynman did \emph{not} claim: that the lowest energy is easy to find on a quantum computer. As we shall see, it is not, in the worst case.}
\onp{\begin{itemize}\item The description of $n$ quantum two-level systems has $2^n$ numbers. \item Lloyd 1996: quantum simulation of time evolution is efficient; Aspuru-Guzik et al.\ 2005: molecular energies by phase estimation. \item Feynman did \emph{not} claim that ground states are easy.\end{itemize}}
\end{frame}

\section{Quantum chemistry for non-chemists}

\begin{frame}\ft{What chemists compute, and why}
\ona{A molecule is a collection of atomic nuclei (heavy, positively charged) and electrons (light, negatively charged). Nuclei are thousands of times heavier than electrons, so one freezes their positions $\boldsymbol R$ and treats only the electrons quantum mechanically (the \emph{Born--Oppenheimer approximation}; \Chap~\chref{C.Motivation}). The electrons then have a lowest possible energy $E_0(\boldsymbol R)$, the \emph{ground-state energy}. Almost everything chemists want follows from this function of the nuclear positions:}
\begin{itemize}\setlength\itemsep{0.2em}
    \item \textbf{Structure:} the positions $\boldsymbol R$ minimising $E_0(\boldsymbol R)$ are the shape of the molecule.
    \item \textbf{Reactions:} the difference of $E_0$ between reactants and products says whether a reaction releases energy; the height of the lowest ``mountain pass'' between them (the \emph{barrier}) says how fast it runs.
    \item \textbf{Binding:} the energy of a drug bound to a protein minus the energies of the two separately.
\end{itemize}
\ona{The natural unit of energy is the \emph{Hartree} ($1\,\mathrm{Ha} \approx 27.2$ eV $\approx 627.5$ kcal/mol). Because rates depend exponentially on barriers, $e^{-\Delta E/k_BT}$, chemists ask for \emph{chemical accuracy}, $1$ kcal/mol $\approx 1.6\times10^{-3}$ Ha, which at room temperature fixes a reaction rate to within a factor of about five. Two features of this target matter for what follows. It is an \emph{absolute} error, and it is demanding even for a small, well-behaved molecule. \citet{eriksen2020ground} organised a blind challenge to compute the ground-state energy of benzene, C$_6$H$_6$, in the standard cc-pVDZ basis, with the six innermost (core) orbitals frozen: $30$ electrons in $108$ orbitals, a state space of the order of $10^{35}$ determinants. The \emph{correlation energy}, the part of the energy beyond the single-determinant Hartree--Fock approximation, is about $-863\times10^{-3}$ Ha, i.e., some $540$ times chemical accuracy; the coupled-cluster hierarchy gives $-862.37\times10^{-3}$ Ha at its highest affordable level (CCSDTQ), and adding triple excitations to CCSD alone lowers the energy by $36.45\times10^{-3}$ Ha. Yet the best available methods, run on up to $1.7$ million core hours, disagree with a standard deviation above $1.3\times10^{-3}$ Ha: comparable to chemical accuracy itself, for a molecule without strong correlation. And the energy is \emph{extensive}: it grows in proportion to the number of atoms, so the relative precision needed shrinks as the molecule grows.}
\onp{\footnotesize Chemical accuracy: $1$ kcal/mol $= 1.6\times10^{-3}$ Ha, \emph{absolute}. Benzene, cc-pVDZ \cite{eriksen2020ground}: $30$ electrons in $108$ orbitals, $\sim10^{35}$ determinants; correlation energy $\approx -0.863$ Ha; the best methods scatter by $1.3\times10^{-3}$ Ha, i.e., by about chemical accuracy. Energy grows with the number of atoms.}
\end{frame}

\begin{frame}\ft{From a molecule to a matrix}
\ona{As in \Chap~\chref{C.Motivation}, one discretises the space in which the electrons move by choosing a finite set of functions, the \emph{spin orbitals}; their number $L$ is proportional to the number of atoms. In the language of quantum computing:}
\begin{itemize}\setlength\itemsep{0.25em}
    \item \textbf{State space.} Each spin orbital is either empty or occupied by one electron, so a \emph{configuration} of the electrons is a bit string of length $L$, and a general state is a unit vector in $\C^{2^L}$: one qubit per spin orbital. With a fixed number $N$ of electrons only the $\binom{L}{N}$ strings with $N$ ones matter; chemists call these basis states \emph{Slater determinants}, or simply \emph{determinants}.
    \item \textbf{Hamiltonian.} The energy is a Hermitian matrix $H$ on this space, the \emph{electronic Hamiltonian}. Its entries are integrals over the orbitals, computed classically in polynomial time, and it is a sum of $\mathcal O(L^4)$ terms, each moving at most two electrons (a sum of $\mathcal O(L^4)$ Pauli strings, \Chap~\chref{C.VQE}).
    \item \textbf{Problem.} Find the smallest eigenvalue $E_0$ of $H$ to precision $\epsilon$; this is the \emph{ground-state energy problem}. The corresponding eigenvector $\ket{\Psi_0}$ is the \emph{ground state}.
\end{itemize}
\ona{The matrix is exponentially large but compactly described, which is exactly the situation of \Chap~\chref{C.Motivation}'s MaxCut, where $H_C$ is diagonal and the eigenvector a bit string. Here $H$ is not diagonal, and the eigenvector is a genuine superposition. Some sizes: the nitrogen molecule N$_2$ in a modest basis has $L = 56$ and $N = 14$, hence $\binom{28}{7}^2 \approx 1.4\times10^{12}$ determinants with the right spin; an \emph{active space} (the chemists' term for the subset of orbitals treated accurately, the rest being frozen) of $50$ electrons in $50$ orbitals already has $\binom{50}{25}^2 \approx 1.6\times10^{28}$. Exact diagonalisation, which chemists call \emph{full configuration interaction} (FCI), is feasible up to roughly $10^{10}$ determinants.}
\onp{\footnotesize $L$ spin orbitals $=$ $L$ qubits; $\binom{L}{N}$ determinants (bit strings with $N$ ones); $H$ is a sum of $\mathcal O(L^4)$ terms. N$_2$: $1.4\times10^{12}$ determinants; $50$ electrons in $50$ orbitals: $1.6\times10^{28}$. Exact diagonalisation (FCI): up to $\sim10^{10}$.}
\end{frame}

\begin{frame}\ft{Weak and strong correlation}
\ona{The ground state is a vector $\ket{\Psi_0} = \sum_x c_x \ket x$ over bit strings $x$. What makes chemistry tractable at all is that, for most molecules, this vector is dominated by a single bit string:}
\onp{\footnotesize}
\begin{itemize}\setlength\itemsep{0.25em}
    \item \textbf{Hartree--Fock (HF).} Optimise the orbitals so that the best single determinant has the lowest possible energy.\ona{ This is a \emph{mean-field} method: each electron sees the average of the others. Cost $\mathcal O(L^3)$ to $\mathcal O(L^4)$. The HF state is the classical analogue of a product state, and the difference between its energy and $E_0$ is the \emph{correlation energy}, typically below one percent of the total but far larger than chemical accuracy (\Chap~\chref{C.Motivation}: $0.021$ Ha for H$_2$).}\onp{ Mean field; the analogue of a product state.}
    \item \textbf{Weakly correlated} molecules, the vast majority of organic chemistry, have $|c_{\rm HF}|^2$ close to one\ona{, and the remaining amplitudes are small corrections that can be computed by expansions around HF}.
    \item \textbf{Strongly correlated} systems have many bit strings of comparable weight\ona{, so that no expansion around a single one converges}. \ona{The typical causes are breaking chemical bonds (\Chap~\chref{C.Motivation}: H$_2$ at large $R$), and \emph{transition metals} such as iron, whose partially filled electron shells offer many configurations of nearly equal energy. Several iron atoms coupled together, as in the iron--sulphur clusters below, are the canonical hard case.}\onp{Bond breaking; transition metals such as iron.}
\end{itemize}
\ona{For computer scientists, a useful caricature: HF is to the ground state what the best single cut is to a probability distribution over cuts. The quantity that will decide the question of this \chap{} is the \emph{overlap} $S = |\braket{\Phi|\Psi_0}|$ of a cheap starting state $\Phi$, such as HF, with the true ground state; chemists call $S^2$ the \emph{weight} of $\Phi$.}
\onp{\footnotesize Overlap $S = |\braket{\Phi|\Psi_0}|$ of a cheap state $\Phi$ (e.g.\ HF) with the ground state; $S^2$ is its \emph{weight}. It decides this lecture.}
\end{frame}

\begin{frame}\ft{A glossary of the classical competition}
\ona{The following names recur in the figures of \citet{lee2023evaluating}. None of them is needed in detail; what matters is that they are \emph{heuristics}: polynomial-time algorithms whose accuracy is not guaranteed but can be checked empirically.}
\onp{\scriptsize}
\begin{itemize}\setlength\itemsep{0.15em}
    \item \textbf{Coupled cluster}, e.g.\ \textbf{CCSD(T)}: writes $\ket{\Psi_0} \approx e^{T}\ket{\rm HF}$, with $T$ a sum of single and double electron moves with unknown coefficients (S, D), plus a perturbative estimate of triple moves, (T). Cost $\mathcal O(L^7)$; ``the gold standard'' for weakly correlated molecules. \emph{Local} variants exploit that distant electrons barely interact and cost $\mathcal O(L)$ for large molecules.
    \item \textbf{Size-extensive}: the error grows only linearly with the size, i.e., it is constant per atom. Coupled cluster is; truncated expansions in bit strings are not.
    \item \textbf{DMRG} (density matrix renormalisation group): minimises the energy over \emph{matrix product states}, i.e., vectors whose amplitudes are products of $D\times D$ matrices, one per orbital (\Chap~\chref{C.PerIteration}). The \emph{bond dimension} $D$ controls the accuracy; cost $\mathrm{poly}(L)\,D^3$. Best for chain-like correlation; used for active spaces of $\sim 100$ orbitals.
    \item \textbf{PEPS} (projected entangled pair states): the same idea on two- and three-dimensional lattices.
    \item \textbf{Quantum Monte Carlo}, e.g.\ AFQMC: estimates $E_0$ by random sampling; exact up to the \emph{sign problem}, which it controls by an approximation.
    \item \textbf{Configuration state function} (CSF): a fixed combination of a few determinants with the correct total spin; a slightly better cheap starting state than one determinant.
    \item \textbf{Model Hamiltonians}: the \emph{Heisenberg model} (one spin per lattice site, $H = J\sum_{\langle ij\rangle}(X_iX_j + Y_iY_j + Z_iZ_j)$) and the \emph{Hubbard model} (electrons hopping between neighbouring sites, with an energy penalty $U$ for two on the same site; \emph{half filling} is one electron per site, \emph{$1/8$ doping} removes one in eight, the regime related to high-temperature superconductivity).
\end{itemize}
\end{frame}

\begin{frame}\ft{The benchmark systems: iron--sulphur clusters}
\ona{The molecules usually quoted to motivate quantum computing for chemistry are \emph{iron--sulphur clusters}: groups of iron and sulphur atoms held in proteins, which move electrons around in photosynthesis, respiration and nitrogen fixation. The largest of them, the iron-molybdenum cofactor \emph{FeMo-co} with seven iron atoms and one molybdenum, is the active site of the enzyme nitrogenase, which turns nitrogen from the air into ammonia at room temperature. Its industrial counterpart, the Haber--Bosch process, runs at high temperature and pressure and is often quoted to consume one to two percent of the world's energy. Understanding the mechanism of FeMo-co is therefore a prize, and \citet{reiher2017elucidating} made it the flagship application of fault-tolerant quantum computing. The smaller clusters, [2Fe-2S] and [4Fe-4S], with two and four iron atoms, are the ``hydrogen atoms'' of the field: small enough to be computed classically, and correlated enough to be hard.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item Each iron contributes several nearly degenerate configurations, and the spins of the iron atoms couple: strong correlation, growing with the number of metal centres.
    \item Active spaces: from about $40$ qubits for [2Fe-2S] to $154$ qubits for FeMo-co in the models of \citet{lee2023evaluating}.
\end{itemize}
\ona{We see their structures in Fig.~\ref{fig:ch02:eqa1}(A) below.}
\end{frame}

\begin{frame}\ft{How hard is the ground-state problem, formally?}
\ona{Complexity theory has a precise answer for the worst case. The \emph{local Hamiltonian problem} asks, given a Hermitian matrix $H = \sum_j H_j$ on $n$ qubits whose $\mathrm{poly}(n)$ terms each act on $k$ qubits, and two numbers $a < b$ with $b - a \ge 1/\mathrm{poly}(n)$, whether $E_0 \le a$ or $E_0 \ge b$. It is the quantum analogue of satisfiability:}
\begin{itemize}\setlength\itemsep{0.25em}
    \item It is \textbf{$\mathbf{QMA}$-complete}, already for $k=2$ \cite{kitaev2002classical,kempe2006complexity}. $\mathbf{QMA}$ is the quantum analogue of $\mathbf{NP}$: problems whose ``yes'' answers can be verified efficiently on a quantum computer given a quantum state as a certificate; here, the ground state itself.
    \item The same holds for the electronic Hamiltonians of chemistry with general coefficients \cite{ogorman2022intractability}.
    \item As $\mathbf{NP}$-hard problems are not believed to be solvable in polynomial time classically, $\mathbf{QMA}$-hard problems are not believed to be solvable in polynomial time on a quantum computer.
\end{itemize}
\ona{So quantum computers do not solve ground-state problems efficiently in general, and any exponential advantage must come from the structure of the instances that occur in chemistry. We shall see exactly where the hardness hides in the quantum algorithm: in the overlap $S$ of the starting state.}
\onp{\footnotesize Worst case is hopeless for quantum computers too; an advantage must come from the structure of chemical instances.}
\end{frame}

\section{Quantum phase estimation}

\begin{frame}\ft{Energies as phases}
\ona{Quantum computers do not diagonalise matrices; they apply unitaries. The link is \emph{time evolution}: for a time step $\tau$, the unitary $U = e^{-iH\tau}$ has the same eigenvectors as $H$, and if $H\ket{\Psi_k} = E_k\ket{\Psi_k}$ then}
\begin{equation}\label{eq:ch02:phase}
    U\ket{\Psi_k} = e^{-iE_k\tau}\ket{\Psi_k} = e^{2\pi i\varphi_k}\ket{\Psi_k}, \qquad \varphi_k = -\frac{E_k\tau}{2\pi} \bmod 1.
\end{equation}
\ona{Measuring an energy is therefore the same as measuring the \emph{phase} $\varphi_k \in [0,1)$ of an eigenvalue of a unitary. To avoid ambiguity modulo $1$, the step $\tau$ must be small enough that the whole spectrum fits in one period, $\tau \le 2\pi/(E_{\max} - E_{\min})$; after shifting $H$ by a multiple of the identity, $\tau \approx \pi/\|H\|$. Implementing $U$, or rather its controlled version, costs $\mathrm{poly}(L)$ gates, by Trotter product formulas or more advanced methods \cite{lloyd1996universal,nielsen2002quantum}.}
\ona{The basic primitive is the \emph{Hadamard test}, already met in \Chap~\chref{C.Motivation}: put an extra qubit (an \emph{ancilla}) into $\ket{+} = (\ket0+\ket1)/\sqrt2$, apply $U^{t}$ to the system controlled on the ancilla, and measure the ancilla in the $\ket\pm$ basis. For an eigenstate the controlled-$U^t$ only multiplies the $\ket1$ branch by $e^{2\pi i t\varphi_k}$ (\emph{phase kickback}), and}
\begin{equation}\label{eq:ch02:hadamardtest}
    \mathrm{Pr}[\text{outcome } +] = \tfrac12\big(1 + \cos(2\pi t\varphi_k)\big).
\end{equation}
\ona{Repeating with a few values of the integer power $t$ determines $\varphi_k$, and hence $E_k$.}
\onp{\footnotesize $U = e^{-iH\tau}$: energy $E_k$ becomes phase $\varphi_k$. Hadamard test with controlled-$U^t$: $\mathrm{Pr}[+] = \frac12(1+\cos 2\pi t\varphi_k)$.}
\end{frame}

\begin{frame}\ft{The phase-estimation circuit}
\ona{Textbook quantum phase estimation (QPE) \cite{kitaev1995quantum,nielsen2002quantum} reads out $m$ binary digits of $\varphi_k$ at once. Ancilla $j$ controls $U^{2^{j}}$, $j = 0, \dots, m-1$, so it picks up the phase $2^j\varphi_k$; the inverse quantum Fourier transform turns these phases into the binary expansion of $\varphi_k$, which is measured:}
\begin{center}

\end{tikzpicture}
\end{center}
\ona{\begin{figure}[htb]\caption{Quantum phase estimation with $m = 3$ ancillas. Ancilla $j$ controls $U^{2^j}$; the inverse quantum Fourier transform maps the kicked-back phases to the binary digits of $\varphi$. The system register ends up approximately in the eigenstate whose phase was measured.}\label{fig:ch02:qpe}\end{figure}}
\begin{itemize}\setlength\itemsep{0.2em}
    \item \textbf{Eigenstate input:} the output is the best $m$-bit approximation of $\varphi_k$ with probability $\ge 4/\pi^2$; with $m = b + \lceil\log_2(2 + \frac1{2\delta})\rceil$ ancillas, it is accurate to $b$ bits with probability $\ge 1-\delta$ \cite{nielsen2002quantum}.
    \item \textbf{General input} $\ket\Phi = \sum_k c_k\ket{\Psi_k}$: by linearity, the output is (an approximation of) $\varphi_k$ with probability $|c_k|^2$, and the system register collapses to (approximately) $\ket{\Psi_k}$. The ground-state energy is obtained with probability $|c_0|^2 = S^2$.
    \item \textbf{Cost:} $2^m - 1$ applications of controlled-$U$, i.e., a total evolution time $T = (2^m-1)\tau$.
\end{itemize}
\ona{A variant with a single ancilla, which measures one digit at a time with the Hadamard test \eqref{eq:ch02:hadamardtest}, has the same total cost \cite{kitaev1995quantum}; the post-QPE methods of \citet{lt21} and \citet{zhang2022computing} (\Chap~\chref{C.Motivation}) replace the Fourier transform by classical post-processing, with shorter circuits.}
\end{frame}

\begin{frame}\ft{The Heisenberg limit}
\ona{Put the pieces together. An energy precision $\epsilon$ is a phase precision $\epsilon\tau/2\pi$, i.e., $b = \log_2(2\pi/(\epsilon\tau))$ bits, so the total evolution time is}
\begin{equation}\label{eq:ch02:heisenberg}
    T \;\approx\; 2^b\,\tau \;=\; \frac{2\pi}{\epsilon},
\end{equation}
\ona{times a factor $\mathcal O(1/\delta)$ for failure probability $\delta$, or $\mathcal O(\log 1/\delta)$ with repetitions and a median. This $1/\epsilon$ scaling, the \emph{Heisenberg limit}, is the source of QPE's power. Any method that estimates $\braket{\Phi|H|\Phi}$ or a single interference probability by sampling, including the variational methods of \Chap~\chref{C.VQE}, needs $\mathcal O(1/\epsilon^2)$ repetitions by the central limit theorem; and, more importantly, it measures the energy of the state one has, not the eigenvalue. QPE measures an eigenvalue, whatever the state, provided it can be repeated until the right eigenvalue comes out.}
\onp{\begin{itemize}\item Total evolution time $T \approx 2\pi/\epsilon$: the \emph{Heisenberg limit}. \item Sampling an expectation value (VQE) costs $\mathcal O(1/\epsilon^2)$, and measures the energy of the state one has, not an eigenvalue.\end{itemize}}
\ona{The number of gates is the evolution time, in units of $\tau$, times the cost of one controlled-$U$:}
\begin{equation}\label{eq:ch02:gates}
    G \;\approx\; \underbrace{\frac{2\pi}{\epsilon\,\tau}}_{\approx\,2\|H\|/\epsilon}\;\times\;\underbrace{\mathrm{poly}(L)}_{\text{one step of }U}\;=\;\mathrm{poly}(L)\,\frac{\|H\|}{\epsilon}.
\end{equation}
\ona{In the most efficient methods (\emph{qubitization}), $\|H\|$ is replaced by the sum $\lambda = \sum_j|h_j|$ of the absolute values of the coefficients of $H = \sum_j h_jP_j$ in the Pauli basis, which is $\mathrm{poly}(L)$ as well.}
\end{frame}

\begin{frame}\ft{What QPE requires as the molecule grows}
\ona{The formulas above hide how the required accuracy depends on the size $L$ and on the dimension $2^L$ of the state space. It is worth going through the requirements one by one, because each is sometimes overlooked.}
\onp{\begin{center}\scriptsize
\begin{tabular}{lll}\toprule
requirement & grows with $L$ as & \\\midrule
phase bits $b$ & $\log_2(\|H\|/\epsilon)$ & cheap \\
depth, evolution time & $\|H\|/\epsilon = \mathrm{poly}(L)$ & absolute $\epsilon$, $\|H\|\propto L^{\ge1}$ \\
eigenvalue spacing & $\mathrm{poly}(L)\,2^{-L}$ & need $\epsilon \lesssim \Delta$ to project \\
leakage of tails & $+\,\mathcal O(\log 1/S^2)$ bits & many levels near $E_0$ \\
repetitions & $1/S^2$ & \textbf{exponential if $S$ decays} \\
Trotter steps, coefficients & $\mathrm{poly}(L)$; $\epsilon/L^4$ each & implementation \\
gate error & $\ll 1/G$, $G = \mathrm{poly}(L)/\epsilon$ & error correction \\\bottomrule
\end{tabular}\end{center}}
\ona{\begin{enumerate}\setlength\itemsep{0.25em}
    \item \textbf{Absolute versus relative precision.} Chemical accuracy is an absolute $\epsilon$, while $|E_0|$ and $\|H\|$ grow at least linearly in $L$. The relative precision $\epsilon/\|H\|$ therefore shrinks as $1/\mathrm{poly}(L)$: only $b = \log_2(\|H\|/\epsilon) + \mathcal O(1)$ bits (cheap), but a circuit depth $\|H\|/\epsilon$ that grows polynomially (expensive). If instead a fixed error \emph{per atom} $\bar\epsilon$ suffices, $\epsilon = \bar\epsilon L$ and the depth grows more slowly.
    \item \textbf{Exponentially dense spectrum.} The $2^L$ eigenvalues lie in an interval of width $2\|H\| = \mathrm{poly}(L)$, so their average spacing is $\mathrm{poly}(L)\,2^{-L}$. Resolving a typical eigenvalue would need exponential precision, hence exponential time. QPE works for the ground state only because it sits at the edge of the spectrum, separated by the \emph{gap} $\Delta = E_1 - E_0$. To project onto $\ket{\Psi_0}$, as opposed to only estimating $E_0$, one needs $\epsilon \lesssim \Delta$; in gapped molecules $\Delta$ stays bounded, but it can close as $1/\mathrm{poly}(L)$ or faster (metals, critical systems, nearly degenerate spin states of iron clusters).
    \item \textbf{Leakage from the rest of the spectrum.} The output distribution of QPE for an eigenphase has tails decaying only as the inverse square of the distance. If $S^2$ is small, the other eigenstates carry total weight $1-S^2 \approx 1$, and their tails can produce outcomes near, and even below, $E_0$ more often than the ground state itself does. Suppressing them below $S^2$ costs additional bits, or tapered variants of QPE, with overhead growing in $\log(1/S^2)$; and it makes ``take the smallest outcome'' a biased estimator.
    \item \textbf{Repetitions.} QPE returns $E_0$ with probability $S^2$: expected $1/S^2$ runs, or $1/S$ with amplitude amplification. If $S$ decays exponentially in $L$, so does the running time, whatever the circuit costs.
    \item \textbf{Accuracy of the implementation.} Approximating $e^{-iH\tau}$ by products of simpler exponentials (Trotterisation) introduces an error in the energy that must be kept below $\epsilon$; the number of Trotter steps grows polynomially in $L$ and in $1/\epsilon$. The coefficients of $H$, $\mathcal O(L^4)$ numbers computed classically, must be accurate to about $\epsilon/L^4$ each, since errors add up in $\|H\|$.
    \item \textbf{Accuracy of the hardware.} A circuit of $G$ gates succeeds only if every gate fails with probability $\ll 1/G$. With $G \sim \mathrm{poly}(L)/\epsilon$ in the billions, and physical error rates around $10^{-3}$, this requires quantum error correction with logical error rates of $10^{-10}$ or below; the number of physical qubits per logical qubit grows as $\log^2 G$. This is why QPE is a \emph{fault-tolerant} algorithm.
\end{enumerate}}
\ona{Items 1, 2, 5 and 6 are polynomial or polylogarithmic in $L$: costly, but not exponential. Item 4 is where an exponential can enter, and it is the subject of the rest of this \chap{}.}
\end{frame}

\begin{frame}\ft{The cost of the fault-tolerant algorithm}
\ona{We can now read the accounting of \citet{lee2023evaluating}. The cost of QPE to precision $\epsilon$ has three components: preparing a trial state $\Phi$, at cost $C$; running the phase-estimation circuit; and repeating until the ground state $\Psi_0$, rather than some other eigenstate, is obtained:}
\begin{equation}\label{eq:ch02:qpecost}
    \mathrm{poly}(1/S)\,\big[\mathrm{poly}(L)\,\mathrm{poly}(1/\epsilon) + C\big],\qquad S = |\braket{\Phi|\Psi_0}|,
\end{equation}
\ona{with $\mathrm{poly}(1/S) = 1/S^2$ for QPE.}
\ona{The informal argument for an exponential advantage quotes the $\mathrm{poly}(L)$ of the circuit and assumes $\exp(L)$ classically, i.e., the cost of exact diagonalisation. But the repetition factor depends on $L$ through the quality of state preparation. Worst-case $\mathbf{QMA}$-hardness is consistent with \eqref{eq:ch02:qpecost}: on hard instances, no efficiently preparable $\Phi$ has $S \ge 1/\mathrm{poly}(L)$. Restricting to ``generic'' chemistry amounts to assuming that good state preparation is not exponentially hard because of some unspecified structure. The same structure could also help classical heuristics; for an exponential advantage, state preparation must be exponentially cheaper than the classical solution. Post-QPE algorithms with lower depth, such as the Fourier methods of \Chap~\chref{C.Motivation}, change constants and the $\epsilon$-dependence but keep the $1/S^2$ factor \cite{lt21,zhang2022computing}.}
\onp{The $\mathrm{poly}(L)$ is the circuit; the $\mathrm{poly}(1/S)$ is the number of repetitions and depends on $L$ through the trial state. $\mathbf{QMA}$-hardness hides in $S$. Post-QPE methods keep the $1/S^2$ \cite{lt21,zhang2022computing}.}
\end{frame}

\section{The hypothesis}

\begin{frame}\ft{What is claimed}
\ona{It is therefore often stated, explicitly or implicitly, that quantum computers offer an exponential advantage for the ground-state energy problem. \citet{lee2023evaluating} make the claim precise and then examine the evidence:}
\begin{center}
\emph{The exponential quantum advantage (EQA) hypothesis:} for a large set of relevant (``generic'') chemical problems, the ground-state energy can be computed exponentially faster, as a function of system size, on a quantum than on a classical computer.
\end{center}
\ona{Because computing ground states of even simple Hamiltonians is $\mathbf{QMA}$-hard, the hypothesis cannot be about worst cases. Its specific content is that \emph{generic} chemistry involves Hamiltonians that are (i) polynomially easy for quantum algorithms, in particular for ground-state preparation, and (ii) simultaneously exponentially hard classically, even for the best classical heuristics. Both halves must hold on the \emph{same} instances, and neither can be proven or disproven rigorously, since ``generic chemistry'' is not a defined set; the question is whether EQA should be considered likely in typical cases of interest, or very fine-tuned. \citet{lee2023evaluating} restrict themselves to fault-tolerant algorithms, the most favourable setting for quantum computing, with no noise.}
\onp{\begin{itemize}\item EQA: generic chemistry is (i) polynomially easy quantumly, incl.\ state preparation, \emph{and} (ii) exponentially hard for every classical heuristic, on the same instances. \item Not provable either way (``generic'' is undefined): the question is likely versus fine-tuned \cite{lee2023evaluating}.\end{itemize}}
\end{frame}

\section{Evaluating the evidence}

\begin{frame}\ft{Half (i): ansatz state preparation and the orthogonality catastrophe}
\ona{The simplest way to obtain $\Phi$ is to prepare a state given by a classical \emph{ansatz} (a guessed functional form), e.g., the Hartree--Fock determinant, which is a single bit string after a change of orbital basis, and hope for good overlap. It works for small molecules. But EQA is a statement about large $L$, and there a simple argument, the \emph{orthogonality catastrophe}, suggests trouble: for $K$ weakly interacting subsystems, the ground state is approximately a product, and if the ansatz has overlap $s < 1$ with each local ground state, the global overlap is $s^K$, exponentially small. For instance, $s^2 = 0.99$ per subsystem gives $S^2 < 10^{-6}$ once $K > 1375$.}
\ona{This does not by itself rule out ansatz preparation: neither the ansatz nor the true state need have product form, and local overlaps can be improved as $L$ grows to reach $1/\mathrm{poly}(L)$ globally. The relevant point for EQA is different: if classical heuristics can efficiently prepare states with such good overlap, they may also efficiently obtain the energy to the desired precision. \citet{lee2023evaluating} quantify this on the iron--sulphur clusters, with up to 154 qubits, using DMRG as the reference for $\Psi_0$:}
\figslide[height=0.4\textheight]{figures/eqa/fig1.pdf}{(A) The iron--sulphur clusters [2Fe-2S], [4Fe-4S], the P-cluster and FeMo-co (iron orange, sulphur yellow). (B) Weight $S^2$ of the best single determinant (purple) and best configuration state function (orange) in the ground state versus the number of metal centres: exponential decay, $\sim 10^{-7}$ for FeMo-co \cite{lee2023evaluating}.}{fig:ch02:eqa1}
\ona{Is the decay fundamental? For the [2Fe-2S] fragment, improving the local overlap to infidelity $I$ (i.e., $1 - s^2 = I$) with a classical method costs $\mathrm{poly}(1/I)$, but the energy error of that method is then already $\mathrm{poly}(I)$: improving the overlap enough for efficient state preparation also solves the problem classically. What would be needed for EQA, and is missing, is a heuristic that efficiently yields $1/\mathrm{poly}(L)$ overlap yet cannot reach the target precision in the energy.}
\onp{\footnotesize Local overlap $s$ $\Rightarrow$ global $s^K$. Improving it classically also solves the problem classically.}
\end{frame}

\begin{frame}\ft{Half (i), continued: adiabatic state preparation}
\ona{Alternatively, prepare $\Phi$ on the quantum computer itself by \emph{adiabatic state preparation} (ASP; \Chap~\chref{C.Annealing}): start from the ground state of a simple Hamiltonian $H(0)$, e.g., the one whose ground state is HF, and change the Hamiltonian slowly along a path $H(s)$, $s \in [0,1]$, to the target $H(1) = H$. The adiabatic theorem guarantees that the state follows the instantaneous ground state if the change is slow compared with the gap $\Delta(s)$ between the two lowest eigenvalues along the path. If the minimal gap is $\ge 1/\mathrm{poly}(L)$ (a \emph{protected} gap), ASP plus QPE is polynomial; but nothing guarantees such a gap for a given path, and a closing gap is exactly how $\mathbf{QMA}$-hardness would manifest itself, so ASP is itself a heuristic. On a 24-qubit [2Fe-2S] model, \citet{lee2023evaluating} estimate the time needed as $T_{\rm ASP} \sim \max_s|\braket{\Upsilon_0(s)|dH/ds|\Upsilon_1(s)}|/\Delta(s)^2$, with $\Upsilon_0(s), \Upsilon_1(s)$ the two lowest eigenvectors of $H(s)$, validated against simulations of the time evolution, for two families of initial Hamiltonians:}
\figslide[height=0.4\textheight]{figures/eqa/fig2.pdf}{Adiabatic state preparation for a [2Fe-2S] model \cite{lee2023evaluating}. (A) Structure and active space. (B) The adiabatic estimate matches simulated ASP times up to $O(1)$. (C) Estimated ASP time for mean-field and for interacting initial Hamiltonians against the weight of the initial ground state in the final one: eight orders of magnitude, scaling as $\mathrm{poly}(1/|\braket{\Upsilon_0|\Psi_0}|)$, reminiscent of unstructured search.}{fig:ch02:eqa2}
\ona{The lowest-energy mean-field start gives $T_{\rm ASP} > T_{\rm QPE}$, i.e., preparing the state takes longer than running phase estimation on it; among interacting starts, 20 of the 24 qubits must be included in the exactly diagonalised initial Hamiltonian before $T_{\rm ASP} < T_{\rm QPE}$. A good starting point again comes from classical heuristics, which must then be exponentially better at starting than at solving.}
\onp{\footnotesize ASP time $\sim\mathrm{poly}(1/S)$ of the initial state: again the overlap decides.}
\end{frame}

\begin{frame}\ft{But annealing can be exponentially faster: Somma, Nagaj, Kieferov\'a}
\ona{The estimate $T_{\rm ASP} \sim 1/\Delta^2$ used above is a \emph{sufficient} condition for adiabatic evolution, not a necessary one. That the difference can be exponential was shown by \citet{nagaj2012quantum}, in \emph{Quantum Speedup by Quantum Annealing}, which we study in full in \Chap~\chref{C.Annealing}:}
\begin{itemize}\setlength\itemsep{0.25em}
    \item On the \emph{glued-trees} problem, an oracle problem on a graph with $2^{\mathcal O(n)}$ vertices, every classical algorithm needs exponentially many oracle queries \cite{childs2003exponential}.
    \item Quantum annealing along a suitable path of \emph{stoquastic} Hamiltonians (those without a sign problem, the most ``classical-looking'' ones) finds the exit in $\mathrm{poly}(n)$ time: an \emph{exponential} speedup over any classical algorithm.
    \item Remarkably, the minimum gap along the path is exponentially small. The schedule does not avoid it; it \emph{exploits} it: the state jumps to the first excited level at one avoided crossing and back down at a symmetric one, while the third level stays polynomially far.
\end{itemize}
\ona{What does this mean for EQA? On the positive side, there is no general obstruction to exponentially fast quantum state preparation; the gap-based estimates can be far too pessimistic, and diabatic shortcuts can beat them. On the negative side, the speedup is relative to an oracle, and it rests on a special symmetry (the evolution stays in a subspace of dimension $2n+2$) that has no known counterpart in molecules. It is thus a proof of principle that exponential speedups in state preparation exist, and, in the terminology of \citet{lee2023evaluating}, an example of a very fine-tuned one. Finding chemically relevant Hamiltonians with such structure is open.}
\onp{\footnotesize Exponential speedup exists for annealing, even with an exponentially small gap, but on an oracle problem with special symmetry: a proof of principle, not generic chemistry (\Chap~\chref{C.Annealing}).}
\end{frame}

\begin{frame}\ft{Half (ii): the scaling of classical heuristics}
\ona{The second half of EQA assumes $\exp(L)$ cost for the best classical heuristics at fixed error $\epsilon$ (or fixed error per unit, $\bar\epsilon$). The critical questions: do the heuristics in their domain of applicability cost $\exp(L)$; does the patchwork of heuristics cover chemical space; and do gaps in coverage require exponential methods? Because heuristics run at $\mathrm{poly}(L)$ cost without a guaranteed accuracy, one must measure how their cost depends on the precision as well: a cost $\mathrm{poly}(L)\exp(1/\bar\epsilon)$ would be $\exp(L)$ at fixed total error $\epsilon = \bar\epsilon L$.}
\figslide[height=0.42\textheight]{figures/eqa/fig3.pdf}{Classical heuristics for molecules \cite{lee2023evaluating}. (A) N$_2$: coupled-cluster error versus a time metric, $\mathrm{poly}(1/\epsilon)$. (B) Local (reduced-scaling) CCSD(T) on $n$-alkanes, chains of carbon atoms with hydrogens, up to C$_{120}$H$_{242}$: near-linear cost, constant error per carbon. (C) A fragment of photosystem II, the protein that splits water in photosynthesis, with thousands of electrons on a few processor cores.}{fig:ch02:eqa3}
\ona{For weakly correlated (``single-reference'') molecules, coupled cluster is size-extensive with $\mathrm{poly}(1/\epsilon)$ precision cost, and local variants cost $\mathcal O(L)$ on gapped systems; the conjectured complexity is $\mathrm{poly}(L)\,\mathrm{poly}(1/\epsilon)$.}
\onp{\footnotesize Weak correlation: coupled cluster, $\mathrm{poly}(L)\,\mathrm{poly}(1/\epsilon)$, near-linear with locality.}
\end{frame}

\begin{frame}\ft{Half (ii), continued: strongly correlated models}
\ona{For strong correlation, the evidence of \citet{lee2023evaluating} comes from lattice models, the Heisenberg and Hubbard models of the glossary. These are very special cases. Their interactions act only between neighbouring sites of a regular lattice, so they have $\mathcal O(L)$ terms rather than the $\mathcal O(L^4)$ of an ab initio Hamiltonian; they are translation invariant, so they are described by one or two parameters, and a tensor network can repeat the same few tensors across the lattice; and their ground states have little entanglement across any cut, as is believed to hold for such local, regular models. This is exactly the structure that tensor networks were designed to exploit: matrix product states and PEPS are efficient precisely because they represent low-entanglement states of local lattice Hamiltonians compactly. Choosing these models as the test cases for strong correlation therefore flatters the classical methods.}
\figslide[height=0.45\textheight]{figures/eqa/fig4.pdf}{Classical heuristics for correlated materials \cite{lee2023evaluating}. (A) PEPS tensor networks on the 3D Heisenberg cube from $3^3$ to $10^3$ sites at bond dimension $D = 4$: constant relative error per site and near-linear time. (B) 2D Hubbard model at half filling and at the hard $1/8$ doping on $4\times4$ to $16\times4$ lattices: $1/\bar\epsilon \sim \mathrm{poly}(D)$ with weak dependence on $L$.}{fig:ch02:eqa4}
\ona{Tensor networks with bond dimension $D$ cost $\mathrm{poly}(L)\,\mathrm{poly}(D)$ per iteration; with the observed $\bar\epsilon \sim 1/\mathrm{poly}(D)$ the empirical cost is $\mathrm{poly}(L)\,\mathrm{poly}(1/\bar\epsilon)$. This observation does not transfer to \emph{ab initio} Hamiltonians, where the Coulomb interaction couples all orbitals with one another, there is no lattice and no translation invariance, and nothing guarantees a low-entanglement ground state. How classical heuristics scale there, for strongly correlated molecules such as the iron--sulphur clusters, is not settled by these data: many such problems remain out of reach classically, and whether the barrier is polynomial or exponential is open.}
\onp{\footnotesize Heisenberg and Hubbard models are very special: local, translation invariant, $\mathcal O(L)$ terms, low entanglement, i.e., the home ground of tensor networks. Empirically $\mathrm{poly}(L)\,\mathrm{poly}(1/\bar\epsilon)$ there; this flatters classical methods and says little about ab initio Hamiltonians.}
\end{frame}

\begin{frame}\ft{Conclusions and frequently asked questions}
\ona{\citet{lee2023evaluating} conclude: heuristics are necessary for efficient quantum state preparation; and since EQA is about the \emph{ratio} of costs, the same structure that makes classical heuristics work likely makes state preparation efficient, without yielding EQA. Numerics are not proofs of asymptotics, and EQA in specific problems is not excluded; identifying a relevant chemical system with strong evidence for it is open. Their FAQ:}
\onp{\footnotesize}
\begin{itemize}\setlength\itemsep{0.2em}
    \item \emph{Does this mean quantum computers are useless for ground states?} No: polynomial or large constant-factor advantages can be very useful; but the degree of polynomial advantage should be compared with that available in other applications.
    \item \emph{Classical heuristics fail on a problem and exact methods are exponential; is that EQA?} Not by itself: one must show that reaching the desired precision classically \emph{requires} exponential effort, and that the quantum algorithm reaches it without exponential effort (state preparation included).
    \item \emph{Improvements to quantum algorithms?} They must change the ratio of quantum to classical cost; introducing locality into a quantum heuristic, for instance, usually helps the classical heuristic too.
    \item \emph{Other tasks, e.g., dynamics?} Nothing follows from this analysis, but heuristics should be considered carefully there as well.
\end{itemize}
\end{frame}

\begin{frame}\ft{What can be guaranteed given the overlap}
\ona{The overlap is also where the positive results live. \citet{zhang2022computing} show that if one \emph{is} given a state with overlap $\eta = S^2$ and a Hamiltonian with gap $\gamma$, then not only the ground-state energy \cite{lt21} but any ground-state property $\braket{\psi_0|O|\psi_0}$ can be estimated to accuracy $\epsilon$ with maximal evolution time $\widetilde{\mathcal O}(\gamma^{-1})$ and total evolution time $\widetilde{\mathcal O}(\gamma^{-1}\epsilon^{-2}\eta^{-2})$, using one ancilla; see \Chap~\chref{C.Motivation}. Two remarks:}
\begin{itemize}\setlength\itemsep{0.3em}
    \item The guarantee is conditional on $\eta$, and the cost scales as $\eta^{-2}$, exactly the $\mathrm{poly}(1/S)$ of \eqref{eq:ch02:qpecost}: the EQA question is precisely whether $\eta$ stays polynomial along families of instances.
    \item Even with the assumptions, the task is not classically easy: estimating ground-state properties to high precision classically would imply $\mathbf P = \mathbf{BQP}$; and without the assumptions, the decision version is $\mathsf{P^{QMA[\log]}}$-complete, harder than energy estimation. So the \emph{form} of a possible advantage is clear: polynomial cost given a good trial state, on instances where classical heuristics do not converge. Its existence for natural families is what remains unproven.
\end{itemize}
\end{frame}

\section{Heuristics, formally, and what this means for optimization}

\begin{frame}\ft{Heuristics, formally}
\ona{The same tension appears in optimization, where the survey \cite{abbas2024challenges} makes the point precisely. Heuristic algorithms are algorithms without provable performance guarantees, run-time guarantees, or both. One way to make claims about them precise is distributional: a pair $(L,D)$ of a language and a family of input distributions is in $\mathbf{HeurBPP}$ if a polynomial-time randomized algorithm $\mathcal A$ satisfies, for all $n$ and $\delta > 0$,}
\begin{equation*}
    \mathrm{Pr}_{x \sim D_n}\Big[\mathrm{Pr}\big[\mathcal{A}(x,0^{1/\delta}) = L(x)\big] \ge 2/3\Big] \ge 1-\delta,
\end{equation*}
\ona{i.e., it may err on a small fraction of instances; $\mathbf{HeurBQP}$ is the quantum analogue. (Here $L$ denotes a language, not the number of orbitals.)}
\begin{itemize}\setlength\itemsep{0.2em}
    \item Super-polynomial separations between the two are known only for contrived distributions, essentially Karp-reduction images of factoring; they have no practical value.
    \item The overlap gap property makes some random constraint satisfaction problems average-case hard for large classes of classical \emph{and} low-depth quantum algorithms (\Chap~\chref{C.Inapprox}); whether deeper quantum algorithms escape is open.
\end{itemize}
\end{frame}

\begin{frame}\ft{What complexity theory does and does not tell us}
\ona{It is widely believed that quantum computers will not offer exponential speedups for $\mathbf{NPO}$-hard problems in the worst case. But in practice one rarely needs the worst case or an exact solution; approximations are acceptable, and the hardness of \emph{approximation} with quantum computers is largely unexplored. The survey's summary applies to chemistry and optimization alike:}
\begin{center}
    \textit{Understanding complexity theory is extremely useful for gauging a possible quantum advantage, but rigorous complexity-theoretic separations are neither necessary nor sufficient when seeking a practical quantum advantage.}
\end{center}
\ona{TSP illustrates the nuance: its decision version is $\mathbf{NP}$-hard and the optimization version $\mathbf{APX}$-hard; Euclidean TSP has a polynomial-time approximation scheme; heuristics solve instances with millions of nodes to near-optimality; and shortest path is in $\mathbf{P}$. In contrast, there exist problems with fewer than 100 variables that are hard classically, such as low-autocorrelation binary sequences. The glued trees of \citet{nagaj2012quantum} show the other side: a rigorous exponential separation exists, but for a problem nobody needs to solve.}
\end{frame}

\begin{frame}\ft{Chemistry and optimization side by side}
\ona{The analysis of \citet{lee2023evaluating} translates almost word for word to the optimization problems of this course:}
\begin{center}\ona{\small}\onp{\scriptsize}
\begin{tabular}{lll}\toprule
 & ground-state chemistry & combinatorial optimization \\\midrule
problem & smallest eigenvalue of $H$ & $\min_x f(x)$, i.e., of diagonal $H_C$ \\
worst case & $\mathbf{QMA}$-hard & $\mathbf{NP}$-hard, often $\mathbf{APX}$-hard \\
trial state & HF, CSF, DMRG state & uniform, warm start (GW cut) \\
overlap $S^2$ & weight of $\Phi$ in $\Psi_0$ & $\mathrm{Pr}[\text{sample is optimal}]$ \\
quantum route & QPE, ASP, VQE & Grover-type search, annealing, QAOA \\
classical baseline & coupled cluster, DMRG, AFQMC & GW; Gurobi, BiqBin, MADAM \\
proven exp.\ speedup & none for generic chemistry & oracle problems only (glued trees) \\\bottomrule
\end{tabular}
\end{center}
\ona{For a diagonal $H_C$, phase estimation on the uniform superposition finds the optimum with probability $S^2 = (\text{number of optima})/2^n$, and amplitude amplification gives the quadratic Grover speedup $1/S$: the orthogonality catastrophe in its purest form.}
\end{frame}

\begin{frame}\ft{Lessons for variational methods}
\onp{\footnotesize}
\begin{enumerate}\setlength\itemsep{0.3em}
    \item \textbf{Claims must be about instance families}, and the classical baseline must be both the best available polynomial-time algorithm and the best classical solvers, not brute force. For MaxCut, these are the Goemans--Williamson algorithm (\Chap~\chref{C.Motivation}) and solvers such as Gurobi, BiqBin and MADAM.
    \item \textbf{State preparation is the crux.} In chemistry it is the overlap $S$; in QAOA it is the initial state and the angles. Warm starts (\Chaps~\chref{C.WSRounded}--\chref{C.WSContinuous}) are the optimization analogue of a good trial state, and variational methods (\Chap~\chref{C.VQE}) are one way to supply trial states for QPE.
    \item \textbf{Mind the precision.} QPE reaches the Heisenberg limit $1/\epsilon$, but needs fault tolerance; variational methods pay $1/\epsilon^2$ in samples and measure the energy of the state they have.
    \item \textbf{Gap estimates can be pessimistic.} Annealing and QAOA can exploit, not merely survive, small gaps (\Chaps~\chref{C.Annealing}--\chref{C.QAOA}).
    \item \textbf{Benchmark empirically and fairly}: model-independent problems, time-to-solution and cost, distributions over instances, and validation on hardware (\Chap~\chref{C.PerIteration}).
    \item \textbf{Polynomial advantages are worth having}, and small improvements matter in applications such as finance (\Chap~\chref{C.Portfolio}); but they must be measured, not assumed.
\end{enumerate}
\end{frame}

\begin{frame}\ft{Exercises}
\begin{exe}[Precision of phase estimation]
Consider a molecule with $\|H\| \approx 250$ Ha (an assumed value, for illustration) and chemical accuracy $\epsilon = 1.6\times10^{-3}$ Ha. (a) How many bits $b$ of the phase are needed? (b) How many applications of controlled-$U$ with $\tau = \pi/\|H\|$ does textbook QPE need, ignoring the $\delta$-dependent bits? (c) How do the answers change for a molecule ten times larger, if the target is (i) chemical accuracy in total, (ii) chemical accuracy per unit of the original size?
\end{exe}
\end{frame}

\begin{frame}\ft{Exercises, continued}
\begin{exe}[Orthogonality catastrophe]
A molecule consists of $K$ identical, non-interacting fragments, and the Hartree--Fock state has weight $s^2$ in each fragment's ground state. (a) Show that the global weight is $S^2 = s^{2K}$. (b) For $s^2 = 0.99$ and $s^2 = 0.9$, for which $K$ is the expected number of QPE repetitions larger than $10^6$? (c) Explain why improving $s$ with a classical method may make QPE unnecessary.
\end{exe}
\begin{exe}[Leakage]
For an eigenphase $\varphi$ that is not an $m$-bit number, the probability that textbook QPE outputs the $m$-bit number at distance $j/2^m$ from $\varphi$ is at most $1/(4j^2)$ for $j\ge1$ (up to rounding). Suppose $S^2 = 10^{-4}$ and all the other weight sits in one eigenstate $10$ bins above the ground state. Estimate how often QPE returns an outcome at or below the ground-state bin coming from the wrong eigenstate, compared with the correct one.
\end{exe}
\end{frame}
