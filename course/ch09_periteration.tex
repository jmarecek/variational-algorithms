\chapter{Per-Iteration Complexity of QAOA}\label{C.PerIteration}

\ona{\Chap~\chref{C.IterationComplexity} counted iterations. This \chap{} asks what one iteration \emph{costs}: copies, circuits, depth, classical simulation and, above all, time. On a device in the cloud, each iteration is a round trip: the circuits travel to the device, wait in its queue, are prepared and executed, and the results travel back, and the next circuits cannot be sent before they have arrived. We take a measured round trip apart, model the queue and derive the waits of a variational method in it, and show how the delays decide how many copies a round should carry and which methods are fast. We then recall from \Chap~\chref{C.VQE} why no per-iteration budget rescues the worst case, since training is $\mathbf{NP}$-hard \citep{bittel2021training}, and follow \citet{marecek2026angles}, \emph{Setting angles in quantum approximate optimization at utility-scale}, where these costs decide how the angles of QAOA are set at 100 qubits or more. The numbers of Sections~\ref{sec:ch09:round}--\ref{sec:ch09:copies} are computed by \texttt{code/ch09\_figures.py}.}

\section{The cost of one iteration}

\begin{frame}\ft{Shots: estimating the energy}
\ona{One evaluation of $F_p(\boldsymbol\theta) = \langle H_C\rangle$ for MaxCut is the average of the cut value $f(x) = \frac12\sum_{(i,j)\in E}w_{ij}(1-z_iz_j)$ over $M$ samples $x$. Since $|f(x)| \le \|w\|_1 = \sum_e|w_e|$, Hoeffding's inequality gives}
\begin{equation}\label{eq:ch09:shots}
    \mathrm{Pr}\big[|\hat F - F_p| \ge \epsilon\big] \le 2\exp\Big(-\frac{M\epsilon^2}{2\|w\|_1^2}\Big),
    \qquad\text{so}\qquad M = \mathcal O\Big(\frac{\|w\|_1^2}{\epsilon^2}\log\frac1\delta\Big)
\end{equation}
\ona{shots suffice for additive error $\epsilon$ with probability $1-\delta$; in terms of approximation ratio, $\epsilon = \eta\cdot\mathrm{OPT}$ with $\mathrm{OPT} \ge \|w\|_1/2$ for positive weights gives $M = \mathcal O(\eta^{-2}\log(1/\delta))$, independent of $n$. The variance $\sigma^2$ of \Chap~\chref{C.IterationComplexity} is $\mathrm{Var}[f(x)]/M$. For general Hamiltonians $H = \sum_kh_kP_k$, terms are grouped into commuting sets and the count becomes $\mathcal O((\sum_k|h_k|)^2/\epsilon^2)$.}
\onp{$M = \mathcal O(\|w\|_1^2\epsilon^{-2}\log(1/\delta))$ shots per evaluation; a relative precision $\eta$ needs $\mathcal O(\eta^{-2})$ shots, independent of $n$; MaxCut is diagonal, one measurement basis.}
\end{frame}

\begin{frame}\ft{Circuits, depth, and time}
\begin{itemize}\setlength\itemsep{0.3em}
    \item \textbf{Circuit evaluations per iteration:} COBYLA $1$, SPSA $2$, parameter-shift gradient $2\cdot 2p$ (\Chap~\chref{C.PSR}); each with $M$ shots.
    \item \textbf{Gates per layer:} $|E|$ two-qubit $ZZ$ rotations and $n$ single-qubit rotations. On hardware with restricted connectivity, each $ZZ$ between non-adjacent qubits costs SWAPs; for a heavy-hex device and a dense problem graph the depth grows by a factor $\mathcal O(n)$, which is why hardware studies use hardware-native (heavy-hex, line-based) graphs.
    \item \textbf{Wall-clock:} on \emph{ibm\_boston}, $t_{\rm shot} \approx 100\,\mu$s and a latency of about $6$~s for each submitted circuit; one iteration of COBYLA with $M = 20\,000$ shots takes about $8$~s, most of it latency\ona{; Section~\ref{sec:ch09:round} takes such a round trip apart}.
    \item \textbf{Total:} $T = N\,(t_{\rm qc} + c\,M\,t_{\rm shot}) + Q\,t_{\rm shot}$ for $N$ iterations of $c$ circuits each and $Q$ final samples, the resource model of the Pareto analysis below; it is \eqref{eq:ch09:round} with $W = 0$ and the latency $t_{\rm qc}$ paid once a round.
\end{itemize}
\end{frame}

\begin{frame}\ft{Classical evaluation of $\langle H_C\rangle$}
\ona{When the loop is run offline, the cost is that of a classical simulator:}
\begin{itemize}\setlength\itemsep{0.3em}
    \item \textbf{Depth one is cheap:} $\braket{Z_iZ_j}$ depends only on the neighbourhoods of $i,j$ and has a closed form; the whole landscape is $\mathcal O(|E|\cdot\deg^2)$ per point.
    \item \textbf{Statevector:} $\mathcal O(2^n)$ memory and $\mathcal O(p(|E|+n)2^n)$ time; 40 qubits took five minutes on 1024 GPUs.
    \item \textbf{Matrix product states} with bond dimension $\chi$: each two-qubit gate costs $\mathcal O(\chi^3)$ after mapping the graph to a line, so $\mathcal O(p\,|E|\,\chi^3)$ per evaluation, polynomial in $n$ but exponential in the entanglement the circuit builds; long-range edges need swaps or a SAT-optimised relabelling. Accuracy is monitored by $F_{\rm approx} = \prod_g\sum_{j\le\chi}\lambda_{j,g}^2$.
    \item \textbf{Pauli propagation:} propagate $H_C$ backwards through the circuit, truncating Pauli strings of weight above $W$ or coefficient below a threshold; cheap for sparse graphs, cost growing fast with density.
    \item \textbf{Empirically}, at $p = 10$ and $\chi = 24$, both scale polynomially in $|E|$, with $100$--$1000$ s per evaluation for a fully connected 100-node graph\ona{, Fig.~\ref{fig:ch09:evaltime}}.
\end{itemize}
\end{frame}

\section{The round trip}\label{sec:ch09:round}

\begin{frame}\ft{A round trip}
\onp{\footnotesize}
\ona{An iteration of a variational method on a device in the cloud is a \emph{round trip}. The classical computer sends circuits to the device; the device queues them, prepares them for execution and executes them; and the results return. The circuits of the next iteration depend on these results, and cannot be sent before the results have returned.}
\begin{defn}\label{defn:ch09:round}
A \emph{round} submits $c$ circuits, with $M$ copies of each, and waits for their results. It takes the time
\begin{equation}\label{eq:ch09:round}
    t_{\rm round} = t_{\rm net}+W+t_{\rm prep}+c\,M\,t_{\rm shot},
\end{equation}
where $t_{\rm net}$ is the time taken to send the circuits and to return the results, $W$ the wait in the queue of the device, $t_{\rm prep}$ the time taken to prepare the circuits for execution, and $t_{\rm shot}$ the time of one copy. A method is \emph{adaptive} if the circuits of each round depend on the results of the earlier rounds.
\end{defn}
\ona{Gradient descent, simultaneous perturbation and COBYLA are adaptive: $K$ iterations take $K$ rounds, one after another, and at least $K(t_{\rm net}+t_{\rm prep})+\sum_kW_k$, however many devices are available. Only the circuits of one round, and the copies of one circuit, can be executed in parallel. We call $t_{\rm lat} = t_{\rm net}+t_{\rm prep}$ the \emph{latency} of a round. It is paid once a round if the $c$ circuits are submitted together, and $c$ times if they are submitted one by one; whether $t_{\rm prep}$ grows with $c$ depends on the device, and we take it to be paid once a round.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item Adaptive methods (gradient descent, simultaneous perturbation, COBYLA): $K$ iterations are $K$ rounds in sequence; only the circuits of a round run in parallel.
    \item The latency $t_{\rm lat} = t_{\rm net}+t_{\rm prep}$: once a round if its circuits are submitted together, $c$ times if one by one.
\end{itemize}
\end{frame}

\section{Waiting in a queue}\label{sec:ch09:queue}

\begin{frame}\ft{A queue and its mean wait}
\onp{\footnotesize}
\ona{A device in the cloud serves many users. We model it as a single server, which executes one job at a time, first come, first served. The jobs of the other users arrive at the times of a Poisson process of rate $\lambda$, independently of one another, and take independent service times distributed as $S$. The \emph{load} $\rho = \lambda\,\mathbb E S$ is the fraction of the time for which the device is busy, and we assume that $\rho<1$, so that the queue has a stationary state \citep{harcholbalter2013performance}.}
\begin{lem}[\citet{little1961proof}]\label{lem:ch09:little}
In a queue in its stationary state, the mean number of jobs waiting is $L_q = \lambda\,\mathbb E W$, where $\mathbb E W$ is the mean wait of a job.
\end{lem}
\begin{propn}[Pollaczek and Khinchine]\label{propn:ch09:pk}
In this queue, in its stationary state, a job that arrives at a time of the Poisson process, or at a time independent of the queue, waits on average
\begin{equation}\label{eq:ch09:pk}
    \mathbb E\,W = \frac{\lambda\,\mathbb E S^2}{2(1-\rho)} = \frac\rho{1-\rho}\cdot\frac{\mathbb E S^2}{2\,\mathbb E S}.
\end{equation}
\end{propn}
\ona{\emph{Proof.} Such a job finds the queue in its stationary state; for the arrivals of a Poisson process, this is the property that Poisson arrivals see time averages \citep[Chapter~13]{harcholbalter2013performance}. Its wait is the work that it finds: the remaining service of the job in service, if any, and the services of the jobs that wait. The device is busy with probability $\rho$, and the job in service is then found at a uniformly random time in its service, a service of length $s$ being found with a probability proportional to $s$; the remaining service has the mean $\mathbb E S^2/(2\,\mathbb E S)$. The services of the $L_q$ waiting jobs do not depend on their number, and have the total mean $L_q\,\mathbb E S = \lambda\,\mathbb E W\,\mathbb E S = \rho\,\mathbb E W$, by Lemma~\ref{lem:ch09:little}. Hence $\mathbb E W = \rho\,\mathbb E S^2/(2\,\mathbb E S)+\rho\,\mathbb E W$. \qed}
\onp{Proof: the work found on arrival is the remaining service, of mean $\rho\,\mathbb ES^2/(2\,\mathbb ES)$, plus $L_q\,\mathbb ES = \rho\,\mathbb EW$ by Little's law.}
\end{frame}

\begin{frame}\ft{The waits of a closed loop}
\onp{\footnotesize}
\ona{A variational method does not submit its rounds at the times of a Poisson process. Each round is submitted as soon as the results of the previous one have returned, and the device has meanwhile served only the work that arrived since.}
\begin{propn}\label{propn:ch09:loop}
In the queue above, let a client submit its first round at a time independent of the queue, and each later round $t_{\rm net}$ after the previous round has been served; let each round occupy the device for $s$, and let $t_c = s+t_{\rm net}$. If the device serves first come, first served, the waits $W_k$ of the rounds satisfy $\mathbb E W_1 = \lambda\,\mathbb E S^2/(2(1-\rho))$ and
\begin{equation}\label{eq:ch09:loop}
    \rho\,(\mathbb E W_k+t_c)-t_{\rm net}\le\mathbb E W_{k+1}\le\rho\,(\mathbb E W_k+t_c),
\end{equation}
so that $\mathbb E W_k\le\rho^{k-1}\,\mathbb E W_1+\rho\,t_c/(1-\rho)$.
\end{propn}
\ona{\emph{Proof.} The first wait is that of Proposition~\ref{propn:ch09:pk}. Let round $k$ be submitted at the time $a_k$, so that $a_{k+1}-a_k = W_k+t_c$. Every job that arrived before $a_k$ is served before round $k$, which ends at $a_k+W_k+s$. The work found at $a_{k+1}$, which is $W_{k+1}$, is therefore that of jobs that arrived in $(a_k,a_{k+1}]$: it is at most their total service $A_k$, and at least $A_k-t_{\rm net}$, since the device can serve them only after round $k$ has ended. The length $W_k+t_c$ of the interval is known at $a_k$, and the arrivals after $a_k$ do not depend on what happened before, so that $\mathbb E[A_k\,|\,W_k] = \lambda\,\mathbb E S\,(W_k+t_c) = \rho\,(W_k+t_c)$. Taking expectations gives \eqref{eq:ch09:loop}, and induction the last bound. \qed}
\onp{Proof: round $k+1$ finds only the work that arrived during $(a_k,a_{k+1}]$, of mean $\rho\,(W_k+t_c)$, less at most $t_{\rm net}$ served meanwhile.}
\end{frame}

\begin{frame}\ft{Sessions}
\onp{\footnotesize}
\ona{A \emph{session} holds the device from the first round of a client to the last. For the client, it removes the waits after the first: at $\rho = 0.8$, the $80$ rounds take $24$ minutes on average in a session, against $111$ minutes first come, first served. \section{Copies and rounds}\label{sec:ch09:copies}

\begin{frame}\ft{How many copies a round}
\onp{\footnotesize}
\ona{\Chap~\chref{C.IterationComplexity} found that, with $N$ copies in all, many iterations with few copies are as good as few iterations with many, as far as the bounds are concerned \citep{sweke2020stochastic}. Latency changes this. Consider an ideal device and unbiased estimates of the gradient from $c$ circuits with $M$ copies each, whose variance is $V = V_1/M$, and let $L$, $\Delta = C(\boldsymbol\theta_0)-C^*$ and $\epsilon$ be as in \Chap~\chref{C.IterationComplexity}.}
\begin{propn}\label{propn:ch09:copies}
By the corollary on unbiased estimates of \Chap~\chref{C.IterationComplexity},
\begin{equation}\label{eq:ch09:K}
    K(M) = \max\{8L\Delta/\epsilon,\ 32L\Delta V_1/(M\epsilon^2)\}
\end{equation}
rounds give an $\epsilon$-stationary point. If each round takes $t_{\rm lat}+cMt_{\rm shot}$, the time
\begin{equation}\label{eq:ch09:copies}
    T(M) = K(M)\,(t_{\rm lat}+cMt_{\rm shot})
\end{equation}
does not increase with $M$ up to $M^* = 4V_1/\epsilon$, and increases beyond. It is constant up to $M^*$ if $t_{\rm lat} = 0$, and
\begin{equation}\label{eq:ch09:ratio}
    \frac{T(1)}{T(M^*)} = M^*\,\frac{t_{\rm lat}+ct_{\rm shot}}{t_{\rm lat}+cM^*t_{\rm shot}}.
\end{equation}
\end{propn}
\ona{\emph{Proof.} For $M\le M^*$, $K(M) = 32L\Delta V_1/(M\epsilon^2)$, and $T(M) = (32L\Delta V_1/\epsilon^2)(t_{\rm lat}/M+ct_{\rm shot})$, which decreases with $M$ if $t_{\rm lat}>0$ and is constant if $t_{\rm lat} = 0$. For $M\ge M^*$, $K(M) = 8L\Delta/\epsilon$, and $T(M)$ increases with $M$. Since $K(1) = M^*K(M^*)$, \eqref{eq:ch09:ratio} follows. \qed}

\ona{Without latency, only the total number of copies counts, up to $M^*$. With latency, a round should carry $M^*$ copies, the most that still reduce the number of rounds; beyond $M^*$ the bound needs $8L\Delta/\epsilon$ rounds whatever $M$, as gradient descent with exact gradients does. When $t_{\rm lat}$ dominates $cM^*t_{\rm shot}$, the ratio \eqref{eq:ch09:ratio} is about $M^*$. In the worked example of \Chap~\chref{C.VQE}, the gate-by-gate shift rule has $c = 16$ and $V_1 = 10\sigma^2 = 20$, and $L = 20$ and $\Delta = 0.923$; with $\epsilon = 0.5$, $M^* = 160$ and $K(M^*) = 295$, and with $t_{\rm lat} = 6$~s the ratio is $153$.}
\onp{\begin{itemize}\setlength\itemsep{0.2em}
    \item Without latency only the total of copies counts, up to $M^*$ (\Chap~\chref{C.IterationComplexity}); with latency a round should carry $M^*$ copies.
    \item Worked example: $c = 16$, $V_1 = 20$, $L = 20$, $\Delta = 0.923$, $\epsilon = 0.5$: $M^* = 160$, $K(M^*) = 295$; with $6$~s a round, $T(1)/T(M^*) = 153$.
\end{itemize}}
\end{frame}

\ona{\citet{menickelly2023latency} design stochastic optimizers that take fewer rounds with more copies each, and choose between them by the ratio of $t_{\rm lat}$ to $t_{\rm shot}$; \citet{ha2025two} estimate the variance and allocate the copies under a budget of time. \citet{karalekas2020quantum} reduce $t_{\rm prep}$ by compiling a parametrised circuit once and updating only its angles on the control system, and \citet{wack2021quality} measure the speed of a device by the circuit layers it executes per second, including the updates of parameters between circuits.}

\section{No budget rescues the worst case: training is NP-hard}

\begin{frame}\ft{No budget rescues the worst case}
\ona{Per-iteration costs are polynomial. Iteration counts, by \Chap~\chref{C.IterationComplexity}, are polynomial to reach a \emph{stationary} point. What no polynomial budget buys is the \emph{global} optimum, in the worst case. \Chap~\chref{C.VQE} proved this, following \citet{bittel2021training}:}
\begin{itemize}\setlength\itemsep{0.25em}
    \item Training is $\mathbf{NP}$-hard on $\mathcal O(\log d)$ qubits, with a single layer, and for QAOA with a single layer. On these instances one evaluation of the energy is a small matrix computation, so the cost of an iteration is negligible, and the difficulty is the number of iterations needed to find the global optimum.
    \item Unless $\mathbf P = \mathbf{NP}$, no polynomial-time training procedure guarantees an optimization error below $(1-\alpha_{\max})/2$ of the spectral width for QAOA, that is $1/34$, or $0.061$ under the Unique Games Conjecture; methods that may stop at a local minimum do no better than $1/4$, as the local minima of the landscape include the local optima of single-vertex search.
\end{itemize}
\ona{What the theorem does \emph{not} say: it does not forbid good angles for typical instances, which is what the rest of this \chap{} finds empirically, nor good solutions from local minima, which is what warm starts guarantee (\Chaps~\chref{C.WSRounded}--\chref{C.WSContinuous}). It says that a per-iteration budget, however generous, is not the bottleneck in the worst case: the number of iterations to the global optimum is.}
\onp{Not forbidden: good angles on typical instances (empirically, below); good solutions from local minima (warm starts). Forbidden: a worst-case guarantee from any polynomial budget (\Chap~\chref{C.VQE}).}
\end{frame}

\section{Angle setting at utility scale}

\begin{frame}\ft{Three ways to set angles without a closed loop}
\ona{Recall the QAOA state $\ket{\psi(\boldsymbol\theta)} = \prod_{k=1}^p U_M(\beta_k)U_C(\gamma_k)\ket{+}^{\otimes n}$ and its energy $\langle H_C\rangle$. Closed-loop optimization of $\boldsymbol\theta$ on hardware is (i) slow, given clock speeds and queueing; (ii) impossible to check, since at utility scale $\langle H_C\rangle$ cannot be simulated exactly; and (iii) \textbf{NP}-hard in the worst case, with many local optima even at $p=1$. Figure~\ref{fig:ch09:strategies} shows the alternatives:}
\figslide[height=0.5\textheight]{figures/numerics/figure1.pdf}{Angle setting at utility scale \citep{marecek2026angles}. Angles are found classically, by evaluating the energy on (a) representative structured problems, (b) small representative instances with exact energy, or (c) the utility-scale instance with an approximate energy; they are transferred to the QAOA circuit (e) for the utility-scale instance (d), and only the final sampling (f) runs on hardware.}{fig:ch09:strategies}
\end{frame}

\begin{frame}\ft{A taxonomy of angle-setting methods}
\ona{Some 30 methods from the literature fall into four classes, with many methods belonging to several:}
\begin{description}\setlength\itemsep{0.2em}
    \item[Physics-inspired] angles follow an annealing schedule: Trotterised quantum annealing (TQA), linear ramps with two slopes $(\Delta_\beta,\Delta_\gamma)$, spectral-gap-informed ramps, counterdiabatic and feedback-based (FALQON) schedules. No energy evaluation needed unless the slopes are optimised.
    \item[Parameter transfer] optimal angles concentrate across similar instances: fixed angles for MaxCut on $k$-regular graphs \citep{Wurtz2021}, transfer from small to large instances, rescaling for weighted problems, surrogate models. Require a database of good angles.
    \item[Iterative] initialise depth $p+1$ from depth $p$: Interp.\ and Fourier \citep{Zhou2020}, transition states, basin hopping. Smooth schedules, good quality, but repeated optimizations.
    \item[Machine learning] predict angles from instances: clustering, regression from $p=1$ to $p>1$, reinforcement learning, diffusion models, graph attention networks. Only as good as their training data.
\end{description}
\ona{Superscripts in what follows: $\dagger$ no optimization to the instance, no superscript only method parameters (e.g., slopes) optimised, $\star$ every angle optimised, always with COBYLA.}
\end{frame}

\begin{frame}\ft{Evaluating the energy when you cannot simulate}
\ona{If a method needs $\langle H_C\rangle$, one of four evaluators supplies it:}
\begin{itemize}\setlength\itemsep{0.2em}
    \item \textbf{Statevector (SV):} exact, up to $\sim 40$ qubits on a GPU cluster (a 40-qubit run takes five minutes on 1024 A100 GPUs), $\sim 20$ on a laptop.
    \item \textbf{Hardware:} sampling the QPU; slow and noisy, but the ground truth for what the QPU will do.
    \item \textbf{Matrix product states (MPS):} tensor-network approximation controlled by the bond dimension; accurate for low-connectivity graphs without short cycles, especially after remapping the problem graph onto the tensor line (a SAT-solved mapping). Tends to \emph{underestimate} the energy, yet still guides the optimizer well.
    \item \textbf{Pauli propagation (PP):} Heisenberg-picture propagation of $H_C$ truncated by Pauli weight $W$ and coefficient threshold; accurate for sparse graphs, expensive for dense ones.
\end{itemize}
\ona{Crucially, methods are only comparable under the same evaluator, and a good approximate energy is not the same as a good hardware energy.}
\end{frame}

\section{Benchmarks}

\begin{frame}\ft{Methodology}
\begin{itemize}\setlength\itemsep{0.2em}
    \item \textbf{Methods:} Interp., Fourier, TQA, recursive transition states (RTS), linear ramps, fixed angles \citep{Wurtz2021}, parameter transfer; 34 configurations, 79\,908 training chains.
    \item \textbf{Problems:} MaxCut and Maximum Independent Set (MIS) on Erd\H{o}s--R\'enyi (ER), heavy-hex (HH, the IBM qubit topology), line-based (LB, a line plus SWAP layers) and random-regular (RR) graphs; small set (20--21 nodes) for SV, large set (50--144 nodes) for MPS and PP; plus low-autocorrelation binary sequences (LABS) up to 21 spins.
    \item \textbf{Depths:} $p \le 10$, since noise limits what can be executed.
    \item \textbf{Metrics:} energy or approximation ratio (against CPLEX optima), \emph{and} the wall-clock cost of finding the angles.
    \item \textbf{Validation:} on \emph{ibm\_boston} and \emph{ibm\_fez} for 144-node heavy-hex and 100-node line-based instances, and a Kruskal--Wallis test with Conover--Iman post-hoc comparisons to decide which methods are statistically distinguishable.
\end{itemize}
\end{frame}

\section{Operational guidance}

\begin{frame}\ft{What a practitioner should do}
\onp{\footnotesize}
\begin{enumerate}\setlength\itemsep{0.2em}
    \item \textbf{Train offline with approximate evaluators} (MPS after remapping the graph to the tensor line; PP for sparse graphs), but \textbf{validate on the QPU}: a method that looks good under an approximate energy may underperform on hardware.
    \item \textbf{Start from transferred or fixed angles.} They cost an $\mathcal O(1)$ look-up, perform well on the problems they were designed for, and are good initial points elsewhere; tailoring them adds $1$--$2\%$ at large cost.
    \item \textbf{Prefer linear ramps} for rugged, higher-order problems (LABS); prefer Interp.\ when quality matters and time does not; be aware that iterative methods hinge on the $p=1$ start, which is cheap to scan exactly.
    \item \textbf{Test several methods at small scale} for a new problem class: rankings change between MaxCut and MIS.
    \item \textbf{Near the noise limit}, small quality differences are unmeasurable; choose the fastest method.
    \item \textbf{Build a database} of good angles when similar problems recur; benchmarking ranks methods and tells you where to spend resources.
    \item \textbf{Count rounds, not circuits:} submit the circuits of an iteration together, carry about $M^*$ copies a round\ona{, run a closed loop in a session, and prefer methods with few rounds when the latency dominates}.
\end{enumerate}
\ona{And keep the yardstick of \Chap~\chref{C.Motivation} in mind: standard MaxCut is solved classically in seconds. The insights should transfer to warm-start and recursive QAOA, which overcome the constant-depth obstructions that standard QAOA faces, and which we turn to next.}
\end{frame}

\begin{frame}\ft{Exercises}

\begin{exe}[The exponential queue]\label{exe:ch09:mm1}
For exponential service times with the mean $1/\mu$, show from \eqref{eq:ch09:pk} that $\mathbb E W = \rho/(\mu-\lambda)$ and that the mean time from arrival to departure is $1/(\mu-\lambda)$. For $\mathbb E S = 60$~s, find the load at which a job waits ten minutes on average, and the load at which it does so when one job in ten is a session of $330$~s on average and the others take $30$~s.
\end{exe}
\begin{exe}[A closed loop in the long run]\label{exe:ch09:loop}
Show that \eqref{eq:ch09:loop} implies $(\rho t_c-t_{\rm net})/(1-\rho)\le\liminf_k\mathbb E W_k\le\limsup_k\mathbb E W_k\le\rho t_c/(1-\rho)$. Evaluate the bounds for the round trip of \ona{Fig.~\ref{fig:ch09:roundtrip}}\onp{the measured round trip} at $\rho = 0.5$, $0.8$ and $0.9$, and compare them with the wait \eqref{eq:ch09:pk} of a single job for the service times of the previous exercise. At $\rho = 0.8$, after how many rounds is the bound on $\mathbb E W_k$ within $10\%$ of its limit?
\end{exe}
\end{frame}

\begin{frame}\ft{Exercises, continued}
\onp{\footnotesize}
\begin{exe}[Circuits one by one]\label{exe:ch09:batch}
If each of the $c$ circuits of a round is submitted on its own, a round takes $c\,(t_{\rm lat}+Mt_{\rm shot})$. Show that \ona{Proposition~\ref{propn:ch09:copies}}\onp{the proposition on copies} holds with the same $M^*$, and that submitting the circuits together divides the time at $M^*$ by $c\,(t_{\rm lat}+M^*t_{\rm shot})/(t_{\rm lat}+cM^*t_{\rm shot})$. The reconstruction of \Chap~\chref{C.PSR} for $d$ parameters, each of degree $1$, submits the $3^d$ circuits of its grid in one round; gradient descent with the shift rule submits $2d$ circuits in each of $K$ rounds. With $M = 1000$ copies of each circuit, $t_{\rm lat} = 6$~s, $t_{\rm shot} = 100\,\mu$s and $K = 50$, find the largest $d$ for which the single round is faster.
