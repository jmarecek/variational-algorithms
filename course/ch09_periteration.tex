\chapter{Per-Iteration Complexity of QAOA}\label{C.PerIteration}

\ona{\Chap~\chref{C.IterationComplexity} counted iterations. This \chap{} asks what one iteration \emph{costs}: copies, circuits, depth, classical simulation and, above all, time. We first prove two tools from applied probability: Hoeffding's inequality, which says how many copies an estimate needs, and the properties of the Poisson process, the model of the arrivals at a shared device. On a device in the cloud, each iteration is a round trip: the circuits travel to the device, wait in its queue, are prepared and executed, and the results travel back, and the next circuits cannot be sent before they have arrived. We take a measured round trip apart, model the queue and derive the waits of a variational method in it, and show how the delays decide how many copies a round should carry and which methods are fast. We then recall from \Chap~\chref{C.VQE} why no per-iteration budget rescues the worst case, since training is $\mathbf{NP}$-hard \citep{bittel2021training}, and follow \citet{marecek2026angles}, \emph{Setting angles in quantum approximate optimization at utility-scale}, where these costs decide how the angles of QAOA are set at 100 qubits or more. The numbers of Section~\ref{sec:ch09:prob} and of Sections~\ref{sec:ch09:round}--\ref{sec:ch09:copies} are computed by \texttt{code/ch09\_figures.py}.}

\section{Preliminaries from applied probability}\label{sec:ch09:prob}

\begin{frame}\ft{Averages of bounded values}
\onp{\footnotesize}
\ona{Each evaluation of a variational cost on a device is an estimate: the mean of the values measured on $M$ copies of a state. How many copies give the precision $\epsilon$ with the probability $1-\delta$? If a measured value $X$ has the mean $\mu$ and the variance $\sigma^2$, the mean $\bar X$ of $M$ independent copies has the variance $\sigma^2/M$, and Chebyshev's inequality gives $\mathrm{Pr}[|\bar X-\mu|\ge\epsilon]\le\sigma^2/(M\epsilon^2)$, which asks for $\sigma^2/(\delta\epsilon^2)$ copies, a number proportional to $1/\delta$. A measured value is bounded, however, since it is an eigenvalue of the observable, and for bounded values the probability of a large deviation decays exponentially in $M$.}
\begin{thm}[\citet{hoeffding1963probability}]\label{thm:ch09:hoeffding}
Let $X_1,\dots,X_M$ be independent random variables with $a\le X_i\le b$, let $\bar X = \frac1M\sum_{i=1}^MX_i$, and let $\mu = \mathbb E\,\bar X$. Then for every $\epsilon>0$,
\begin{equation}\label{eq:ch09:hoeffding}
    \mathrm{Pr}\big[\bar X-\mu\ge\epsilon\big]\le e^{-2M\epsilon^2/(b-a)^2}
    \qquad\text{and}\qquad
    \mathrm{Pr}\big[|\bar X-\mu|\ge\epsilon\big]\le 2e^{-2M\epsilon^2/(b-a)^2}.
\end{equation}
\end{thm}
\ona{Equating the right-hand side with $\delta$, the mean of}
\onp{Hence}
\begin{equation}\label{eq:ch09:hcopies}
    M\ge\frac{(b-a)^2}{2\epsilon^2}\ln\frac2\delta
\end{equation}
\ona{copies is within $\epsilon$ of $\mu$ with the probability at least $1-\delta$. Only the range $b-a$ enters, not the distribution. Since $\mathrm{Var}\,X\le\mathbb E(X-\frac{a+b}2)^2\le\frac{(b-a)^2}4$, Chebyshev's inequality with the same information asks for $(b-a)^2/(4\delta\epsilon^2)$ copies. At $\delta = 0.05$, the two counts are $1.84$ and $5$ times $(b-a)^2/\epsilon^2$; at $\delta = 10^{-6}$, they are $7.25$ and $250\,000$ times. The logarithm of $1/\delta$ is what makes a union bound over many estimates affordable, as in the reconstruction from a grid of \Chap~\chref{C.PSR}.}
\onp{copies suffice for $|\bar X-\mu|<\epsilon$ with probability $1-\delta$; only the range $b-a$ enters. Chebyshev, with $\mathrm{Var}\,X\le(b-a)^2/4$, asks for $(b-a)^2/(4\delta\epsilon^2)$: at $\delta = 10^{-6}$, $250\,000$ against $7.25$ times $(b-a)^2/\epsilon^2$.}
\end{frame}

\begin{frame}\ft{Hoeffding's lemma}
\onp{\footnotesize}
\ona{The proof rests on a bound on the moment generating function of a single bounded value.}
\begin{lem}[\citet{hoeffding1963probability}]\label{lem:ch09:hoeffding}
If $a\le X\le b$ and $\mu = \mathbb E\,X$, then $\mathbb E\,e^{s(X-\mu)}\le e^{s^2(b-a)^2/8}$ for every real $s$.
\end{lem}
\ona{\emph{Proof.} Let $p = (\mu-a)/(b-a)$, which lies in $[0,1]$, and $h = s(b-a)$. The function $x\mapsto e^{sx}$ is convex, so that on $[a,b]$ it lies below its chord: $e^{sx}\le\frac{b-x}{b-a}e^{sa}+\frac{x-a}{b-a}e^{sb}$. Taking expectations, $\mathbb E\,e^{sX}\le(1-p)e^{sa}+pe^{sb}$, which is $\mathbb E\,e^{sY}$ for the variable $Y$ that takes the value $b$ with the probability $p$ and the value $a$ otherwise, and has the same mean $\mu$. Since $a-\mu = -p(b-a)$, multiplying by $e^{-s\mu}$ gives $\mathbb E\,e^{s(X-\mu)}\le e^{\varphi(h)}$ with $\varphi(h) = -ph+\ln(1-p+pe^h)$. Here $\varphi(0) = 0$; $\varphi'(h) = -p+u$ with $u = pe^h/(1-p+pe^h)$, which lies in $[0,1]$, so that $\varphi'(0) = 0$; and $\varphi''(h) = u(1-u)\le\frac14$. By Taylor's theorem, $\varphi(h)\le\frac{h^2}8 = \frac{s^2(b-a)^2}8$. \qed}
\onp{Proof: $e^{sx}$ lies below its chord on $[a,b]$, so $\mathbb E\,e^{sX}\le(1-p)e^{sa}+pe^{sb}$ with $p = (\mu-a)/(b-a)$, the value for the two-point variable on $\{a,b\}$ with the mean $\mu$. With $h = s(b-a)$, multiplying by $e^{-s\mu}$ gives $e^{\varphi(h)}$, $\varphi(h) = -ph+\ln(1-p+pe^h)$, with $\varphi(0) = \varphi'(0) = 0$ and $\varphi''\le\frac14$; Taylor's theorem gives $\varphi(h)\le h^2/8$.}

\end{frame}

\begin{frame}\ft{The proof of Hoeffding's inequality}
\onp{\footnotesize}
\ona{The theorem follows by the method of \citet{chernoff1952measure}: Markov's inequality, applied to an exponential of the sum, and Lemma~\ref{lem:ch09:hoeffding} for each term.}
\ona{\emph{Proof of Theorem~\ref{thm:ch09:hoeffding}.} Let $\mu_i = \mathbb E\,X_i$ and $s>0$. The variable $e^{sM(\bar X-\mu)} = \prod_ie^{s(X_i-\mu_i)}$ is positive, and by Markov's inequality, by independence and by Lemma~\ref{lem:ch09:hoeffding},}
\onp{For $s>0$, by Markov's inequality, independence and the lemma,}
\begin{equation}\label{eq:ch09:chernoff}
    \mathrm{Pr}\big[\bar X-\mu\ge\epsilon\big]\le e^{-sM\epsilon}\,\mathbb E\,e^{sM(\bar X-\mu)} = e^{-sM\epsilon}\prod_{i=1}^M\mathbb E\,e^{s(X_i-\mu_i)}\le e^{-sM\epsilon+Ms^2(b-a)^2/8}.
\end{equation}
\ona{The exponent is a convex quadratic in $s$, least at $s = 4\epsilon/(b-a)^2$, where it equals $-2M\epsilon^2/(b-a)^2$. The variables $-X_i$ lie in $[-b,-a]$, an interval of the same length, which gives the same bound on $\mathrm{Pr}[\bar X-\mu\le-\epsilon]$, and the two-sided bound is the sum of the two. \qed}
\onp{The exponent is least at $s = 4\epsilon/(b-a)^2$, where it is $-2M\epsilon^2/(b-a)^2$; the lower tail follows from the $-X_i$, in $[-b,-a]$.}

\end{frame}

\begin{frame}\ft{The Poisson process}
\onp{\footnotesize}
\ona{The users of a device in the cloud submit their jobs independently of one another, at times that nobody schedules. The standard model of such arrivals is the Poisson process \citep[Chapter~11]{harcholbalter2013performance}, which Section~\ref{sec:ch09:queue} takes for the jobs of the other users.}
\begin{defn}\label{defn:ch09:poisson}
Let $\lambda>0$, and let $X_1,X_2,\dots$ be independent and exponentially distributed with the mean $1/\lambda$: $\mathrm{Pr}[X_k>x] = e^{-\lambda x}$ for $x\ge0$. The times $T_k = X_1+\dots+X_k$ are the \emph{arrivals} of a \emph{Poisson process} of \emph{rate} $\lambda$, the $X_k$ are its \emph{gaps}, and $N(t) = |\{k : T_k\le t\}|$ is the number of arrivals up to the time $t$.
\end{defn}
\onp{\begin{itemize}\setlength\itemsep{0.3em}
    \item The model of arrivals that nobody schedules: the jobs of the other users of a device.
    \item The gaps have the mean $1/\lambda$; on average, $\lambda$ arrivals per unit of time.
\end{itemize}}
\end{frame}

\begin{frame}\ft{Properties of the Poisson process}
\onp{\footnotesize}
\begin{propn}\label{propn:ch09:poisson}
A Poisson process of rate $\lambda$ has the following properties.
\begin{itemize}\setlength\itemsep{0.1em}
    \item[(a)] It forgets: $\mathrm{Pr}[X_k>x+y\mid X_k>y] = e^{-\lambda x}$ for all $x,y\ge0$.
    \item[(b)] It restarts: for every time $u\ge0$, the arrivals after $u$, less $u$, form a Poisson process of rate $\lambda$, independent of the arrivals up to $u$. The same holds when $u$ is a \emph{stopping time}, a random time decided by what has happened up to it, such as the time at which a client submits a round.
    \item[(c)] Its counts are Poisson: $\mathrm{Pr}[N(u+t)-N(u) = k] = e^{-\lambda t}(\lambda t)^k/k!$, with the mean $\lambda t$, and the counts in disjoint intervals are independent.
    \item[(d)] If the $k$th arrival brings the work $S_k$, and $S_1,S_2,\dots$ are distributed as $S$ and independent of the arrivals, the work that arrives in an interval of length $t$ has the mean $\lambda t\,\mathbb ES$.
    \item[(e)] Independent Poisson processes of the rates $\lambda_1,\dots,\lambda_m$ together form a Poisson process of rate $\lambda_1+\dots+\lambda_m$.
\end{itemize}
\end{propn}
\end{frame}

\begin{frame}\ft{The properties of the Poisson process, proved}
\onp{\footnotesize}
\ona{\emph{Proof.} (a) $e^{-\lambda(x+y)}/e^{-\lambda y} = e^{-\lambda x}$. (b) Given $N(u) = k$ and $T_1,\dots,T_k$, all that is known of $X_{k+1}$ is that $X_{k+1}>u-T_k$. By (a), the time $T_{k+1}-u$ to the next arrival is then exponential with the mean $1/\lambda$, and the gaps $X_{k+2},X_{k+3},\dots$ are independent of it and of the past. The same argument applies at a stopping time $u$, which on the event $N(u) = k$ is decided by $T_1,\dots,T_k$ and by what is independent of the arrivals. (c) By induction on $k$, $T_k$ has the density $\lambda^ks^{k-1}e^{-\lambda s}/(k-1)!$, since $\int_0^s\frac{\lambda^kr^{k-1}e^{-\lambda r}}{(k-1)!}\,\lambda e^{-\lambda(s-r)}\,dr = \frac{\lambda^{k+1}s^ke^{-\lambda s}}{k!}$. Hence, for $k\ge1$, $\mathrm{Pr}[N(t) = k] = \mathrm{Pr}[T_k\le t<T_k+X_{k+1}] = \int_0^t\frac{\lambda^ks^{k-1}e^{-\lambda s}}{(k-1)!}\,e^{-\lambda(t-s)}\,ds = e^{-\lambda t}\frac{(\lambda t)^k}{k!}$, and $\mathrm{Pr}[N(t) = 0] = \mathrm{Pr}[X_1>t] = e^{-\lambda t}$. By (b), the same holds for $N(u+t)-N(u)$, and the count in an interval is independent of the counts in the intervals before it. (d) The work $\sum_kS_k\mathbf 1[u<T_k\le u+t]$ has the mean $\mathbb ES\cdot\mathbb E[N(u+t)-N(u)] = \lambda t\,\mathbb ES$, by (c). (e) For $m = 2$, the first arrival of either process comes after $\min\{X,X'\}$, for the independent first gaps $X$ and $X'$ of the rates $\lambda_1$ and $\lambda_2$, and $\mathrm{Pr}[\min\{X,X'\}>x] = e^{-\lambda_1x}e^{-\lambda_2x}$. By (b), both processes restart at that arrival, a stopping time, so that the gaps of the merged process are independent and exponential with the rate $\lambda_1+\lambda_2$. Induction on $m$ completes the proof. \qed}
\onp{\begin{itemize}\setlength\itemsep{0.3em}
    \item[(a)] $e^{-\lambda(x+y)}/e^{-\lambda y} = e^{-\lambda x}$.
    \item[(b)] Given $N(u) = k$ and $T_1,\dots,T_k$, only $X_{k+1}>u-T_k$ is known; by (a), $T_{k+1}-u$ is exponential, and the later gaps are independent of the past.
    \item[(c)] $T_k$ has the density $\lambda^ks^{k-1}e^{-\lambda s}/(k-1)!$, and $N(t) = k$ if and only if $T_k\le t<T_k+X_{k+1}$: $\int_0^t\frac{\lambda^ks^{k-1}e^{-\lambda s}}{(k-1)!}e^{-\lambda(t-s)}\,ds = e^{-\lambda t}\frac{(\lambda t)^k}{k!}$.
    \item[(d)] The mean of $\sum_kS_k\mathbf 1[u<T_k\le u+t]$ is $\mathbb ES\cdot\lambda t$, by (c).
    \item[(e)] $\mathrm{Pr}[\min\{X,X'\}>x] = e^{-(\lambda_1+\lambda_2)x}$, and both processes restart at the first arrival, by (b).
\end{itemize}}

\end{frame}

\begin{frame}\ft{The gap that covers a time}
\onp{\footnotesize}
\ona{A time fixed in advance is more likely to fall in a long gap than in a short one, since long gaps cover more of the line. This is the \emph{inspection paradox}; it finds the service in progress in the proof of Proposition~\ref{propn:ch09:pk}.}
\begin{lem}\label{lem:ch09:inspection}
In a Poisson process of rate $\lambda$, let $u>0$ be a fixed time. The time from $u$ to the next arrival is exponential with the mean $1/\lambda$. The time back from $u$ to the last arrival, or to $0$ if there is none, is distributed as $\min\{X,u\}$ with $X$ exponential with the mean $1/\lambda$, and is independent of the time to the next arrival. The gap that covers $u$ has the mean $(2-e^{-\lambda u})/\lambda$, and as $u\to\infty$ its length has the density
\begin{equation}\label{eq:ch09:sizebias}
    \lambda^2xe^{-\lambda x} = \frac{x\,f(x)}{\mathbb E\,X},\qquad f(x) = \lambda e^{-\lambda x}.
\end{equation}
\end{lem}
\ona{\emph{Proof.} By Proposition~\ref{propn:ch09:poisson}(b), the arrivals after $u$ form a Poisson process independent of the arrivals up to $u$, so that the time to the next arrival is the first gap of a Poisson process, independent of the time back. For $0\le x<u$, the time back exceeds $x$ if and only if there is no arrival in $(u-x,u]$, which by Proposition~\ref{propn:ch09:poisson}(c) has the probability $e^{-\lambda x}$; it never exceeds $u$. The mean of $\min\{X,u\}$ is $\int_0^ue^{-\lambda x}\,dx = (1-e^{-\lambda u})/\lambda$. As $u\to\infty$, the gap is the sum of two independent exponential variables, with the density $\int_0^x\lambda e^{-\lambda y}\,\lambda e^{-\lambda(x-y)}\,dy = \lambda^2xe^{-\lambda x}$. \qed}

\ona{The gap that covers a time is thus, on average, twice as long as a gap: a gap of length $x$ is found with a probability proportional to $x$. The same holds in the long run for independent gaps of any distribution with a finite mean, since the gaps of lengths in $[x,x+dx]$ cover the fraction $xf(x)\,dx/\mathbb EX$ of the time, and the part of the covering gap that remains after a random time then has the mean $\mathbb EX^2/(2\,\mathbb EX)$. For the exponential distribution this is $1/\lambda$ again, and for the service in progress in Proposition~\ref{propn:ch09:pk} it is $\mathbb ES^2/(2\,\mathbb ES)$.}
\onp{Proof: by (b), the future after $u$ is a fresh process, independent of the past; the time back exceeds $x<u$ if and only if $(u-x,u]$ has no arrival, with probability $e^{-\lambda x}$ by (c).}
\onp{\begin{itemize}\setlength\itemsep{0.1em}
    \item A gap of length $x$ is found with a probability $\propto x$: on average twice as long as a gap.
    \item For any gaps, the rest of the covering gap has the mean $\mathbb EX^2/(2\,\mathbb EX)$: $1/\lambda$ here, $\mathbb ES^2/(2\,\mathbb ES)$ for a service found in progress.
\end{itemize}}
\end{frame}

\section{The cost of one iteration}

\begin{frame}\ft{Shots: estimating the energy}
\ona{One evaluation of $F_p(\boldsymbol\theta) = \langle H_C\rangle$ for MaxCut is the average of the cut value $f(x) = \frac12\sum_{(i,j)\in E}w_{ij}(1-z_iz_j)$ over $M$ samples $x$. Each edge contributes $0$ or $w_{ij}$ to $f(x)$, so that the values of $f$ lie in an interval of length $\|w\|_1 = \sum_e|w_e|$, and Theorem~\ref{thm:ch09:hoeffding} gives}
\begin{equation}\label{eq:ch09:shots}
    \mathrm{Pr}\big[|\hat F - F_p| \ge \epsilon\big] \le 2\exp\Big(-\frac{2M\epsilon^2}{\|w\|_1^2}\Big),
    \qquad\text{so}\qquad M = \mathcal O\Big(\frac{\|w\|_1^2}{\epsilon^2}\log\frac1\delta\Big)
\end{equation}
\ona{shots suffice for additive error $\epsilon$ with probability $1-\delta$; in terms of approximation ratio, $\epsilon = \eta\cdot\mathrm{OPT}$ with $\mathrm{OPT} \ge \|w\|_1/2$ for positive weights gives $M = \mathcal O(\eta^{-2}\log(1/\delta))$, independent of $n$. The variance $\sigma^2$ of \Chap~\chref{C.IterationComplexity} is $\mathrm{Var}[f(x)]/M$. For general Hamiltonians $H = \sum_kh_kP_k$, terms are grouped into commuting sets and the count becomes $\mathcal O((\sum_k|h_k|)^2/\epsilon^2)$.}
\onp{$M = \mathcal O(\|w\|_1^2\epsilon^{-2}\log(1/\delta))$ shots per evaluation; a relative precision $\eta$ needs $\mathcal O(\eta^{-2})$ shots, independent of $n$; MaxCut is diagonal, one measurement basis.}
\end{frame}

\begin{frame}\ft{Circuits, depth, and time}
\begin{itemize}\setlength\itemsep{0.3em}
    \item \textbf{Circuit evaluations per iteration:} COBYLA $1$, SPSA $2$, parameter-shift gradient $2\cdot 2p$ (\Chap~\chref{C.PSR}); each with $M$ shots.
    \item \textbf{Gates per layer:} $|E|$ two-qubit $ZZ$ rotations and $n$ single-qubit rotations. On hardware with restricted connectivity, each $ZZ$ between non-adjacent qubits costs SWAPs; for a heavy-hex device and a dense problem graph the depth grows by a factor $\mathcal O(n)$, which is why hardware studies use hardware-native (heavy-hex, line-based) graphs.
    \item \textbf{Wall-clock:} on \emph{ibm\_boston}, $t_{\rm shot} \approx 100\,\mu$s and a latency of about $6$~s for each submitted circuit; one iteration of COBYLA with $M = 20\,000$ shots takes about $8$~s, most of it latency\ona{; Section~\ref{sec:ch09:round} takes such a round trip apart}.
    \item \textbf{Total:} $T = N\,(t_{\rm qc} + c\,M\,t_{\rm shot}) + Q\,t_{\rm shot}$ for $N$ iterations of $c$ circuits each and $Q$ final samples, the resource model of the Pareto analysis below; it is \eqref{eq:ch09:round} with $W = 0$ and the latency $t_{\rm qc}$ paid once a round.
\end{itemize}
\end{frame}

\begin{frame}\ft{Classical evaluation of $\langle H_C\rangle$}
\ona{When the loop is run offline, the cost is that of a classical simulator:}
\begin{itemize}\setlength\itemsep{0.3em}
    \item \textbf{Depth one is cheap:} $\braket{Z_iZ_j}$ depends only on the neighbourhoods of $i,j$ and has a closed form; the whole landscape is $\mathcal O(|E|\cdot\deg^2)$ per point.
    \item \textbf{Statevector:} $\mathcal O(2^n)$ memory and $\mathcal O(p(|E|+n)2^n)$ time; 40 qubits took five minutes on 1024 GPUs.
    \item \textbf{Matrix product states} with bond dimension $\chi$: each two-qubit gate costs $\mathcal O(\chi^3)$ after mapping the graph to a line, so $\mathcal O(p\,|E|\,\chi^3)$ per evaluation, polynomial in $n$ but exponential in the entanglement the circuit builds; long-range edges need swaps or a SAT-optimised relabelling. Accuracy is monitored by $F_{\rm approx} = \prod_g\sum_{j\le\chi}\lambda_{j,g}^2$.
    \item \textbf{Pauli propagation:} propagate $H_C$ backwards through the circuit, truncating Pauli strings of weight above $W$ or coefficient below a threshold; cheap for sparse graphs, cost growing fast with density.
    \item \textbf{Empirically}, at $p = 10$ and $\chi = 24$, both scale polynomially in $|E|$, with $100$--$1000$ s per evaluation for a fully connected 100-node graph\ona{ \citep{marecek2026angles}}.
\end{itemize}
\end{frame}

\section{The round trip}\label{sec:ch09:round}

\begin{frame}\ft{A round trip}
\onp{\footnotesize}
\ona{An iteration of a variational method on a device in the cloud is a \emph{round trip}. The classical computer sends circuits to the device; the device queues them, prepares them for execution and executes them; and the results return. The circuits of the next iteration depend on these results, and cannot be sent before the results have returned.}
\begin{defn}\label{defn:ch09:round}
A \emph{round} submits $c$ circuits, with $M$ copies of each, and waits for their results. It takes the time
\begin{equation}\label{eq:ch09:round}
    t_{\rm round} = t_{\rm net}+W+t_{\rm prep}+c\,M\,t_{\rm shot},
\end{equation}
where $t_{\rm net}$ is the time taken to send the circuits and to return the results, $W$ the wait in the queue of the device, $t_{\rm prep}$ the time taken to prepare the circuits for execution, and $t_{\rm shot}$ the time of one copy. A method is \emph{adaptive} if the circuits of each round depend on the results of the earlier rounds.
\end{defn}
\ona{Gradient descent, simultaneous perturbation and COBYLA are adaptive: $K$ iterations take $K$ rounds, one after another, and at least $K(t_{\rm net}+t_{\rm prep})+\sum_kW_k$, however many devices are available. Only the circuits of one round, and the copies of one circuit, can be executed in parallel. We call $t_{\rm lat} = t_{\rm net}+t_{\rm prep}$ the \emph{latency} of a round. It is paid once a round if the $c$ circuits are submitted together, and $c$ times if they are submitted one by one; whether $t_{\rm prep}$ grows with $c$ depends on the device, and we take it to be paid once a round.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item Adaptive methods (gradient descent, simultaneous perturbation, COBYLA): $K$ iterations are $K$ rounds in sequence; only the circuits of a round run in parallel.
    \item The latency $t_{\rm lat} = t_{\rm net}+t_{\rm prep}$: once a round if its circuits are submitted together, $c$ times if one by one.
\end{itemize}
\end{frame}

\section{Waiting in a queue}\label{sec:ch09:queue}

\begin{frame}\ft{A queue and its mean wait}
\onp{\footnotesize}
\ona{A device in the cloud serves many users. We model it as a single server, which executes one job at a time, first come, first served. The jobs of the other users arrive at the times of a Poisson process of rate $\lambda$ (Definition~\ref{defn:ch09:poisson}), independently of one another, and take independent service times distributed as $S$. The \emph{load} $\rho = \lambda\,\mathbb E S$ is the fraction of the time for which the device is busy, and we assume that $\rho<1$, so that the queue has a stationary state \citep{harcholbalter2013performance}.}
\begin{lem}[\citet{little1961proof}]\label{lem:ch09:little}
In a queue in its stationary state, the mean number of jobs waiting is $L_q = \lambda\,\mathbb E W$, where $\mathbb E W$ is the mean wait of a job.
\end{lem}
\begin{propn}[Pollaczek and Khinchine]\label{propn:ch09:pk}
In this queue, in its stationary state, a job that arrives at a time of the Poisson process, or at a time independent of the queue, waits on average
\begin{equation}\label{eq:ch09:pk}
    \mathbb E\,W = \frac{\lambda\,\mathbb E S^2}{2(1-\rho)} = \frac\rho{1-\rho}\cdot\frac{\mathbb E S^2}{2\,\mathbb E S}.
\end{equation}
\end{propn}
\ona{\emph{Proof.} Such a job finds the queue in its stationary state; for the arrivals of a Poisson process, this is the property that Poisson arrivals see time averages \citep[Chapter~13]{harcholbalter2013performance}. Its wait is the work that it finds: the remaining service of the job in service, if any, and the services of the jobs that wait. The device is busy with probability $\rho$, and the job in service is then found at a uniformly random time in its service, a service of length $s$ being found with a probability proportional to $s$, as a gap is in Lemma~\ref{lem:ch09:inspection}; the remaining service has the mean $\mathbb E S^2/(2\,\mathbb E S)$. The services of the $L_q$ waiting jobs do not depend on their number, and have the total mean $L_q\,\mathbb E S = \lambda\,\mathbb E W\,\mathbb E S = \rho\,\mathbb E W$, by Lemma~\ref{lem:ch09:little}. Hence $\mathbb E W = \rho\,\mathbb E S^2/(2\,\mathbb E S)+\rho\,\mathbb E W$. \qed}
\onp{Proof: the work found on arrival is the remaining service, of mean $\rho\,\mathbb ES^2/(2\,\mathbb ES)$, plus $L_q\,\mathbb ES = \rho\,\mathbb EW$ by Little's law.}
\end{frame}

\begin{frame}\ft{The waits of a closed loop}
\onp{\footnotesize}
\ona{A variational method does not submit its rounds at the times of a Poisson process. Each round is submitted as soon as the results of the previous one have returned, and the device has meanwhile served only the work that arrived since.}
\begin{propn}\label{propn:ch09:loop}
In the queue above, let a client submit its first round at a time independent of the queue, and each later round $t_{\rm net}$ after the previous round has been served; let each round occupy the device for $s$, and let $t_c = s+t_{\rm net}$. If the device serves first come, first served, the waits $W_k$ of the rounds satisfy $\mathbb E W_1 = \lambda\,\mathbb E S^2/(2(1-\rho))$ and
\begin{equation}\label{eq:ch09:loop}
    \rho\,(\mathbb E W_k+t_c)-t_{\rm net}\le\mathbb E W_{k+1}\le\rho\,(\mathbb E W_k+t_c),
\end{equation}
so that $\mathbb E W_k\le\rho^{k-1}\,\mathbb E W_1+\rho\,t_c/(1-\rho)$.
\end{propn}
\ona{\emph{Proof.} The first wait is that of Proposition~\ref{propn:ch09:pk}. Let round $k$ be submitted at the time $a_k$, so that $a_{k+1}-a_k = W_k+t_c$. Every job that arrived before $a_k$ is served before round $k$, which ends at $a_k+W_k+s$. The work found at $a_{k+1}$, which is $W_{k+1}$, is therefore that of jobs that arrived in $(a_k,a_{k+1}]$: it is at most their total service $A_k$, and at least $A_k-t_{\rm net}$, since the device can serve them only after round $k$ has ended. The length $W_k+t_c$ of the interval is known at $a_k$, and by Proposition~\ref{propn:ch09:poisson}(b) and (d) the arrivals after the stopping time $a_k$ do not depend on what happened before, so that $\mathbb E[A_k\,|\,W_k] = \lambda\,\mathbb E S\,(W_k+t_c) = \rho\,(W_k+t_c)$. Taking expectations gives \eqref{eq:ch09:loop}, and induction the last bound. \qed}
\onp{Proof: round $k+1$ finds only the work that arrived during $(a_k,a_{k+1}]$, of mean $\rho\,(W_k+t_c)$, less at most $t_{\rm net}$ served meanwhile.}
\end{frame}

\section{Copies and rounds}\label{sec:ch09:copies}

\begin{frame}\ft{How many copies a round}
\onp{\footnotesize}
\ona{\Chap~\chref{C.IterationComplexity} found that, with $N$ copies in all, many iterations with few copies are as good as few iterations with many, as far as the bounds are concerned \citep{sweke2020stochastic}. Latency changes this. Consider an ideal device and unbiased estimates of the gradient from $c$ circuits with $M$ copies each, whose variance is $V = V_1/M$, and let $L$, $\Delta = C(\boldsymbol\theta_0)-C^*$ and $\epsilon$ be as in \Chap~\chref{C.IterationComplexity}.}
\begin{propn}\label{propn:ch09:copies}
By the corollary on unbiased estimates of \Chap~\chref{C.IterationComplexity},
\begin{equation}\label{eq:ch09:K}
    K(M) = \max\{8L\Delta/\epsilon,\ 32L\Delta V_1/(M\epsilon^2)\}
\end{equation}
rounds give an $\epsilon$-stationary point. If each round takes $t_{\rm lat}+cMt_{\rm shot}$, the time
\begin{equation}\label{eq:ch09:copies}
    T(M) = K(M)\,(t_{\rm lat}+cMt_{\rm shot})
\end{equation}
does not increase with $M$ up to $M^* = 4V_1/\epsilon$, and increases beyond. It is constant up to $M^*$ if $t_{\rm lat} = 0$, and
\begin{equation}\label{eq:ch09:ratio}
    \frac{T(1)}{T(M^*)} = M^*\,\frac{t_{\rm lat}+ct_{\rm shot}}{t_{\rm lat}+cM^*t_{\rm shot}}.
\end{equation}
\end{propn}
\ona{\emph{Proof.} For $M\le M^*$, $K(M) = 32L\Delta V_1/(M\epsilon^2)$, and $T(M) = (32L\Delta V_1/\epsilon^2)(t_{\rm lat}/M+ct_{\rm shot})$, which decreases with $M$ if $t_{\rm lat}>0$ and is constant if $t_{\rm lat} = 0$. For $M\ge M^*$, $K(M) = 8L\Delta/\epsilon$, and $T(M)$ increases with $M$. Since $K(1) = M^*K(M^*)$, \eqref{eq:ch09:ratio} follows. \qed}

\ona{Without latency, only the total number of copies counts, up to $M^*$. With latency, a round should carry $M^*$ copies, the most that still reduce the number of rounds; beyond $M^*$ the bound needs $8L\Delta/\epsilon$ rounds whatever $M$, as gradient descent with exact gradients does. When $t_{\rm lat}$ dominates $cM^*t_{\rm shot}$, the ratio \eqref{eq:ch09:ratio} is about $M^*$. In the worked example of \Chap~\chref{C.VQE}, the gate-by-gate shift rule has $c = 16$ and $V_1 = 10\sigma^2 = 20$, and $L = 20$ and $\Delta = 0.923$; with $\epsilon = 0.5$, $M^* = 160$ and $K(M^*) = 295$, and with $t_{\rm lat} = 6$~s the ratio is $153$.}
\onp{\begin{itemize}\setlength\itemsep{0.2em}
    \item Without latency only the total of copies counts, up to $M^*$ (\Chap~\chref{C.IterationComplexity}); with latency a round should carry $M^*$ copies.
    \item Worked example: $c = 16$, $V_1 = 20$, $L = 20$, $\Delta = 0.923$, $\epsilon = 0.5$: $M^* = 160$, $K(M^*) = 295$; with $6$~s a round, $T(1)/T(M^*) = 153$.
\end{itemize}}
\end{frame}

\section{No budget rescues the worst case: training is NP-hard}

\begin{frame}\ft{No budget rescues the worst case}
\ona{Per-iteration costs are polynomial. Iteration counts, by \Chap~\chref{C.IterationComplexity}, are polynomial to reach a \emph{stationary} point. What no polynomial budget buys is the \emph{global} optimum, in the worst case. \Chap~\chref{C.VQE} proved this, following \citet{bittel2021training}:}
\begin{itemize}\setlength\itemsep{0.25em}
    \item Training is $\mathbf{NP}$-hard on $\mathcal O(\log d)$ qubits, with a single layer, and for QAOA with a single layer. On these instances one evaluation of the energy is a small matrix computation, so the cost of an iteration is negligible, and the difficulty is the number of iterations needed to find the global optimum.
    \item Unless $\mathbf P = \mathbf{NP}$, no polynomial-time training procedure guarantees an optimization error below $(1-\alpha_{\max})/2$ of the spectral width for QAOA, that is $1/34$, or $0.061$ under the Unique Games Conjecture; methods that may stop at a local minimum do no better than $1/4$, as the local minima of the landscape include the local optima of single-vertex search.
\end{itemize}
\ona{What the theorem does \emph{not} say: it does not forbid good angles for typical instances, which is what the rest of this \chap{} finds empirically, nor good solutions from local minima, which is what warm starts guarantee (\Chaps~\chref{C.WSRounded}--\chref{C.WSContinuous}). It says that a per-iteration budget, however generous, is not the bottleneck in the worst case: the number of iterations to the global optimum is.}
\onp{Not forbidden: good angles on typical instances (empirically, below); good solutions from local minima (warm starts). Forbidden: a worst-case guarantee from any polynomial budget (\Chap~\chref{C.VQE}).}
\end{frame}

\begin{frame}\ft{Three ways to set angles without a closed loop}
\ona{Recall the QAOA state $\ket{\psi(\boldsymbol\theta)} = \prod_{k=1}^p U_M(\beta_k)U_C(\gamma_k)\ket{+}^{\otimes n}$ and its energy $\langle H_C\rangle$. Closed-loop optimization of $\boldsymbol\theta$ on hardware is (i) slow, given clock speeds and queueing; (ii) impossible to check, since at utility scale $\langle H_C\rangle$ cannot be simulated exactly; and (iii) \textbf{NP}-hard in the worst case, with many local optima even at $p=1$. Figure~\ref{fig:ch09:strategies} shows the alternatives \citep{marecek2026angles}: the angles are found classically, by evaluating the energy (a) on representative structured problems, (b) on small representative instances with the exact energy, or (c) on the utility-scale instance with an approximate energy. They are transferred to the circuit for the utility-scale instance, and only the final sampling runs on the device.}
\ona{\begin{figure}[htb]\centering\begin{tikzpicture}\stratfig\end{tikzpicture}
\caption{Three sources of angles for QAOA at utility scale, after \citet{marecek2026angles}: (a) a class of problems, such as $3$-regular graphs, (b) small instances with the exact energy, and (c) the instance itself, a heavy-hex graph, with an approximate energy. A classical optimiser turns them into the angles $\gamma_1,\dots,\gamma_p$ and $\beta_1,\dots,\beta_p$, which go to the device once.}\label{fig:ch09:strategies}\end{figure}}
\onp{\begin{center}\begin{tikzpicture}[scale=0.7,transform shape]\stratfig\end{tikzpicture}\end{center}
\footnotesize\begin{itemize}\setlength\itemsep{0.2em}
    \item Angles from (a) a table for a class, (b) small instances, exactly, or (c) the instance, approximately.
    \item The loop runs classically; the device receives the angles once and only samples.
\end{itemize}}
\end{frame}

\begin{frame}\ft{A taxonomy of angle-setting methods}
\ona{Some 30 methods from the literature fall into four classes, with many methods belonging to several:}
\begin{description}\setlength\itemsep{0.2em}
    \item[Physics-inspired] angles follow an annealing schedule: Trotterised quantum annealing (TQA), linear ramps with two slopes $(\Delta_\beta,\Delta_\gamma)$, spectral-gap-informed ramps, counterdiabatic and feedback-based (FALQON) schedules. No energy evaluation needed unless the slopes are optimised.
    \item[Parameter transfer] optimal angles concentrate across similar instances: fixed angles for MaxCut on $k$-regular graphs \citep{Wurtz2021}, transfer from small to large instances, rescaling for weighted problems, surrogate models. Require a database of good angles.
    \item[Iterative] initialise depth $p+1$ from depth $p$: Interp.\ and Fourier \citep{Zhou2020}, transition states, basin hopping. Smooth schedules, good quality, but repeated optimizations.
    \item[Machine learning] predict angles from instances: clustering, regression from $p=1$ to $p>1$, reinforcement learning, diffusion models, graph attention networks. Only as good as their training data.
\end{description}
\ona{Superscripts in what follows: $\dagger$ no optimization to the instance, no superscript only method parameters (e.g., slopes) optimised, $\star$ every angle optimised, always with COBYLA.}
\end{frame}

\begin{frame}\ft{Evaluating the energy when you cannot simulate}
\ona{If a method needs $\langle H_C\rangle$, one of four evaluators supplies it:}
\begin{itemize}\setlength\itemsep{0.2em}
    \item \textbf{Statevector (SV):} exact, up to $\sim 40$ qubits on a GPU cluster (a 40-qubit run takes five minutes on 1024 A100 GPUs), $\sim 20$ on a laptop.
    \item \textbf{Hardware:} sampling the QPU; slow and noisy, but the ground truth for what the QPU will do.
    \item \textbf{Matrix product states (MPS):} tensor-network approximation controlled by the bond dimension; accurate for low-connectivity graphs without short cycles, especially after remapping the problem graph onto the tensor line (a SAT-solved mapping). Tends to \emph{underestimate} the energy, yet still guides the optimizer well.
    \item \textbf{Pauli propagation (PP):} Heisenberg-picture propagation of $H_C$ truncated by Pauli weight $W$ and coefficient threshold; accurate for sparse graphs, expensive for dense ones.
\end{itemize}
\ona{Crucially, methods are only comparable under the same evaluator, and a good approximate energy is not the same as a good hardware energy.}
\end{frame}

\begin{frame}\ft{Methodology}
\begin{itemize}\setlength\itemsep{0.2em}
    \item \textbf{Methods:} Interp., Fourier, TQA, recursive transition states (RTS), linear ramps, fixed angles \citep{Wurtz2021}, parameter transfer; 34 configurations, 79\,908 training chains.
    \item \textbf{Problems:} MaxCut and Maximum Independent Set (MIS) on Erd\H{o}s--R\'enyi (ER), heavy-hex (HH, the IBM qubit topology), line-based (LB, a line plus SWAP layers) and random-regular (RR) graphs; small set (20--21 nodes) for SV, large set (50--144 nodes) for MPS and PP; plus low-autocorrelation binary sequences (LABS) up to 21 spins.
    \item \textbf{Depths:} $p \le 10$, since noise limits what can be executed.
    \item \textbf{Metrics:} energy or approximation ratio (against CPLEX optima), \emph{and} the wall-clock cost of finding the angles.
    \item \textbf{Validation:} on \emph{ibm\_boston} and \emph{ibm\_fez} for 144-node heavy-hex and 100-node line-based instances, and a Kruskal--Wallis test with Conover--Iman post-hoc comparisons to decide which methods are statistically distinguishable.
\end{itemize}
\end{frame}

\section{Operational guidance}

\begin{frame}\ft{What a practitioner should do}
\onp{\footnotesize}
\begin{enumerate}\setlength\itemsep{0.2em}
    \item \textbf{Train offline with approximate evaluators} (MPS after remapping the graph to the tensor line; PP for sparse graphs), but \textbf{validate on the QPU}: a method that looks good under an approximate energy may underperform on hardware.
    \item \textbf{Start from transferred or fixed angles.} They cost an $\mathcal O(1)$ look-up, perform well on the problems they were designed for, and are good initial points elsewhere; tailoring them adds $1$--$2\%$ at large cost.
    \item \textbf{Prefer linear ramps} for rugged, higher-order problems (LABS); prefer Interp.\ when quality matters and time does not; be aware that iterative methods hinge on the $p=1$ start, which is cheap to scan exactly.
    \item \textbf{Test several methods at small scale} for a new problem class: rankings change between MaxCut and MIS.
    \item \textbf{Near the noise limit}, small quality differences are unmeasurable; choose the fastest method.
    \item \textbf{Build a database} of good angles when similar problems recur; benchmarking ranks methods and tells you where to spend resources.
    \item \textbf{Count rounds, not circuits:} submit the circuits of an iteration together, carry about $M^*$ copies a round\ona{, run a closed loop in a session, and prefer methods with few rounds when the latency dominates}.
\end{enumerate}
\ona{And keep the yardstick of \Chap~\chref{C.Motivation} in mind: standard MaxCut is solved classically in seconds. The insights should transfer to warm-start and recursive QAOA, which overcome the constant-depth obstructions that standard QAOA faces, and which we turn to next.}
\end{frame}

\ona{\section*{Exercises}}

\begin{frame}\ft{Exercises}
\onp{\footnotesize}
\begin{exe}[The exponential queue]\label{exe:ch09:mm1}
For exponential service times with the mean $1/\mu$, show from \eqref{eq:ch09:pk} that $\mathbb E W = \rho/(\mu-\lambda)$ and that the mean time from arrival to departure is $1/(\mu-\lambda)$. For $\mathbb E S = 60$~s, find the load at which a job waits ten minutes on average, and the load at which it does so when one job in ten is a session of $330$~s on average and the others take $30$~s.
\end{exe}
\begin{exe}[A closed loop in the long run]\label{exe:ch09:loop}
Show that \eqref{eq:ch09:loop} implies \[\frac{\rho t_c-t_{\rm net}}{1-\rho}\le\liminf_k\mathbb E W_k\le\limsup_k\mathbb E W_k\le\frac{\rho t_c}{1-\rho}.\] Evaluate the bounds for the round trip of \ona{Fig.~\ref{fig:ch09:roundtrip}}\onp{the measured round trip} at $\rho = 0.5$, $0.8$ and $0.9$, and compare them with the wait \eqref{eq:ch09:pk} of a single job for the service times of the previous exercise. At $\rho = 0.8$, after how many rounds is the bound on $\mathbb E W_k$ within $10\%$ of its limit?
\end{exe}
\end{frame}

\begin{frame}\ft{Exercises, continued}
\onp{\footnotesize}
\begin{exe}[Circuits one by one]\label{exe:ch09:batch}
If each of the $c$ circuits of a round is submitted on its own, a round takes $c\,(t_{\rm lat}+Mt_{\rm shot})$. Show that \ona{Proposition~\ref{propn:ch09:copies}}\onp{the proposition on copies} holds with the same $M^*$, and that submitting the circuits together divides the time at $M^*$ by $c\,(t_{\rm lat}+M^*t_{\rm shot})/(t_{\rm lat}+cM^*t_{\rm shot})$. Evaluate this for the worked example, with $c = 16$, $M^* = 160$, $t_{\rm lat} = 6$~s and $t_{\rm shot} = 100\,\mu$s.
\end{exe}
\begin{exe}[One round or many]\label{exe:ch09:grid}
The reconstruction of \Chap~\chref{C.PSR} for $d$ parameters, each of degree $1$, submits the $3^d$ circuits of its grid in one round; gradient descent with the shift rule submits $2d$ circuits in each of $K$ rounds. With $M = 1000$ copies of each circuit, $t_{\rm lat} = 6$~s, $t_{\rm shot} = 100\,\mu$s and $K = 50$, find the largest $d$ for which the single round is faster. How does the answer change if the latency falls to $0.1$~s?
\end{exe}
\end{frame}
