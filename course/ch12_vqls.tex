\chapter{Applications: Variational Solvers for Systems of Linear Equations}\label{C.VQLS}

\ona{This \chap{} applies the variational template to a problem outside combinatorial optimization: solving $Ax = b$. It follows the variational quantum linear solver (VQLS) of \citet{bravoprieto2023vqls}. We prove that its two costs certify the same solutions, that the cost bounds the error within a factor of the condition number in either direction, and how the Hadamard test evaluates it; we run it on an Ising-inspired worked example, measure the barren plateau of the global cost against the local one, and compare it with conjugate gradients, in the spirit of \Chap~\chref{C.EQA}. The numbers of this \chap{} are computed by \texttt{code/ch12\_figures.py}.}

\section{The problem and its fault-tolerant solution}

\begin{frame}\ft{Quantum linear systems}
\ona{Given $A \in \C^{N\times N}$, $N = 2^n$, and $b \in \C^N$, the \emph{quantum linear systems problem} asks for the normalised state}
\begin{equation}
    \ket{x} \propto A^{-1}\ket{b},
\end{equation}
\ona{not the vector $x$ itself: reading out all $N$ amplitudes would cost $\Omega(N)$. The output is useful for computing expectations $\braket{x|M|x}$, overlaps, or sampling from $|x_i|^2$.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item The HHL algorithm \citep{harrow2009quantum} prepares $\ket x$ in time $\mathrm{poly}(\log N,\kappa,1/\epsilon)$ for sparse $A$ with condition number $\kappa$, given oracles for $A$ and a state-preparation routine for $\ket b$; improvements bring the $\kappa$ and $\epsilon$ dependence to near-optimal \citep{childs2017quantum}.
    \item Caveats: the oracles, the state preparation, the read-out, and the $\kappa$ dependence each can erase the exponential advantage; the algorithm needs fault tolerance.
\end{itemize}
\ona{VQLS is the near-term, heuristic alternative: a variational ansatz for $\ket x$ and a cost that vanishes exactly at the solution.}
\end{frame}

\section{The variational quantum linear solver}

\begin{frame}\ft{Cost functions}
\onp{\footnotesize}
\ona{Assume $A = \sum_{l=1}^{L} c_lA_l$ with real $c_l$ and Hermitian unitaries $A_l$, such as Pauli strings, and $\ket b = U_b\ket 0$. For a trial state $\ket{\psi(\boldsymbol\theta)} = V(\boldsymbol\theta)\ket0$, let $\ket\phi = A\ket{\psi}$. The \emph{global} cost is the infidelity of $\ket\phi$ with $\ket b$,}
\begin{equation}\label{eq:ch12:global}
    C_G(\boldsymbol\theta) = 1 - \frac{|\braket{b|\phi}|^2}{\braket{\phi|\phi}} = \frac{\braket{\phi|(\one - \ket b\!\bra b)|\phi}}{\braket{\phi|\phi}},
\end{equation}
\ona{which is zero if and only if $A\ket\psi \propto \ket b$. Global costs suffer from barren plateaus (\Chap~\chref{C.VQE}), and \citet{bravoprieto2023vqls} introduce the \emph{local} cost}
\begin{equation}\label{eq:ch12:local}
    C_L(\boldsymbol\theta) = \frac{\braket{\phi|H_L|\phi}}{\braket{\phi|\phi}},\qquad H_L = U_b\Big(\one - \frac1n\sum_{j=1}^n\ket0\!\bra0_j\otimes\one_{\bar j}\Big)U_b^\dagger,
\end{equation}
\ona{whose evaluation, after undoing $U_b$, needs only the probability that each qubit reads $0$.}
\begin{propn}\label{propn:ch12:costs}
$0\le C_L\le C_G\le nC_L$. In particular, $C_L$ and $C_G$ vanish at the same states, those with $A\ket\psi\propto\ket b$.
\end{propn}
\ona{\emph{Proof.} Let $\ket\chi = U_b^\dagger\ket\phi/\|\phi\|$, $P_0 = \ket{0^n}\!\bra{0^n}$ and $P_j = \ket0\!\bra0_j\otimes\one_{\bar j}$. Then $C_G = \braket{\chi|\one-P_0|\chi}$ and $C_L = \braket{\chi|\one-\frac1n\sum_jP_j|\chi}$. All these projectors are diagonal in the computational basis, so it suffices to compare their diagonals. A string other than $0^n$ has a $1$ in some position, so $\one-P_0\le\sum_j(\one-P_j) = n(\one-\frac1n\sum_jP_j)$, which gives $C_G\le nC_L$; and $P_0\le P_j$ for every $j$, so $P_0\le\frac1n\sum_jP_j$, which gives $C_L\le C_G$. \qed}
\onp{Proof: with $\ket\chi = U_b^\dagger\ket\phi/\|\phi\|$, compare the diagonal projectors $\one-\ket{0^n}\!\bra{0^n}$ and $\one-\frac1n\sum_j\ket0\!\bra0_j$.}
\end{frame}

\begin{frame}\ft{The Hadamard test}
\onp{\footnotesize}
\ona{Expanding $\ket\phi = \sum_lc_lA_l\ket\psi$, the two costs are sums of matrix elements of unitaries,}
\begin{equation}\label{eq:ch12:terms}
    \braket{\phi|\phi} = \sum_{l,l'}c_lc_{l'}\,\mathrm{Re}\braket{\psi|A_lA_{l'}|\psi},\qquad
    \braket{b|\phi} = \sum_lc_l\braket{0|U_b^\dagger A_lV|0},
\end{equation}
\ona{and each is estimated on a quantum computer by the following test.}
\begin{lem}\label{lem:ch12:hadamard}
Prepare an ancilla in $\ket0$ and $\ket\psi$, apply $H$ to the ancilla, then $U$ to $\ket\psi$ controlled by the ancilla, then $H$ to the ancilla, and measure it. Then $\mathrm{Pr}[0]-\mathrm{Pr}[1] = \mathrm{Re}\braket{\psi|U|\psi}$; with $S^\dagger$ applied to the ancilla before the controlled $U$, it is $\mathrm{Im}\braket{\psi|U|\psi}$.
\end{lem}
\ona{\emph{Proof.} After the controlled $U$ the state is $(\ket0\ket\psi+\ket1U\ket\psi)/\sqrt2$, and after the second $H$ it is $\frac12\ket0(\ket\psi+U\ket\psi)+\frac12\ket1(\ket\psi-U\ket\psi)$. Hence $\mathrm{Pr}[0] = \frac14\|\psi+U\psi\|^2 = \frac12(1+\mathrm{Re}\braket{\psi|U|\psi})$ and $\mathrm{Pr}[1] = \frac12(1-\mathrm{Re}\braket{\psi|U|\psi})$. The phase $S^\dagger$ multiplies the second branch by $-i$, and $\mathrm{Re}(-iz) = \mathrm{Im}\,z$. \qed}
\onp{Proof: the ancilla interferes $\ket\psi$ with $U\ket\psi$; $\mathrm{Pr}[0] = \frac12(1+\mathrm{Re}\braket{\psi|U|\psi})$.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item $\braket{\phi|\phi}$: $L(L+1)/2$ tests with $U = A_lA_{l'}$; the local cost needs as many for each of the $n$ projectors.
    \item Each test is estimated with copies, with the noise and bias of \Chap~\chref{C.IterationComplexity}; gradients follow from the shift rule of \Chap~\chref{C.PSR} applied to each test.
\end{itemize}
\end{frame}

\begin{frame}\ft{Ansatz, optimizer, and reported scaling}
\onp{\footnotesize}
\begin{itemize}\setlength\itemsep{0.3em}
    \item \textbf{Ansatz.} Hardware-efficient layers of $R_Y$ rotations and CZ gates for generic $A$; a problem-inspired alternating operator ansatz built from the $A_l$ and $U_b$ when $A$ has structure, as in QAOA.
    \item \textbf{Optimizer.} Gradient descent with gradients from the shift rule, or COBYLA; the local cost has larger gradients than the global one for shallow circuits\ona{, as Section~\ref{sec:ch12:plateau} shows}.
    \item \textbf{Reported scaling.} On Ising-inspired systems with up to $50$ qubits, simulated with tensor networks, \citet{bravoprieto2023vqls} observed a time to a fixed precision that grows linearly in $\kappa$, logarithmically in $1/\epsilon$ and polylogarithmically in $N$, for that family.
\end{itemize}
\end{frame}

\section{A worked example}\label{sec:ch12:example}

\begin{frame}\ft{An Ising-inspired system}
\onp{\footnotesize}
\ona{Following \citet{bravoprieto2023vqls}, we take}
\begin{equation}\label{eq:ch12:ising}
    A = \frac1\zeta\Big(\sum_{j=1}^nX_j+J\sum_{j=1}^{n-1}Z_jZ_{j+1}+\eta\one\Big),\qquad J = 0.1,\qquad\ket b = H^{\otimes n}\ket0 = \ket+^{\otimes n},
\end{equation}
\ona{with $\eta$ and $\zeta$ chosen so that the eigenvalues of $A$ fill $[1/\kappa,1]$. For $n = 3$ and $\kappa = 10$, the Ising part has the eigenvalues from $-3.005$ to $3.005$, so that $\eta = 3.6728$ and $\zeta = 6.6778$, and $A$ has $L = 6$ terms: the three $X_j$ with the coefficient $0.1498$, the two $Z_jZ_{j+1}$ with $0.0150$, and the identity with $0.5500$. The solution $\ket x$ has the amplitudes between $0.328$ and $0.381$. It is close to $\ket b$, with $\braket{b|x} = 0.9986$, because $\ket b$ is the top eigenvector of $\sum_jX_j$ and the couplings $J$ perturb it only a little. The ansatz $V(\boldsymbol\theta)$ has three layers of $R_Y$ rotations on all qubits, separated by CZ gates on the pairs $(1,2)$ and $(2,3)$ alternately, and a final layer: $12$ angles. All computations are exact simulations by \texttt{code/ch12\_figures.py}.}
\onp{\begin{itemize}\setlength\itemsep{0.2em}
    \item $n = 3$, $\kappa = 10$: $\eta = 3.6728$, $\zeta = 6.6778$; $L = 6$ terms; the eigenvalues of $A$ fill $[0.1,1]$.
    \item $\braket{b|x} = 0.9986$: $\ket b$ is the top eigenvector of $\sum_jX_j$, and $J = 0.1$ is small.
    \item Ansatz: three layers of $R_Y$ and CZ, and a final layer of $R_Y$; $12$ angles.
\end{itemize}}
\end{frame}

\section{Certificates}\label{sec:ch12:cert}

\begin{frame}\ft{The cost certifies the solution}
\onp{\footnotesize}
\ona{Let $\kappa = \|A\|\,\|A^{-1}\|$ be the condition number of $A$, and let $D(\psi,x) = \sqrt{1-|\braket{x|\psi}|^2}$ be the trace distance between the pure states $\ket\psi$ and $\ket x$.}
\begin{propn}\label{propn:ch12:cert}
For every state $\ket\psi$,
\begin{equation}\label{eq:ch12:cert}
    \frac{\sqrt{C_G}}\kappa\le D(\psi,x)\le\kappa\sqrt{C_G},
\end{equation}
and hence $D(\psi,x)\le\kappa\sqrt{nC_L}$.
\end{propn}
\ona{\emph{Proof.} Write $\ket\psi = \alpha\ket x+\beta\ket{x^\perp}$, with $\ket{x^\perp}$ a unit vector orthogonal to $\ket x$, so that $D = |\beta|$, and let $P_b^\perp = \one-\ket b\!\bra b$, $P_x^\perp = \one-\ket x\!\bra x$. Since $A\ket x\propto\ket b$, the component of $\ket\phi = A\ket\psi$ along $\ket b$ is mapped by $A^{-1}$ to a multiple of $\ket x$, and
\begin{equation*}
    D = \|P_x^\perp\psi\| = \|P_x^\perp A^{-1}P_b^\perp\phi\|\le\|A^{-1}\|\,\|P_b^\perp\phi\| = \|A^{-1}\|\sqrt{C_G}\,\|\phi\|\le\|A^{-1}\|\,\|A\|\sqrt{C_G}.
\end{equation*}
Conversely, $P_b^\perp\phi = \beta P_b^\perp A\ket{x^\perp}$, so $\sqrt{C_G}\,\|\phi\| = \|P_b^\perp\phi\|\le D\,\|A\|$, while $\|\phi\| = \|A\psi\|\ge\|A^{-1}\|^{-1}$; hence $\sqrt{C_G}\le\kappa D$. The last bound follows from Proposition~\ref{propn:ch12:costs}. \qed}
\onp{Proof: $D = \|P_x^\perp A^{-1}P_b^\perp\phi\|\le\|A^{-1}\|\sqrt{C_G}\|\phi\|$, and $\sqrt{C_G}\|\phi\| = |\beta|\,\|P_b^\perp Ax^\perp\|\le D\|A\|$.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item The cost is a certificate: $C_G\le\epsilon$ guarantees $D\le\kappa\sqrt\epsilon$, whatever the optimizer did.
    \item To guarantee $D\le\epsilon$, the cost must be driven, and estimated, below $\epsilon^2/\kappa^2$.
\end{itemize}
\end{frame}

\begin{frame}\ft{A ground-state problem}
\onp{\footnotesize}
\ona{The cost \eqref{eq:ch12:global} is the generalised Rayleigh quotient $C_G = \braket{\psi|H_G|\psi}/\braket{\psi|A^\dagger A|\psi}$ of the Hamiltonian}
\begin{equation}\label{eq:ch12:HG}
    H_G = A^\dagger(\one - \ket b\!\bra b)A.
\end{equation}
\begin{propn}\label{propn:ch12:gap}
$H_G\succeq0$ has the ground state $\ket x$, with the eigenvalue $0$, and its next eigenvalue is at least $\lambda_{\min}(A^\dagger A) = \|A^{-1}\|^{-2}$, which is $1/\kappa^2$ if $\|A\| = 1$.
\end{propn}
\ona{\emph{Proof.} $H_G = A^\dagger A-\ket a\!\bra a$ with $\ket a = A^\dagger\ket b$, and $H_G\ket x\propto A^\dagger(\one-\ket b\!\bra b)\ket b = 0$. By the Courant--Fischer theorem, the second eigenvalue is the maximum over vectors $u$ of the minimum of the Rayleigh quotient of $H_G$ over unit vectors orthogonal to $u$. Choosing $u = a$, the rank-one term vanishes on these vectors, and the minimum is at least $\lambda_{\min}(A^\dagger A)$. \qed}
\ona{\citet{zhang2022computing} exploit this. Instead of optimising an ansatz to the ground state, they prepare any state with an overlap of order one with $\ket x$, by a constant-precision run of an adiabatic linear-systems solver, which takes $\widetilde{\mathcal O}(\kappa)$ time. They then estimate $\braket{x|M|x}$ by ground-state property estimation, evaluating the $M$-weighted cumulative distribution function directly at the energy zero, where the gap \eqref{eq:ch12:HG} separates $\ket x$ from the rest of the spectrum.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item Gap amplification: $\sigma^+\otimes A^\dagger(\one-\ket b\!\bra b)+\sigma^-\otimes(\one-\ket b\!\bra b)A$ has the eigenvalues $\pm\sqrt{\lambda_j(H_G)}$, so the gap improves from $\kappa^{-2}$ to $\kappa^{-1}$.
    \item The result: depth $\widetilde{\mathcal O}(\kappa)$ and expected time $\widetilde{\mathcal O}(\kappa\epsilon^{-2}\alpha^2)$ for an $\alpha$-block-encoded $M$, with a guarantee on $\epsilon$, no optimizer and no local minima.
\end{itemize}
\end{frame}

\section{Global and local plateaus}\label{sec:ch12:plateau}

\begin{frame}\ft{Barren plateaus, global and local}
\onp{\footnotesize}
\ona{The proposition above makes the two costs equivalent as certificates, but not as landscapes. \citet{McClean_2018} showed that, for parametrised circuits random enough to form approximate $2$-designs, the variance of every derivative of a cost vanishes exponentially in the number of qubits; \citet{Cerezo_2021_costfunct} showed that global costs, which compare all $n$ qubits at once, have such barren plateaus already for shallow circuits, while local costs, which compare one qubit at a time, have derivatives whose variance vanishes at most polynomially in $n$ when the depth grows as $\mathcal O(\log n)$. The cost $C_L$ was designed for this. Figure~\ref{fig:ch12:plateau} measures it on the family \eqref{eq:ch12:ising}, with four layers, for $n = 2,\dots,12$: the variance over $400$ random angles of the derivative of each cost with respect to the first angle.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item Global costs: the variance of the derivatives vanishes exponentially in $n$ \citep{McClean_2018,Cerezo_2021_costfunct}.
    \item Local costs with logarithmic depth: at most polynomially \citep{Cerezo_2021_costfunct}.
    \item A derivative of standard deviation $s$ needs about $1/s^2$ copies to be resolved: the plateau multiplies the copies.
\end{itemize}
\end{frame}

\begin{frame}\ft{Against conjugate gradients}
\ona{Conjugate gradients \citep{hestenes1952methods} solve a positive definite system with $\kappa$ by reducing the error in the norm of $A$ by the factor $2((\sqrt\kappa-1)/(\sqrt\kappa+1))^k$ after $k$ iterations, each a product with $A$ \citep{saad2003iterative}: about $\frac12\sqrt\kappa\ln(2/\epsilon)$ iterations for the relative error $\epsilon$. Figure~\ref{fig:ch12:kappa} counts iterations on the family \eqref{eq:ch12:ising}. On $n = 10$ qubits, that is $1024$ unknowns, conjugate gradients reach $C_G\le10^{-8}$ in $3$ iterations at every $\kappa$ for $\ket b = \ket+^{\otimes10}$, since $\ket x$ is almost $\ket b$, and in $6$ to $21$ iterations for a random $\ket b$, as $\kappa$ grows from $2$ to $500$, below the bound. VQLS on $n = 4$ qubits, with six layers and BFGS on the exact local cost, reaches $C_G\le10^{-6}$ in a median of $50$ iterations at $\kappa = 2$ and $73$ at $\kappa = 100$.}
\ona{The comparison of iterations is not a comparison of cost. An iteration of conjugate gradients is one product with $A$, of the order of $nN$ operations here. An iteration of VQLS needs $L(L+1)/2$ Hadamard tests for each term of each cost, $21$ in the worked example, each with $M$ copies, and twice as many for every angle of a gradient. By Proposition~\ref{propn:ch12:cert}, guaranteeing $D\le\epsilon$ requires $C_G\le\epsilon^2/\kappa^2$, estimated to that precision: for $\epsilon = 0.01$, $10^{-6}$ at $\kappa = 10$ and $10^{-8}$ at $\kappa = 100$, which Hoeffding's inequality turns into about $10^{12}$ and $10^{16}$ copies. Where VQLS may pay off is where the classical algorithm cannot run at all: $N = 2^n$ too large to store, $\ket b$ available only as a state, and only a few expectations of $\ket x$ needed, as \citet{bravoprieto2023vqls} argue; the honest comparison is then with quantum-inspired and Monte Carlo methods, instance family by instance family (\Chap~\chref{C.EQA}).}

\section{Assessment}

\begin{frame}\ft{What can and cannot be claimed}
\onp{\footnotesize}
\begin{itemize}\setlength\itemsep{0.3em}
    \item \textbf{Certificates.} The cost certifies the solution, $D\le\kappa\sqrt{C_G}$, and on the worked example the bound is attained: guaranteeing $D\le\epsilon$ means estimating the cost below $\epsilon^2/\kappa^2$.
    \item \textbf{Trainability.} The local cost avoids the barren plateau of the global one at shallow depth, as \ona{Fig.~\ref{fig:ch12:plateau}}\onp{the plateau figure} shows up to $n = 12$; problem-inspired ans\"atze face the local minima of QAOA (\Chap~\chref{C.QAOA}).
    \item \textbf{Cost per iteration.} $\mathcal O(L^2)$ Hadamard tests, each with $M$ copies, and twice the angles for a gradient: the accounting of \Chap~\chref{C.PerIteration}, with its latency, applies verbatim.
    \item \textbf{Classical baseline.} Conjugate gradients need $\mathcal O(\sqrt\kappa\log(1/\epsilon))$ products with $A$, and on the Ising-inspired family with $\ket b = \ket+^{\otimes n}$ only three; quantum-inspired sampling algorithms match the polylogarithmic scaling in $N$ for low-rank $A$. The comparison must be made per instance family (\Chap~\chref{C.EQA}).
    \item \textbf{Where it may pay off.} When $\ket b$ is available only as a quantum state, $N$ is too large to store, and only a few expectations of $\ket x$ are needed, as a subroutine of a larger quantum algorithm.
\end{itemize}
\ona{The template is the same as in every \chap{} of this course: a cost with a certificate at zero, a circuit family, a noisy optimizer, and a classical competitor that must be beaten instance by instance.}
\end{frame}
