\chapter{The Parameter-Shift Rule}\label{C.PSR}

\ona{This \chap{} is concerned with the derivatives of the cost of a variational algorithm. Several of the optimizers of \Chap~\chref{C.QAOA} require gradients, and the worked example of \Chap~\chref{C.VQE} obtained one from sixteen circuits with shifted angles, by a rule that was stated there without proof. We prove the rule, the \emph{parameter-shift rule}, and extend it in three directions: to gates whose generators have more than two eigenvalues, to gates whose generator does not commute with the rest of the gate, and to second derivatives. Following \citet{sakos2026global}, we then write the cost as a trigonometric polynomial in all the parameters, which can be reconstructed from its values on a grid and minimised globally when the number of parameters is constant. We then examine what the estimates cost in measured copies, how they compare with finite differences, what the noise of a device does to them, and how the same structure gives an exact minimisation along one coordinate. The \chap{} ends with two derivative-free estimators, simultaneous perturbation and Gaussian smoothing, which use two circuits whatever the number of parameters. The iteration complexity of all these methods under noisy and biased estimates is the subject of \Chap~\chref{C.IterationComplexity}. The numbers quoted in this \chap{} are computed by \texttt{code/ch07\_figures.py}, on the worked example of \Chap~\chref{C.VQE}.}

\section{Derivatives from estimates}\label{sec:ch07:estimates}

\begin{frame}\ft{The setting}
\onp{\footnotesize}
\ona{We use the form of the variational loop of \Chap~\chref{C.VQE}. A circuit with $d$ parameters prepares the state}
\begin{equation}\label{eq:ch07:ansatz}
    \ket{\psi(\boldsymbol\theta)} = U_d(\theta_d)\cdots U_1(\theta_1)\ket{\psi_0},\qquad U_j(\theta_j) = e^{-i\theta_jH_j},
\end{equation}
\ona{and the optimizer minimises the cost}
\begin{equation}\label{eq:ch07:cost}
    C(\boldsymbol\theta) = \braket{\psi(\boldsymbol\theta)|O|\psi(\boldsymbol\theta)},\qquad\boldsymbol\theta\in\R^d.
\end{equation}
\ona{In the notation of \Chap~\chref{C.VQE}, $d$ is the number $L$ of layers; we keep the letter $L$ for a Lipschitz constant, as in \Chap~\chref{C.IterationComplexity}. Fixed gates between the layers are absorbed into the generators and the observable, as a fixed initial state was absorbed in \Chap~\chref{C.VQE}. A Pauli rotation $R_P(\theta) = e^{-i\theta P/2}$ has the generator $P/2$, with the eigenvalues $\pm\frac12$; a parameter that controls several gates, as $\gamma$ controls all the $ZZ$ rotations of a layer of QAOA, has the sum of their generators. The cost is available only through estimates. We assume that $O$ is measured in its eigenbasis, as the cost Hamiltonian of QAOA is measured in the computational basis; a sum of Pauli strings that do not commute is measured in several groups, and the variances of the groups add. A measurement then returns a random value with mean $C(\boldsymbol\theta)$ and variance}
\begin{equation}\label{eq:ch07:var}
    \sigma^2(\boldsymbol\theta) = \braket{\psi(\boldsymbol\theta)|O^2|\psi(\boldsymbol\theta)} - C(\boldsymbol\theta)^2,
\end{equation}
\ona{and the mean $\hat C(\boldsymbol\theta)$ of $M$ such values, obtained from $M$ copies of the state, each prepared afresh and measured once, is an unbiased estimate of $C(\boldsymbol\theta)$ with variance $\sigma^2(\boldsymbol\theta)/M$.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item An estimator of a derivative is a combination of such estimates at several points. Its cost has two parts: the number of circuits submitted, and the number $N$ of copies measured in all.
    \item In the worked example of \Chap~\chref{C.VQE}, with $d = 2$, an iteration required $17$ circuits and $17\,000$ copies, and an estimated energy had a standard error of $0.03$.
    \item We compare estimators by their bias and variance for a given $N$, and by the number of circuits.
\end{itemize}
\end{frame}

\begin{frame}\ft{Why a circuit is not differentiated as a program}
\onp{\footnotesize}
\ona{On a classical computer, the gradient of a function computed by a program can be obtained by reverse-mode automatic differentiation, at a cost of a small constant multiple of the cost of one evaluation of the function, whatever the number of parameters \citep{griewank2008evaluating}. The method stores the intermediate values of the forward computation and propagates derivatives backwards through them. A quantum circuit does not permit this. Its intermediate states cannot be copied, and measuring them destroys them. \citet{abbas2023backpropagation} show that, with access to a single copy of the state at a time, the scaling of reverse-mode differentiation cannot be achieved; with several copies at once it can be, in the quantum resources, by means of shadow tomography. The methods of this \chap{} use one copy at a time, and their cost grows with the number of parameters: the parameter-shift rule requires two circuits for each parameter, and the derivative-free estimators of Section~\ref{sec:ch07:dfo}, which require two circuits in all, pay for this with a variance that grows with the number of parameters.}
\begin{itemize}\setlength\itemsep{0.25em}
    \item Reverse mode: a constant multiple of the cost of one evaluation, for any $d$ \citep{griewank2008evaluating}.
    \item It stores the intermediate values. Intermediate quantum states can be neither copied nor stored, and measuring them destroys them.
    \item With one copy at a time, the scaling of reverse mode cannot be achieved \citep{abbas2023backpropagation}.
    \item Hence $2d$ circuits for the parameter-shift rule, or two circuits with a large variance for the derivative-free estimators.
\end{itemize}
\end{frame}

\begin{frame}\ft{Finite differences under shot noise}
\onp{\footnotesize}
\ona{The simplest estimator of a partial derivative is the central difference. Fix all the parameters but one, write $f(\theta)$ for the cost as a function of the remaining one, and $\sigma_\pm^2$ for the variances \eqref{eq:ch07:var} at $\theta\pm h$. With $M$ copies at each of the two points, Taylor's theorem gives}
\begin{equation}\label{eq:ch07:fd}
    \hat g_h = \frac{\hat f(\theta+h)-\hat f(\theta-h)}{2h},\qquad \mathbb E\,\hat g_h - f'(\theta) = \frac{h^2}6f'''(\theta) + \mathcal O(h^4),\qquad \mathrm{Var}\,\hat g_h = \frac{\sigma_+^2+\sigma_-^2}{4h^2M}.
\end{equation}
\ona{The bias decreases with $h$, and the variance increases as $h$ decreases. If $\sigma_\pm\approx\sigma$, the mean squared error $\frac{h^4}{36}f'''(\theta)^2 + \frac{\sigma^2}{2h^2M}$ is least at $h^6 = 9\sigma^2/(Mf'''(\theta)^2)$, where it is $\frac1{12}f'''(\theta)^2h^4$, of the order of $M^{-2/3}$.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item The best step depends on the third derivative, which is not known.
    \item At the best step, the error decreases as $M^{-1/3}$, not as $M^{-1/2}$.
    \item A step of $10^{-7}$, usual in floating-point arithmetic, gives a variance of the order of $10^{14}\sigma^2/M$: the two estimates differ by far less than their errors.
\end{itemize}
\ona{The parameter-shift rule removes the dilemma: for the functions that circuits compute, a difference with a step of order one, suitably rescaled, is exact.}
\end{frame}

\section{The parameter-shift rule}\label{sec:ch07:psr}

\begin{frame}\ft{One parameter at a time}
\onp{\footnotesize}
\ona{To differentiate with respect to $\theta_j$, absorb the later layers into the observable and the earlier ones into the state:}
\begin{equation}\label{eq:ch07:heis}
    C(\boldsymbol\theta) = \braket{\phi_j|e^{i\theta_jH_j}O_j\,e^{-i\theta_jH_j}|\phi_j},\qquad \ket{\phi_j} = U_{j-1}\cdots U_1\ket{\psi_0},\quad O_j = U_{j+1}^\dagger\cdots U_d^\dagger\,O\,U_d\cdots U_{j+1}.
\end{equation}
\ona{As a function of $\theta_j$ alone, the cost therefore has the form $f(\theta) = \braket{\phi|e^{i\theta H}O'e^{-i\theta H}|\phi}$, for a state $\ket\phi$, a Hermitian generator $H$ and a Hermitian observable $O'$ that do not depend on $\theta$. Neither $\ket\phi$ nor $O'$ is known explicitly: the circuit prepares the one and measures the other.}
\onp{As a function of $\theta_j$ alone: $f(\theta) = \braket{\phi|e^{i\theta H}O'e^{-i\theta H}|\phi}$, with $\ket\phi$ and $O'$ prepared and measured by the circuit, not known.}
\begin{lem}\label{lem:ch07:freq}
Let $H = \sum_{k=1}^m\lambda_k\Pi_k$ be the spectral decomposition of $H$, with distinct eigenvalues $\lambda_k$ and orthogonal projectors $\Pi_k$. Then
\begin{equation}\label{eq:ch07:freq}
    f(\theta) = \sum_{k,l=1}^m c_{kl}\,e^{i(\lambda_k-\lambda_l)\theta},\qquad c_{kl} = \braket{\phi|\Pi_kO'\Pi_l|\phi},\quad c_{lk} = \overline{c_{kl}}.
\end{equation}
In particular, if $H$ has exactly two eigenvalues, which differ by $\Omega>0$, then $f(\theta) = a + b\cos\Omega\theta + c\sin\Omega\theta$ for real $a$, $b$, $c$.
\end{lem}
\ona{\emph{Proof.} Insert $e^{-i\theta H} = \sum_le^{-i\theta\lambda_l}\Pi_l$ and its adjoint. With two eigenvalues, the terms with $k = l$ give $a = c_{11}+c_{22}$, and the other two give $2\,\mathrm{Re}(c_{12}e^{i\Omega\theta})$. \qed}

\ona{The frequencies of $f$ are the differences of the eigenvalues of the generator. When they are integers, $f$ is a Hermitian trigonometric polynomial in the sense of \Chap~\chref{C.VQE}. For a Pauli rotation, $\Omega = 1$, and the cost is a sinusoid of period $2\pi$ in the angle of the gate.}
\end{frame}

\begin{frame}\ft{The parameter-shift rule}
\onp{\footnotesize}
\begin{thm}[\citet{Mitarai_2018,schuld2019evaluating}]\label{thm:ch07:psr}
If $H$ has exactly two eigenvalues, which differ by $\Omega>0$, then for every $\theta$ and every $s$ with $\sin\Omega s\ne0$,
\begin{equation}\label{eq:ch07:psrs}
    f'(\theta) = \Omega\,\frac{f(\theta+s)-f(\theta-s)}{2\sin\Omega s}.
\end{equation}
In particular, for a Pauli rotation $R_P(\theta) = e^{-i\theta P/2}$,
\begin{equation}\label{eq:ch07:psr}
    f'(\theta) = \tfrac12\big[f(\theta+\tfrac\pi2)-f(\theta-\tfrac\pi2)\big].
\end{equation}
\end{thm}
\ona{\emph{Proof.} By Lemma~\ref{lem:ch07:freq}, $f(\theta) = a + b\cos\Omega\theta + c\sin\Omega\theta$, and the addition theorems give $f(\theta+s)-f(\theta-s) = 2\sin\Omega s\,(c\cos\Omega\theta - b\sin\Omega\theta) = \frac{2\sin\Omega s}\Omega f'(\theta)$. \qed}
\onp{Proof: $f = a + b\cos\Omega\theta + c\sin\Omega\theta$, and $f(\theta+s)-f(\theta-s) = 2\sin\Omega s\,(c\cos\Omega\theta-b\sin\Omega\theta) = \frac{2\sin\Omega s}{\Omega}f'(\theta)$.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item The rule is exact. It is a statement about expectations, and the shift $\pi/2$ is of order one.
    \item For such $f$, the central difference \eqref{eq:ch07:fd} is $\frac{\sin\Omega h}{\Omega h}f'(\theta)$; the rule is the central difference with this factor removed.
    \item With $M$ copies at each point, the variance of the estimate is $\Omega^2(\sigma_+^2+\sigma_-^2)/(4M\sin^2\Omega s)$, least at $\Omega s = \pi/2$; for a Pauli rotation it is $(\sigma_+^2+\sigma_-^2)/(4M)$.
\end{itemize}
\end{frame}

\begin{frame}\ft{A second proof, by the commutator}
\onp{\footnotesize}
\ona{The rule can also be derived without the spectral decomposition. The derivation is needed in Section~\ref{sec:ch07:general}.}
\begin{lem}\label{lem:ch07:comm}
For a Pauli string $P$ and every operator $O$,
\begin{equation}\label{eq:ch07:comm}
    i[P,O] = R_P(\tfrac\pi2)^\dagger\,O\,R_P(\tfrac\pi2) - R_P(-\tfrac\pi2)^\dagger\,O\,R_P(-\tfrac\pi2).
\end{equation}
\end{lem}
\ona{\emph{Proof.} Since $P^2 = \one$, $R_P(\pm\frac\pi2) = (\one\mp iP)/\sqrt2$, and so $R_P(\pm\frac\pi2)^\dagger OR_P(\pm\frac\pi2) = \frac12(\one\pm iP)O(\one\mp iP) = \frac12\big(O + POP \pm i[P,O]\big)$. The difference of the two is $i[P,O]$. \qed}

\ona{For $f(\theta) = \braket{\phi|R_P(\theta)^\dagger O'R_P(\theta)|\phi}$, differentiation gives $f'(\theta) = \braket{\phi|R_P(\theta)^\dagger\,\frac i2[P,O']\,R_P(\theta)|\phi}$. By the lemma, and since $R_P(\pm\frac\pi2)R_P(\theta) = R_P(\theta\pm\frac\pi2)$, this is \eqref{eq:ch07:psr}.}
\onp{Since $\frac{d}{d\theta}R_P(\theta)^\dagger O'R_P(\theta) = R_P(\theta)^\dagger\,\frac i2[P,O']\,R_P(\theta)$ and $R_P(\pm\frac\pi2)R_P(\theta) = R_P(\theta\pm\frac\pi2)$, the lemma gives \eqref{eq:ch07:psr}.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item The derivative is the expectation of the observable $\frac i2[P,O']$ in the state at $\theta$. It could be measured with an additional qubit and controlled gates; the rule requires neither.
    \item The shifted circuits are the original circuit with one angle changed. No new gates are compiled, and the same calibration serves.
\end{itemize}
\end{frame}

\begin{frame}\ft{Assembling the gradient}
\onp{\footnotesize}
\ona{For a circuit of Pauli rotations, \eqref{eq:ch07:heis} and \eqref{eq:ch07:psr} give every partial derivative from two circuits,}
\begin{equation}\label{eq:ch07:grad}
    \frac{\partial C}{\partial\theta_j}(\boldsymbol\theta) = \tfrac12\big[C(\boldsymbol\theta+\tfrac\pi2\boldsymbol e_j) - C(\boldsymbol\theta-\tfrac\pi2\boldsymbol e_j)\big],\qquad j = 1,\dots,d,
\end{equation}
\ona{and so the gradient from $2d$ circuits. If a parameter controls several gates, as in QAOA, each gate is given its own angle $\phi_g$, which depends on the parameter, and the chain rule gives}
\begin{equation}\label{eq:ch07:chain}
    \frac{\partial C}{\partial\theta_j} = \sum_g\frac{\partial\phi_g}{\partial\theta_j}\cdot\tfrac12\big[C_{g+} - C_{g-}\big],
\end{equation}
\ona{where $C_{g\pm}$ is the cost with the angle of gate $g$ alone shifted by $\pm\frac\pi2$. This is the computation of the worked example of \Chap~\chref{C.VQE}: $\gamma$ is the angle of four gates $R_Z(\gamma)$ and $2\beta$ that of four gates $R_X(2\beta)$, so that $\partial\phi_g/\partial\beta = 2$, and the gradient required $16$ circuits.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item For QAOA of depth $p$ on a graph $(V,E)$, gate by gate: $2p(|E|+|V|)$ circuits for a gradient of $2p$ parameters.
    \item \ona{Section~\ref{sec:ch07:general}}\onp{The next section} gives rules that treat a parameter of several gates as one, with fewer circuits.
    \item The estimate of each term is unbiased, and so is the sum. Its variance is the sum of the variances of the terms, since the circuits are measured independently.
\end{itemize}
\end{frame}

\begin{frame}\ft{Second derivatives}
\onp{\footnotesize}
\ona{Applying \eqref{eq:ch07:psr} to each partial derivative \eqref{eq:ch07:grad} gives the second derivatives \citep{Mari_2021}. For $j\ne k$,}
\begin{equation}\label{eq:ch07:hessmixed}
    \frac{\partial^2C}{\partial\theta_j\partial\theta_k} = \tfrac14\big[C(\boldsymbol\theta+\tfrac\pi2\boldsymbol e_j+\tfrac\pi2\boldsymbol e_k) - C(\boldsymbol\theta+\tfrac\pi2\boldsymbol e_j-\tfrac\pi2\boldsymbol e_k) - C(\boldsymbol\theta-\tfrac\pi2\boldsymbol e_j+\tfrac\pi2\boldsymbol e_k) + C(\boldsymbol\theta-\tfrac\pi2\boldsymbol e_j-\tfrac\pi2\boldsymbol e_k)\big].
\end{equation}
\ona{For $j = k$, the sinusoid $f = a + b\cos\theta + c\sin\theta$ satisfies $f'' = a - f$ and $f(\theta+\pi) = 2a - f(\theta)$, and so}
\begin{equation}\label{eq:ch07:hessdiag}
    \frac{\partial^2C}{\partial\theta_j^2} = \tfrac12\big[C(\boldsymbol\theta+\pi\boldsymbol e_j) - C(\boldsymbol\theta)\big].
\end{equation}
\begin{itemize}\setlength\itemsep{0.2em}
    \item The Hessian of $d$ parameters requires $4\binom d2 + d + 1 = 2d^2-d+1$ circuits, the value $C(\boldsymbol\theta)$ included. For $d = 20$ this is $781$ circuits.
    \item Higher derivatives follow in the same way; each order multiplies the number of circuits by about $2d$.
    \item The Newton, trust-region and cubic-regularisation methods of \Chap~\chref{C.QAOA} need these estimates, and they need them more accurately than gradients, since the Hessian is inverted.
\end{itemize}
\end{frame}

\begin{frame}\ft{The frequencies of common gates}
\onp{\footnotesize}
\ona{By Lemma~\ref{lem:ch07:freq}, the frequencies of the cost in one parameter are the positive differences of the eigenvalues of its generator. Table~\ref{tab:ch07:freq} lists them for the gates of \Chaps~\chref{C.VQE} and~\chref{C.QAOA}. For MaxCut, $\mathrm{MC}$ denotes the maximum cut; in the worked example of \Chap~\chref{C.VQE}, on the 4-cycle, the cut values are $0$, $2$ and $4$, so that the frequencies in $\gamma$ are $2$ and $4$, and $R = 2$ with $\Omega_0 = 2$.}

\begin{frame}\ft{The general parameter-shift rule}
\onp{\footnotesize}
\begin{thm}[\citet{Wierichs2021}]\label{thm:ch07:general}
Let $f(x) = a_0 + \sum_{\ell=1}^R(a_\ell\cos\ell x + b_\ell\sin\ell x)$ with real coefficients, and let $x_\mu = \frac{(2\mu-1)\pi}{2R}$ for $\mu = 1,\dots,R$. Then
\begin{equation}\label{eq:ch07:general}
    f'(0) = \sum_{\mu=1}^Ry_\mu\big[f(x_\mu)-f(-x_\mu)\big],\qquad y_\mu = \frac{(-1)^{\mu-1}}{4R\sin^2(x_\mu/2)},
\end{equation}
and $\sum_{\mu=1}^R2|y_\mu| = R$.
\end{thm}
\ona{At another point $x_0$, the rule is applied to $x\mapsto f(x_0+x)$, which has the same frequencies. It requires $2R$ values. For $R = 1$ it is the rule \eqref{eq:ch07:psr}, with $x_1 = \frac\pi2$ and $y_1 = \frac12$. For $R = 2$, as for the excitations of the unitary coupled-cluster ansatz, the shifts are $\frac\pi4$ and $\frac{3\pi}4$, and the weights $y_1 = \frac12(1+\frac1{\sqrt2}) = 0.854$ and $y_2 = -\frac12(1-\frac1{\sqrt2}) = -0.146$; \citet{Kottmann2021} gave a rule with four values for these excitations. For the parameter $\gamma$ of the worked example, with $x = 2\gamma$,}
\onp{$R = 1$: \eqref{eq:ch07:psr}. $R = 2$: shifts $\frac\pi4$, $\frac{3\pi}4$, weights $0.854$, $-0.146$. In the worked example, with $x = 2\gamma$:}
\begin{equation}\label{eq:ch07:c4gamma}
    \frac{\partial F}{\partial\gamma} = \big(1+\tfrac1{\sqrt2}\big)\big[F(\gamma+\tfrac\pi8) - F(\gamma-\tfrac\pi8)\big] - \big(1-\tfrac1{\sqrt2}\big)\big[F(\gamma+\tfrac{3\pi}8) - F(\gamma-\tfrac{3\pi}8)\big],
\end{equation}
\ona{four circuits in place of eight, which give $0.7633$ at $(\beta,\gamma) = (0.1,-0.1)$, the exact value. For $\beta$, with frequencies $2,4,6,8$, the rule with $R = 4$ requires eight circuits, as many as the gates.}
\ona{The formula \eqref{eq:ch07:general} is classical in approximation theory: it is the interpolation formula of M.~Riesz for the derivative of a trigonometric polynomial, from which Bernstein's inequality follows \citep{zygmund2003trigonometric}.}
\end{frame}

\begin{frame}\ft{Proof of the general rule}
\onp{\footnotesize}
\ona{We use the \emph{Chebyshev polynomials}: $T_R$, of degree $R$, with $T_R(\cos x) = \cos Rx$, and $U_{\ell-1}$, of degree $\ell-1$, with $\sin\ell x = \sin x\,U_{\ell-1}(\cos x)$; both identities follow from de Moivre's formula.}
\onp{Chebyshev polynomials: $T_R(\cos x) = \cos Rx$, of degree $R$, and $\sin\ell x = \sin x\,U_{\ell-1}(\cos x)$, with $U_{\ell-1}$ of degree $\ell-1$.}

\ona{\emph{Proof of Theorem~\ref{thm:ch07:general}.}}
\begin{enumerate}\setlength\itemsep{0.2em}
    \item The odd part $g(x) = \frac12[f(x)-f(-x)] = \sum_\ell b_\ell\sin\ell x$ has $g'(0) = f'(0)$, and $g(x) = \sin x\,p(\cos x)$ for a polynomial $p = \sum_\ell b_\ell U_{\ell-1}$ of degree at most $R-1$. Differentiating, $g'(0) = p(1)$.
    \item The points $t_\mu = \cos x_\mu$, $\mu = 1,\dots,R$, are distinct, and $T_R(t_\mu) = \cos\frac{(2\mu-1)\pi}2 = 0$: they are the zeros of $T_R$, the \emph{Chebyshev nodes}. Interpolation at $R$ nodes is exact for $p$, so $p(1) = \sum_\mu p(t_\mu)\ell_\mu(1)$ with the Lagrange polynomials $\ell_\mu(t) = \frac{T_R(t)}{(t-t_\mu)T_R'(t_\mu)}$.
    \item $T_R(1) = 1$ and $1-t_\mu = 2\sin^2\frac{x_\mu}2$, and differentiating $T_R(\cos x) = \cos Rx$ gives $T_R'(t_\mu) = R\sin Rx_\mu/\sin x_\mu = R(-1)^{\mu-1}/\sin x_\mu$. Since $p(t_\mu) = g(x_\mu)/\sin x_\mu$,
    \begin{equation*}
        f'(0) = p(1) = \sum_\mu\frac{g(x_\mu)}{\sin x_\mu}\cdot\frac{(-1)^{\mu-1}\sin x_\mu}{2R\sin^2(x_\mu/2)} = \sum_\mu\frac{(-1)^{\mu-1}}{2R\sin^2(x_\mu/2)}\,g(x_\mu),
    \end{equation*}
    which is \eqref{eq:ch07:general}.
    \item For $f = \sin Rx$, $f'(0) = R$ and $f(\pm x_\mu) = \pm(-1)^{\mu-1}$, the sign of $y_\mu$. The rule gives $R = \sum_\mu2|y_\mu|$.\ona{ \qed}
\end{enumerate}
\end{frame}

\begin{frame}\ft{No rule has smaller weights}
\onp{\scriptsize}
\begin{propn}\label{propn:ch07:l1}
If $f'(0) = \sum_kw_kf(z_k)$ for every real trigonometric polynomial $f$ of degree at most $R$, then $\sum_k|w_k|\ge R$.
\end{propn}
\ona{\emph{Proof.} For $f = \sin Rx$, $R = f'(0) = \sum_kw_k\sin Rz_k\le\sum_k|w_k|$. \qed}
\onp{Proof: take $f = \sin Rx$.}
\begin{lem}\label{lem:ch07:alloc}
If the values $f(z_k)$ are estimated from $N_k$ copies each, with variances $\sigma_k^2$ of a single value, and $\sum_kN_k = N$, then the variance $\sum_kw_k^2\sigma_k^2/N_k$ of the estimate of $f'(0)$ is least for $N_k\propto|w_k|\sigma_k$, where it is $\big(\sum_k|w_k|\sigma_k\big)^2/N$.
\end{lem}
\ona{\emph{Proof.} By the Cauchy--Schwarz inequality, $\big(\sum_k|w_k|\sigma_k\big)^2 = \big(\sum_k\frac{|w_k|\sigma_k}{\sqrt{N_k}}\sqrt{N_k}\big)^2\le\sum_k\frac{w_k^2\sigma_k^2}{N_k}\sum_kN_k$, with equality when $N_k$ is proportional to $|w_k|\sigma_k$. \qed}
\begin{itemize}\setlength\itemsep{0.2em}
    \item With $\sigma_k\le\sigma$, the general rule has variance at most $R^2\sigma^2/N$, and by the proposition no exact rule has a smaller bound. Gate by gate, a parameter of $k$ gates has weights whose absolute values sum to $k\ge R$.
    \item The worked example, $\partial F/\partial\gamma$: with $1000$ copies for each circuit, the standard errors are $0.044$ gate by gate ($8$ circuits) and $0.077$ by \eqref{eq:ch07:c4gamma} ($4$ circuits, equal copies for unequal weights). With copies as in the lemma, $N$ times the variance is $15.4$ and $16.0$.
\end{itemize}
\ona{The general rule thus saves circuits, not copies. \citet{Theis2021} formulates the search for the rule of least variance as a pair of convex optimisation problems, and establishes the optimality of known rules for several families of circuits.}
\end{frame}

\begin{frame}\ft{Gates whose terms do not commute}
\onp{\footnotesize}
\ona{A gate realised by a pulse often has the form $e^{-i(\theta A+B)}$, where only the term $A$ carries the parameter, and a drift term $B$, which need not commute with $A$, acts throughout. Lemma~\ref{lem:ch07:freq} does not apply, since the eigenvectors of $\theta A+B$ depend on $\theta$, and the cost is not a trigonometric polynomial in $\theta$. The commutator form of the rule still gives an estimator.}
\begin{thm}[\citet{Banchi_2021}]\label{thm:ch07:stoch}
Let $A = P/2$ for a Pauli string $P$, let $B$ be Hermitian, and let $f(\theta)$ be the expectation of $O$ in the state $e^{-i(\theta A+B)}\ket\psi$. For $t\in[0,1]$, let $f_\pm(t)$ be the expectation of $O$ in the state $e^{-it(\theta A+B)}R_P(\pm\frac\pi2)e^{-i(1-t)(\theta A+B)}\ket\psi$. Then
\begin{equation}\label{eq:ch07:stoch}
    f'(\theta) = \tfrac12\int_0^1\big[f_+(t)-f_-(t)\big]\,dt.
\end{equation}
\end{thm}
\ona{\emph{Proof.} Write $G = \theta A+B$. The derivative of the exponential is $\frac{d}{d\theta}e^{-iG} = -i\int_0^1e^{-itG}Ae^{-i(1-t)G}\,dt$, which follows on differentiating $s\mapsto e^{-isG(\theta)}e^{-i(1-s)G(\theta_0)}$, integrating over $s\in[0,1]$, dividing by $\theta-\theta_0$ and letting $\theta\to\theta_0$. With $\ket{\phi_t} = e^{-i(1-t)G}\ket\psi$ and $O_t = e^{itG}Oe^{-itG}$, that is, cutting the gate at the fraction $t$ as the circuit was cut in Fig.~\ref{fig:ch07:heis}(a), and as in the derivation of Lemma~\ref{lem:ch07:comm}, this gives $f'(\theta) = \int_0^1\braket{\phi_t|i[A,O_t]|\phi_t}\,dt$, and Lemma~\ref{lem:ch07:comm} with $i[A,O_t] = \frac i2[P,O_t]$ gives \eqref{eq:ch07:stoch}. \qed}

\section{The cost as a trigonometric polynomial}\label{sec:ch07:trig}

\begin{frame}\ft{The cost as a trigonometric polynomial}
\onp{\scriptsize}
\ona{Theorem~\ref{thm:ch07:general} applies to the cost as a function of one parameter, once its degree $R$ is known. \citet{sakos2026global} determine the degree, and the form of the cost as a function of all the parameters, for circuits in which a parameter may control several gates that do not commute. Such a circuit applies $K\ge d$ gates between fixed unitaries $W_k$, the $k$-th gate being controlled by the parameter $\theta_{j_k}$:}
\onp{$K\ge d$ gates, the $k$-th controlled by the parameter $\theta_{j_k}$, between fixed unitaries $W_k$:}
\begin{equation}\label{eq:ch07:shared}
    \ket{\psi(\boldsymbol\theta)} = W_Ke^{-i\theta_{j_K}H_K}\cdots W_1e^{-i\theta_{j_1}H_1}\ket{\psi_0}.
\end{equation}
\ona{We write $\Delta_k$ for the \emph{spectral width} of $H_k$, the difference of its largest and least eigenvalues; a Pauli rotation has $\Delta_k = 1$.}
\begin{propn}[\citet{sakos2026global}]\label{propn:ch07:trig}
Suppose that the eigenvalues of each $H_k$ differ by integers, and let $R_j = \sum_{k:\,j_k = j}\Delta_k$\onp{, where $\Delta_k$ is the spectral width of $H_k$}. Then the cost is a Hermitian trigonometric polynomial,
\begin{equation}\label{eq:ch07:trig}
    C(\boldsymbol\theta) = \sum_{\boldsymbol\ell}c_{\boldsymbol\ell}\,e^{i\boldsymbol\ell^\top\boldsymbol\theta},\qquad c_{-\boldsymbol\ell} = \overline{c_{\boldsymbol\ell}},
\end{equation}
the sum running over the $\boldsymbol\ell\in\mathbb Z^d$ with $|\ell_j|\le R_j$. In particular, for every $\boldsymbol\theta\in\R^d$ and every $j$, the function $f(x) = C(\boldsymbol\theta+x\boldsymbol e_j)$ is a real trigonometric polynomial of degree at most $R_j$, and \ona{Theorem~\ref{thm:ch07:general}}\onp{the general rule} applies to it with $R = R_j$.
\end{propn}
\ona{\emph{Proof.} As in the proof of Lemma~\ref{lem:ch07:freq}, $e^{-i\theta H_k} = \sum_\lambda e^{-i\theta\lambda}\Pi_{k,\lambda}$, where $\Pi_{k,\lambda}$ is the projector onto the eigenspace of $H_k$ for the eigenvalue $\lambda$. Expanding \eqref{eq:ch07:shared} gives $\ket{\psi(\boldsymbol\theta)} = \sum_{\boldsymbol\lambda}e^{-i\sum_k\lambda_k\theta_{j_k}}P_{\boldsymbol\lambda}\ket{\psi_0}$, the sum running over the choices $\boldsymbol\lambda = (\lambda_1,\dots,\lambda_K)$ of an eigenvalue of each generator, with $P_{\boldsymbol\lambda} = W_K\Pi_{K,\lambda_K}\cdots W_1\Pi_{1,\lambda_1}$. Hence}
\ona{\begin{equation*}
    C(\boldsymbol\theta) = \sum_{\boldsymbol\lambda,\boldsymbol\lambda'}e^{i\sum_k(\lambda_k-\lambda'_k)\theta_{j_k}}\braket{\psi_0|P_{\boldsymbol\lambda}^\dagger\,O\,P_{\boldsymbol\lambda'}|\psi_0}.
\end{equation*}}
\ona{The coefficient of $\theta_j$ in the exponent, $\ell_j = \sum_{k:\,j_k = j}(\lambda_k-\lambda'_k)$, is an integer with $|\ell_j|\le R_j$, and exchanging $\boldsymbol\lambda$ and $\boldsymbol\lambda'$ conjugates the term. Collecting the terms with equal $\boldsymbol\ell$ gives \eqref{eq:ch07:trig}. Collecting instead the terms with $\ell_j = \pm\ell$, for $\ell = 0,\dots,R_j$, gives $f$ in the form of Theorem~\ref{thm:ch07:general}, with real coefficients, since $f$ is real. \qed}
\onp{Proof: expand each gate in the projectors of its generator, as in the lemma on frequencies; the frequency of a term in $\theta_j$ is a sum of differences of eigenvalues over the gates that $\theta_j$ controls.}
\end{frame}

\begin{frame}\ft{When the cost is a trigonometric polynomial}
\onp{\footnotesize}
\ona{The proposition rests on three conditions, each of which can fail in practice.}
\begin{itemize}\setlength\itemsep{0.3em}
    \item Each parameter enters only as the angle of gates $e^{-i\theta_{j_k}H_k}$ whose generators do not depend on the parameters. An angle $c\,\theta_j$ or $\theta_j+c$, for a constant $c$, is admissible: the factor is absorbed into the generator, and the constant into $W_k$. A gate $e^{-i(\theta A+B)}$ with a drift $B$ that does not commute with $A$ is not of this form, and its cost is not a trigonometric polynomial (\ona{Theorem~\ref{thm:ch07:stoch}}\onp{the stochastic rule}).
    \item The eigenvalues of each generator differ by integers, possibly after a rescaling of each parameter, which requires that the differences for all the gates that $\theta_j$ controls be integer multiples of a single base frequency. Otherwise, as for a generator with the eigenvalues $0$, $1$ and $\sqrt2$, the cost is a sum of exponentials with incommensurable frequencies, which is in general not periodic, and neither the grid nor the hierarchy applies.
    \item The cost is the expectation of a fixed observable, and so linear in the state. Quantum channels that do not depend on $\boldsymbol\theta$ may act between the gates, since the proof applies to density matrices, as in \ona{Proposition~\ref{propn:ch07:noise}}\onp{the proposition on noise in the device}. An objective that is not linear in the distribution of the measured values, such as the CVaR objective of \Chap~\chref{C.QAOA}, is not of this form.
\end{itemize}
\ona{The number of parameters, the number of gates and the norm of the observable do not enter these conditions; they enter only the cost of the algorithm below.}
\end{frame}

\begin{frame}\ft{The degree in a shared parameter}
\onp{\footnotesize}
\begin{itemize}\setlength\itemsep{0.3em}
    \item The degree in $\theta_j$ adds the widths of all the gates that $\theta_j$ controls, whether or not they commute. With one gate for each parameter, this is \ona{Lemma~\ref{lem:ch07:freq} applied to each parameter}\onp{the lemma on frequencies}.
    \item In the worked example, $\gamma$ controls four rotations $R_Z(\gamma)$, of width $1$, and $\beta$ four rotations $R_X(2\beta)$, of width $2$: $R_\gamma\le4$ and $R_\beta\le8$. All the frequencies are even, and in $x = 2\gamma$ and $y = 2\beta$ the degrees are at most $2$ and $4$, as in \ona{Table~\ref{tab:ch07:freq}}\onp{the table of frequencies}.
    \onp{\item The general rule with $R = R_j$ gives $\partial C/\partial\theta_j$ from $2R_j$ circuits: for $k$ Pauli rotations, $2k$, as gate by gate; for $\gamma$, in $x = 2\gamma$, four.}
\end{itemize}
\end{frame}

\begin{frame}\ft{Reconstructing the cost from its values on a grid}
\onp{\scriptsize}
\ona{Theorem~\ref{thm:ch07:general} obtains the derivative $f'(0)$ from $2R$ values of $f$. The $2R+1$ values at the equally spaced points $\xi_k = 2\pi k/(2R+1)$, $k = 0,\dots,2R$, determine $f$ itself,}
\onp{$2R$ values give $f'(0)$; the $2R+1$ values at $\xi_k = 2\pi k/(2R+1)$, $k = 0,\dots,2R$, give $f$:}
\begin{equation}\label{eq:ch07:interp}
    a_0 = \frac1{2R+1}\sum_{k=0}^{2R}f(\xi_k),\qquad a_\ell = \frac2{2R+1}\sum_{k=0}^{2R}f(\xi_k)\cos\ell\xi_k,\qquad b_\ell = \frac2{2R+1}\sum_{k=0}^{2R}f(\xi_k)\sin\ell\xi_k,
\end{equation}
\ona{since $\sum_ke^{im\xi_k} = 0$ for every integer $m$ with $0<|m|\le2R$. Applied to each coordinate in turn, \eqref{eq:ch07:interp} gives the coefficients of \eqref{eq:ch07:trig} from the values of $C$ on the grid $\mathcal G$ of the points $\big(\frac{2\pi k_1}{2R_1+1},\dots,\frac{2\pi k_d}{2R_d+1}\big)$,}
\onp{and on the grid $\mathcal G$ of the points $\big(\frac{2\pi k_1}{2R_1+1},\dots,\frac{2\pi k_d}{2R_d+1}\big)$,}
\begin{equation}\label{eq:ch07:dft}
    c_{\boldsymbol\ell} = \frac1{|\mathcal G|}\sum_{\boldsymbol\phi\in\mathcal G}C(\boldsymbol\phi)\,e^{-i\boldsymbol\ell^\top\boldsymbol\phi},\qquad |\mathcal G| = \prod_{j=1}^d(2R_j+1),
\end{equation}
\ona{a discrete Fourier transform in $d$ dimensions, which the fast Fourier transform computes in $\mathcal O(|\mathcal G|\log|\mathcal G|)$ operations \citep{cooley1965algorithm}. The values are estimated, and the following lemma bounds the effect of their errors.}
\onp{\begin{itemize}\setlength\itemsep{0.3em}
    \item The $2R+1$ values determine $f$, as $2R$ values determine $f'(0)$; $|\mathcal G|$ values determine $C$.
    \item A discrete Fourier transform in $d$ dimensions: $\mathcal O(|\mathcal G|\log|\mathcal G|)$ operations by the fast Fourier transform \citep{cooley1965algorithm}.
\end{itemize}}
\end{frame}

\begin{frame}\ft{The error of the reconstruction}
\onp{\footnotesize}
\begin{lem}\label{lem:ch07:grid}
If the values in \eqref{eq:ch07:dft} are replaced by real estimates with errors of absolute value at most $\eta$, then the resulting trigonometric polynomial $\hat C$ is real, and $|\hat C(\boldsymbol\theta)-C(\boldsymbol\theta)|\le\sqrt{|\mathcal G|}\,\eta$ for every $\boldsymbol\theta$. In particular, $|\min\hat C-\min C|\le\sqrt{|\mathcal G|}\,\eta$.
\end{lem}
\ona{\emph{Proof.} The errors $e(\boldsymbol\phi)$ of the values change the coefficients by $\hat c_{\boldsymbol\ell}-c_{\boldsymbol\ell} = |\mathcal G|^{-1}\sum_{\boldsymbol\phi}e(\boldsymbol\phi)e^{-i\boldsymbol\ell^\top\boldsymbol\phi}$. By the identity behind \eqref{eq:ch07:interp}, applied in each coordinate, the $|\mathcal G|$ vectors $(e^{-i\boldsymbol\ell^\top\boldsymbol\phi})_{\boldsymbol\phi\in\mathcal G}$ are orthogonal, each of squared norm $|\mathcal G|$. By Parseval's identity, $\sum_{\boldsymbol\ell}|\hat c_{\boldsymbol\ell}-c_{\boldsymbol\ell}|^2 = |\mathcal G|^{-1}\sum_{\boldsymbol\phi}e(\boldsymbol\phi)^2\le\eta^2$, and by the Cauchy--Schwarz inequality, $\sum_{\boldsymbol\ell}|\hat c_{\boldsymbol\ell}-c_{\boldsymbol\ell}|\le\sqrt{|\mathcal G|}\,\eta$, which bounds $|\hat C(\boldsymbol\theta)-C(\boldsymbol\theta)|$ at every $\boldsymbol\theta$. Since the estimates are real, $\hat c_{-\boldsymbol\ell} = \overline{\hat c_{\boldsymbol\ell}}$. Finally, if $C$ is least at $\boldsymbol\theta^*$ and $\hat C$ at $\hat{\boldsymbol\theta}$, then $\min\hat C-\min C\le\hat C(\boldsymbol\theta^*)-C(\boldsymbol\theta^*)$ and $\min C-\min\hat C\le C(\hat{\boldsymbol\theta})-\hat C(\hat{\boldsymbol\theta})$. \qed}
\onp{Proof: Parseval's identity for \eqref{eq:ch07:dft} and the Cauchy--Schwarz inequality.}

\ona{\citet{sakos2026global} bound each coefficient by the largest error, which gives $|\mathcal G|\eta$ in place of $\sqrt{|\mathcal G|}\,\eta$. If $O$ has spectral width $w$, a measured value lies in an interval of length $w$, and by Hoeffding's inequality the mean of $M$ such values differs from its expectation by more than $\eta$ with probability at most $2e^{-2M\eta^2/w^2}$ \citep{hoeffding1963probability}. By the union bound, all the $|\mathcal G|$ estimates are within $\eta$ with probability at least $1-\delta$ if}
\onp{By Hoeffding's inequality and the union bound, all the estimates are within $\eta$ with probability at least $1-\delta$ with}
\begin{equation}\label{eq:ch07:hoeffding}
    M\ge\frac{w^2}{2\eta^2}\ln\frac{2|\mathcal G|}{\delta}
\end{equation}
\ona{copies are measured at each point.}
\onp{copies at each point, where $w$ is the spectral width of $O$ \citep{hoeffding1963probability}.}

\begin{frame}\ft{Minimising a trigonometric polynomial}
\onp{\footnotesize}
\ona{After the reconstruction, training is a classical problem, the minimisation of a known trigonometric polynomial over the torus $[-\pi,\pi)^d$. The \emph{sum-of-squares hierarchy} bounds its minimum from below. Writing $z_j = e^{i\theta_j}$, every polynomial $p$ in $z_1,\dots,z_d$ gives a non-negative function $|p|^2$, and so every $\lambda$ with}
\begin{equation}\label{eq:ch07:sos}
    C(\boldsymbol\theta)-\lambda = \sum_i\big|p_i(e^{i\theta_1},\dots,e^{i\theta_d})\big|^2,\qquad p_i\ \text{a polynomial of degree at most $r_j$ in $z_j$},
\end{equation}
\ona{is a lower bound on $\min C$. The largest such $\lambda$, which we denote $C^*_{\boldsymbol r}$, is computed by semidefinite programming: if $v$ is the vector of the $\prod_j(r_j+1)$ monomials of these degrees, the right-hand side is $v^\dagger Qv$ for a Hermitian positive semidefinite matrix $Q$, and \eqref{eq:ch07:sos} is a system of linear equations in $\lambda$ and the entries of $Q$, one for each frequency \citep{dumitrescu2007positive,josz2018lasserre}.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item $C^*_{\boldsymbol r}\le\min C$, and $C^*_{\boldsymbol r}$ increases with $\boldsymbol r$. In one variable it is exact at $r = R$: by the Fej\'er--Riesz theorem, a non-negative real trigonometric polynomial $f$ of degree $R$ is $|p(e^{ix})|^2$ for a polynomial $p$ of degree $R$ \citep{dumitrescu2007positive}.
    \item In several variables it need not be exact, but $\min C-C^*_{\boldsymbol r} = \mathcal O(1/r^2)$ for $r_j = r$ and every $C$, and the gap decays exponentially in $r$ if $C$ has finitely many global minimisers, with invertible Hessians \citep{bach2023exponential}. When a condition on the ranks of the solution of the dual problem holds, the minimisers can be read off from it \citep{sakos2026global}.
    \item The worked example: $F+3 = \frac14|e^{2i\gamma}+e^{4i\beta}|^2+\frac14|1-e^{2i(2\beta+\gamma)}|^2$, of degrees $2$ in $e^{2i\beta}$ and $1$ in $e^{2i\gamma}$. The hierarchy is exact at its lowest level, and the minimisers are where both terms vanish. For $\hat F$ from one simulated run, $C^*_{\boldsymbol r}$ at the lowest level, $(4,2)$, equals $\min\hat F = -3.046$.
\end{itemize}
\end{frame}

\begin{frame}\ft{Global optimisation for a constant number of parameters}
\onp{\footnotesize}
\begin{thm}[\citet{sakos2026global}]\label{thm:ch07:fpras}
Consider circuits \eqref{eq:ch07:shared} on $n$ qubits whose generators have eigenvalues that differ by integers, with a fixed number $d$ of parameters, and with $K$, the widths $\Delta_k$ and $\|O\|$ bounded by polynomials in $n$. There is an algorithm that, for all $\epsilon,\delta>0$, measures copies of the states of a set of circuits fixed in advance and returns, with probability at least $1-\delta$, a number within $\epsilon$ of $\min C$, in time and with a number of copies polynomial in $n$, $1/\epsilon$ and $\log(1/\delta)$.
\end{thm}
\ona{\emph{Proof (sketch).} Let $\Delta_{\max} = \max_k\Delta_k$. Each $R_j\le K\Delta_{\max}$, so that $|\mathcal G|\le(2K\Delta_{\max}+1)^d$, a polynomial in $n$ for fixed $d$. By Lemma~\ref{lem:ch07:grid} and \eqref{eq:ch07:hoeffding} with $\eta = \epsilon/(2\sqrt{|\mathcal G|})$, the reconstruction $\hat C$ satisfies $|\min\hat C-\min C|\le\epsilon/2$ with probability at least $1-\delta$, from $|\mathcal G|M = \mathcal O(|\mathcal G|^2w^2\epsilon^{-2}\log(|\mathcal G|/\delta))$ copies, where $w\le2\|O\|$. The explicit constants of \citet{bach2023exponential} give a level $r$, polynomial in $n$ and $1/\epsilon$, at which $\hat C^*_{\boldsymbol r}$ is within $\epsilon/2$ of $\min\hat C$, and the semidefinite program then has matrices of order $(r+1)^d$, polynomial for fixed $d$. \qed}
\onp{Proof: Lemma and \eqref{eq:ch07:hoeffding} with $\eta = \epsilon/(2\sqrt{|\mathcal G|})$, $|\mathcal G|\le(2K\max_k\Delta_k+1)^d$; the rate of \citet{bach2023exponential} with explicit constants.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item The device is used once, on $|\mathcal G|$ circuits fixed in advance, and the number of copies is known before the first is measured. The optimisation is then classical.
    \item The worked example, with $\epsilon = 0.1$, $\delta = 0.05$ and $w = 4$: $\eta = 0.0075$, and $M\approx1.1\times10^6$ copies at each point, $4.9\times10^7$ in all; with the bound of \citet{sakos2026global}, $2.2\times10^9$. The simulated errors with $1000$ copies at each point are far smaller than the guarantee requires.
    \item The effort grows exponentially with $d$, through $|\mathcal G|$ and the order of the semidefinite program. The gradient methods of this \chap{} scale polynomially with $d$, but find only stationary points.
\end{itemize}
\end{frame}

\begin{frame}\ft{What a constant number of parameters cannot express}
\onp{\footnotesize}
\ona{The theorem limits the problems that such circuits can encode. In the \emph{gap version} of training, one is given a circuit as in Theorem~\ref{thm:ch07:fpras} and rationals $a<b$, with $b-a$ at least the inverse of a polynomial in $n$, and asked to decide whether $\min C\le a$ or $\min C\ge b$, it being promised that one of the two holds. Running the algorithm with $\epsilon = (b-a)/3$ and $\delta = \frac13$, and comparing its result with $\frac12(a+b)$, decides this with probability at least $\frac23$ in polynomial time, with a quantum computer in place of the device. The gap version therefore lies in $\mathbf{BQP}$, and the following is a consequence.}
\onp{The gap version: given a circuit as in the theorem and $b-a\ge1/\mathrm{poly}(n)$, decide whether $\min C\le a$ or $\min C\ge b$. The algorithm with $\epsilon = (b-a)/3$ decides it on a quantum computer: it is in $\mathbf{BQP}$.}
\begin{cor}[\citet{sakos2026global}]\label{cor:ch07:bqp}
If MaxCut, in its decision version, reduces in polynomial time to the gap version of training circuits with a fixed number of parameters, then $\mathbf{NP}\subseteq\mathbf{BQP}$.
\end{cor}
\begin{itemize}\setlength\itemsep{0.2em}
    \item This does not contradict \Chap~\chref{C.VQE}. There, training is $\mathbf{NP}$-hard with a number of parameters that grows with the instance; the reduction of Bittel and Kliesch with a single parameter uses frequencies $m^{-j}$, which, rescaled to integers, give a degree exponential in the size of the instance.
    \item Unless $\mathbf{NP}\subseteq\mathbf{BQP}$, training is thus tractable for a constant number of parameters and $\mathbf{NP}$-hard when their number grows linearly with $n$; for $d$ of the order of $\log n$, the question is open \citep{sakos2026global}.
\end{itemize}
\end{frame}

\section{Noise}\label{sec:ch07:noise}

\begin{frame}\ft{Shot noise and stochastic gradients}
\onp{\footnotesize}
\ona{Each estimate in \eqref{eq:ch07:grad} is unbiased, and so the estimated gradient is an unbiased estimate of $\nabla C$, with a variance of $(\sigma_+^2+\sigma_-^2)/(4M)$ in each component for $M$ copies at each point. Gradient descent with such estimates is stochastic gradient descent, whose analysis does not require accurate gradients, only unbiased ones of bounded variance. \citet{sweke2020stochastic} show that the guarantees of stochastic gradient descent apply to variational algorithms for any number of copies at each point, down to a single one, and that the terms of the observable and the shifts of the general rule may themselves be sampled, which gives a doubly stochastic method.}
\begin{itemize}\setlength\itemsep{0.25em}
    \item Unbiased, with variance $(\sigma_+^2+\sigma_-^2)/(4M)$ per component: stochastic gradient descent applies, for any $M\ge1$ \citep{sweke2020stochastic}.
    \item More copies reduce the variance, not the bias; fewer copies per step allow more steps for the same $N$.
    \item Unbiasedness does not help on a barren plateau (\Chap~\chref{C.VQE}): if the partial derivatives are of order $2^{-n/2}$, of the order of $2^n$ copies are needed to determine even their signs.
    \item The number of steps, as a function of $\sigma^2$, $M$ and the bias of the device, is the subject of \Chap~\chref{C.IterationComplexity}.
\end{itemize}
\end{frame}

\begin{frame}\ft{Noise in the device}
\onp{\footnotesize}
\begin{propn}\label{propn:ch07:noise}
Suppose that the gate $R_P(\theta)$ is applied exactly, and that every other operation of the circuit, including the preparation of the state and the measurement, is replaced by a quantum channel that does not depend on $\theta$. Then the expectation measured by the device is a sinusoid in $\theta$, and \eqref{eq:ch07:psr} gives its derivative exactly.
\end{propn}
\ona{\emph{Proof.} The measured expectation is $\tilde f(\theta) = \Tr\big[\mathcal E_2^\dagger(O)\,R_P(\theta)\,\mathcal E_1(\rho)\,R_P(\theta)^\dagger\big]$, where $\mathcal E_1$ and $\mathcal E_2$ are the channels before and after the gate, and $\mathcal E_2^\dagger$ is the adjoint of $\mathcal E_2$. This is $f$ with the mixed state $\mathcal E_1(\rho)$ and the observable $\mathcal E_2^\dagger(O)$, and Lemma~\ref{lem:ch07:freq} holds for mixed states, which are convex combinations of pure ones. \qed}

\ona{The rule thus gives the gradient of the cost that the device actually computes, which differs from the ideal one. If the state is depolarised before the measurement with probability $p$, the measured cost is $(1-p)C + p\Tr(O)/2^n$: the gradient is multiplied by $1-p$, and the minimisers do not move (Fig.~\ref{fig:ch07:noise}). An asymmetric read-out of one qubit, which reports $1$ for $0$ with probability $q_0$ and $0$ for $1$ with probability $q_1$, maps $\braket Z$ to $(1-q_0-q_1)\braket Z + q_1-q_0$, again an affine map. For a product $Z_iZ_j$ of two such qubits, however, the measured value $(1-q_0-q_1)^2\braket{Z_iZ_j} + (q_1-q_0)(1-q_0-q_1)(\braket{Z_i}+\braket{Z_j}) + (q_1-q_0)^2$ mixes in other terms, and the minimisers move. If the noise of the gate itself depends on $\theta$, as when the duration of a pulse grows with the angle, the proposition does not apply, and the rule is biased even for the noisy cost. These are the biases of \Chap~\chref{C.IterationComplexity}.}

\section{Minimisation along one coordinate}\label{sec:ch07:coord}

\begin{frame}\ft{Minimising along one coordinate}
\onp{\footnotesize}
\ona{The values that give the derivative give more. A sinusoid $a + b\cos\theta + c\sin\theta$ has three coefficients, and three values determine it, together with its minimiser.}
\begin{propn}[\citet{nakanishi2020sequential,ostaszewski2021structure}]\label{propn:ch07:roto}
Let $f(\theta) = a + b\cos\theta + c\sin\theta$ be non-constant, and let $f_0 = f(\theta_0)$ and $f_\pm = f(\theta_0\pm\frac\pi2)$. Then $f$ is least at
\begin{equation}\label{eq:ch07:roto}
    \theta^* = \theta_0 + \pi - \operatorname{atan2}\big(f_--f_+,\ 2f_0-f_+-f_-\big)\pmod{2\pi},
\end{equation}
where $f(\theta^*) = \frac12(f_++f_-) - \frac12\sqrt{(2f_0-f_+-f_-)^2+(f_+-f_-)^2}$.
\end{propn}
\ona{\emph{Proof.} Write $f(\theta) = a + A\cos(\theta-\varphi)$ with $A>0$. Then $f_\pm = a\mp A\sin(\theta_0-\varphi)$, so that $a = \frac12(f_++f_-)$, $2f_0-f_+-f_- = 2A\cos(\theta_0-\varphi)$ and $f_--f_+ = 2A\sin(\theta_0-\varphi)$. Hence $\theta_0-\varphi$ is the $\operatorname{atan2}$ in \eqref{eq:ch07:roto}, $A$ is half the square root, and the minimum $a - A$ is attained at $\theta^* = \varphi+\pi$. \qed}
\begin{itemize}\setlength\itemsep{0.2em}
    \item Sweeping over the coordinates, $\theta_j\leftarrow\theta_j^*$, is a coordinate descent method with exact line minimisation, at two or three circuits for each coordinate. In exact arithmetic the cost never increases.
    \item Like every coordinate method, it may stop at a stationary point that is minimal along each coordinate but is a saddle point, as $x^2+y^2-3xy$ is at the origin.
    \item For a coordinate with $R$ frequencies, $2R+1$ values determine the trigonometric polynomial, which is then minimised classically \citep{Wierichs2021}. \citet{koczor2022quantum} build a model of the cost in all the coordinates near a point from shifted values, and descend on the model.
\end{itemize}
\end{frame}

\section{Derivative-free estimators}\label{sec:ch07:dfo}

\begin{frame}\ft{Simultaneous perturbation}
\onp{\footnotesize}
\ona{The estimators of the previous sections require at least two circuits for each parameter. The estimator of simultaneous perturbation stochastic approximation (SPSA) \citep{spall1992multivariate}, used in \Chap~\chref{C.QAOA}, perturbs all the parameters at once, along a random vector $\boldsymbol\Delta$ with independent components uniform on $\{\pm1\}$, and uses two circuits whatever the number $d$ of parameters:}
\begin{equation}\label{eq:ch07:spsa}
    \hat g_i = \frac{\hat C(\boldsymbol\theta+c\boldsymbol\Delta)-\hat C(\boldsymbol\theta-c\boldsymbol\Delta)}{2c\,\Delta_i},\qquad i = 1,\dots,d.
\end{equation}
\begin{lem}[\citet{spall1992multivariate}]\label{lem:ch07:spsa}
If $C$ is three times continuously differentiable and each value is estimated from $M$ copies, then $\mathbb E\,\hat{\boldsymbol g} = \nabla C(\boldsymbol\theta) + \mathcal O(c^2)$ and
\begin{equation}\label{eq:ch07:spsamse}
    \mathbb E\,\|\hat{\boldsymbol g}-\nabla C(\boldsymbol\theta)\|^2 = (d-1)\,\|\nabla C(\boldsymbol\theta)\|^2 + d\,\frac{\sigma_+^2+\sigma_-^2}{4c^2M} + \mathcal O(c^2).
\end{equation}
\end{lem}
\ona{\emph{Proof.} By Taylor's theorem, $\frac1{2c}[C(\boldsymbol\theta+c\boldsymbol\Delta)-C(\boldsymbol\theta-c\boldsymbol\Delta)] = \nabla C^\top\boldsymbol\Delta + \mathcal O(c^2)$. Since $1/\Delta_i = \Delta_i$, the noiseless part of $\hat g_i$ is $\sum_j\partial_jC\,\Delta_j\Delta_i + \mathcal O(c^2)$, whose mean is $\partial_iC + \mathcal O(c^2)$, as $\mathbb E\,\Delta_j\Delta_i = \delta_{ij}$. Its deviation $\sum_{j\ne i}\partial_jC\,\Delta_j\Delta_i$ has the second moment $\sum_{j\ne i}(\partial_jC)^2$, since the products $\Delta_j\Delta_i$, $j\ne i$, are independent signs; summed over $i$, this gives $(d-1)\|\nabla C\|^2$. The shot noise contributes $(\sigma_+^2+\sigma_-^2)/(4c^2M)$ to each component, independently of the rest. \qed}
\begin{itemize}\setlength\itemsep{0.2em}
    \item An estimate is a gradient only on average. Each one lies along the random diagonal $\boldsymbol\Delta$, and its error is of the order of $\sqrt d$ times the gradient.
    \item The bias of order $c^2$ calls for a small $c$, the shot noise for a large one, as for the central difference.
\end{itemize}
\end{frame}

\begin{frame}\ft{Gaussian smoothing}
\onp{\footnotesize}
\ona{The two-point estimator \citep{duchi2015optimal,Nesterov2015} perturbs along a Gaussian direction $\boldsymbol u\sim\mathcal N(0,\one_d)$ and compares with the unperturbed point,}
\begin{equation}\label{eq:ch07:2pfa}
    \hat{\boldsymbol g} = \frac{\hat C(\boldsymbol\theta+c\boldsymbol u)-\hat C(\boldsymbol\theta)}{c}\,\boldsymbol u.
\end{equation}
\begin{lem}\label{lem:ch07:smooth}
Without shot noise, $\mathbb E\,\hat{\boldsymbol g} = \nabla C_c(\boldsymbol\theta)$, the gradient of the \emph{Gaussian smoothing} $C_c(\boldsymbol\theta) = \mathbb E_{\boldsymbol u}C(\boldsymbol\theta+c\boldsymbol u)$. If $C = \sum_{\boldsymbol\omega}c_{\boldsymbol\omega}e^{i\boldsymbol\omega^\top\boldsymbol\theta}$ is a Hermitian trigonometric polynomial, then
\begin{equation}\label{eq:ch07:smoothtrig}
    C_c(\boldsymbol\theta) = \sum_{\boldsymbol\omega}c_{\boldsymbol\omega}\,e^{-c^2\|\boldsymbol\omega\|^2/2}\,e^{i\boldsymbol\omega^\top\boldsymbol\theta}.
\end{equation}
\end{lem}
\ona{\emph{Proof.} The standard Gaussian density $\varphi$ satisfies $\nabla\varphi(\boldsymbol u) = -\boldsymbol u\varphi(\boldsymbol u)$. Writing $C_c(\boldsymbol\theta) = c^{-d}\int C(\boldsymbol v)\varphi((\boldsymbol v-\boldsymbol\theta)/c)\,d\boldsymbol v$ and differentiating under the integral gives $\nabla C_c(\boldsymbol\theta) = \frac1c\mathbb E[C(\boldsymbol\theta+c\boldsymbol u)\boldsymbol u]$, and $\mathbb E[C(\boldsymbol\theta)\boldsymbol u] = 0$. The second claim follows from $\mathbb E\,e^{ic\boldsymbol\omega^\top\boldsymbol u} = e^{-c^2\|\boldsymbol\omega\|^2/2}$, the characteristic function of the Gaussian. \qed}
\begin{itemize}\setlength\itemsep{0.2em}
    \item The estimator is unbiased for the smoothed cost, not for the cost. For $L$-smooth $C$, $|C_c-C|\le\frac{c^2}2Ld$ and $\|\nabla C_c-\nabla C\|\le\frac c2L(d+3)^{3/2}$ \citep{Nesterov2015}.
    \item Smoothing damps the frequency $\boldsymbol\omega$ by $e^{-c^2\|\boldsymbol\omega\|^2/2}$. In the worked example, with frequencies $(\pm4,\pm2)$, $F_c = -2+e^{-10c^2}(F+2)$: the landscape is flattened, but its minimisers do not move. At $(0.2,-0.3)$ with $c = 0.2$, the mean of $2\times10^5$ estimates is $(-1.060,0.801)$, and $\nabla F_c = (-1.055,0.794)$.
\end{itemize}
\end{frame}

\section{Conclusions}

\begin{frame}\ft{A short history}
\onp{\scriptsize}
\ona{The rule and its variants have several origins; the following list, after \citet{kungurtsev2022iteration} and \citet{Wierichs2021}, gives the main steps.}
\begin{description}\setlength\itemsep{0.1em}
    \item[2005, 2017] Gradients of expectations with respect to control parameters in quantum optimal control \citep{Khaneja2005,Li_2017}.
    \item[2018] The rule for Pauli rotations, in quantum circuit learning \citep{Mitarai_2018}; a larger class of circuits whose derivatives are obtained without ancillas \citep{Theis2018}.
    \item[2019] The rule on hardware, compared with measurements with an ancilla \citep{schuld2019evaluating}; gates decomposed into gates to which the rule applies \citep{Crooks2019}.
    \item[2020] Exact minimisation along one coordinate \citep{nakanishi2020sequential}; stochastic gradient descent with few copies \citep{sweke2020stochastic}.
    \item[2021] Higher derivatives \citep{Mari_2021}; the stochastic rule \citep{Banchi_2021}; a rule with four values for fermionic excitations \citep{Kottmann2021}; algebraic extensions to general unitaries \citep{izmaylov2021analytic}; coordinate minimisation with structure optimisation \citep{ostaszewski2021structure}; rules of least variance \citep{Theis2021}.
    \item[2022] The general rule for arbitrary spectra \citep{Wierichs2021}; models of the cost from shifted values \citep{koczor2022quantum}.
    \item[2023] With single copies, the scaling of backpropagation cannot be achieved \citep{abbas2023backpropagation}.
    \item[2026] Global optimisation of the cost as a trigonometric polynomial, for a constant number of parameters \citep{sakos2026global}.
\end{description}
\end{frame}

\begin{frame}\ft{Conclusions}
\onp{\footnotesize}
\begin{itemize}\setlength\itemsep{0.2em}
    \item For a gate generated by a Pauli string, the cost is a sinusoid in the angle, and two circuits with the angle shifted by $\pm\frac\pi2$ give its derivative exactly; $2d$ circuits give the gradient of $d$ parameters, and about $2d^2$ the Hessian.
    \item A generator with $R$ frequencies requires $2R$ circuits. The weights of the general rule, which is the interpolation formula of M.~Riesz, are the least possible, and the variance for $N$ copies is at most $R^2\sigma^2/N$; the general rule saves circuits, not copies.
    \item Shot noise makes the estimates random but leaves them unbiased. Noise that does not depend on the angle leaves the rule exact for the cost that the device computes, which is not the ideal cost.
    \item The cost is a trigonometric polynomial, given by its values on a grid. With a constant number of parameters, it is minimised to any accuracy in polynomial time, and the circuits cannot encode $\mathbf{NP}$-hard problems unless $\mathbf{NP}\subseteq\mathbf{BQP}$.
    \item The same values give the exact minimum along one coordinate.
    \item Simultaneous perturbation and Gaussian smoothing use two circuits whatever $d$, at the price of a bias of order $c^2$ or of a smoothed cost, and of an error of the order of $d\|\nabla C\|^2$. How these trade-offs play out over many iterations is the subject of \Chap~\chref{C.IterationComplexity}.
\end{itemize}
\end{frame}
