\chapter{Warm-Started Quantum Optimization: Rounded}\label{C.WSRounded}

\ona{This \chap{} and the next follow \citet{egger2021warm}, \emph{Warm-starting quantum optimization}. Classical solvers for integer programs employ relaxations, and under the Unique Games Conjecture these relaxations often provide the best performance ratios available in polynomial time (\Chap~\chref{C.Inapprox}). Here we warm-start QAOA from a \emph{rounded} solution of a relaxation, the Goemans--Williamson cut of \Chap~\chref{C.Motivation}, so that the quantum algorithm inherits the classical guarantee. We picture the rounding and the warm start on a worked instance with six nodes, prove that the warm start can always return its cut, run the recursion that builds on it, view roundings as Brownian motions, and test the method by our own simulations. The numbers of this \chap{} are computed by \texttt{code/ch10\_figures.py}.}

\section{Relaxations and roundings}

\begin{frame}\ft{QUBO and two relaxations}
\ona{Quadratic Unconstrained Binary Optimization is}
\begin{equation}\label{eq:ch10:qubo}\tag{QUBO}
    \min_{x \in \{0,1\}^n} x^\top\Sigma x + \mu^\top x, \qquad \Sigma \in \R^{n\times n}\text{ symmetric},\ \mu\in\R^n.
\end{equation}
\ona{It is \textbf{NP}-hard; even checking local optimality is \textbf{NP}-hard. Two continuous relaxations:}
\begin{itemize}\setlength\itemsep{0.2em}
    \item If $\Sigma \succeq 0$, drop integrality:
    \begin{equation}\label{eq:ch10:qp}\tag{QP}
        \min_{x\in[0,1]^n} x^\top\Sigma x + \mu^\top x,
    \end{equation}
    a convex quadratic program with optimum $c^\star \in [0,1]^n$, solvable by any QP solver.
    \item In general, lift to matrices: with $z = 1-2x \in \{\pm1\}^n$ and $Y = zz^\top$, relax to
    \begin{equation}\label{eq:ch10:sdp}\tag{SDP}
        \max_{Y \in \mathbb S^n}\ \Tr(\Sigma' Y)\quad\text{s.t.}\quad \diag(Y) = e,\ Y \succeq 0,
    \end{equation}
    the semidefinite relaxation of \Chap~\chref{C.Motivation}, solvable to any fixed precision in polynomial time, even for $\Sigma$ with $10^{13}$ entries on a laptop.
\end{itemize}
\ona{MaxCut is the special case $\Sigma_{ij} = -w_{ij}$, and its \eqref{eq:ch10:sdp} is the Goemans--Williamson relaxation with its $\alpha_{\rm GW}\approx 0.878$ rounding, tight under the Unique Games Conjecture; with real-valued weights the hardness factor is $11/13$ \citep{charikar2004maximizing}.}
\end{frame}

\begin{frame}\ft{Encoding a cut as a product state}
\ona{QAOA associates each binary variable $x_i$ with a qubit via $x_i = (1-z_i)/2$, and the final measurement is a randomized rounding: measuring the product state}
\begin{equation}\label{eq:ch10:init}
    \ket{\phi} = \bigotimes_{i=1}^n R_Y(\theta_i)\ket0, \qquad \theta_i = 2\arcsin\sqrt{c_i},\qquad c_i \in [0,1],
\end{equation}
\ona{returns $x_i = 1$ with probability $c_i$, independently. A cut $z \in \{\pm1\}^n$ is the special case $c_i \in \{0,1\}$, but then the qubit is frozen: $H_C$ contains only $Z_iZ_j$ terms, so a computational basis state is an eigenstate of $e^{-i\gamma H_C}$ and nothing moves. Hence one \emph{regularises} with $\varepsilon \in (0, 0.5)$:}
\begin{equation}\label{eq:ch10:eps}
    c_i = \begin{cases}1-\varepsilon & z_i = -1\ (x_i = 1),\\ \varepsilon & z_i = +1\ (x_i = 0),\end{cases}
\end{equation}
\ona{so that the state is biased towards the cut but can leave it. At $\varepsilon = 0.5$ we recover $\ket{+}^{\otimes n}$: $\varepsilon$ interpolates between warm and cold start.}
\onp{Product state with $\mathrm{Pr}[x_i = 1] = c_i$; a cut has $c_i \in \{0,1\}$ and would be frozen; regularise to $c_i \in \{\varepsilon, 1-\varepsilon\}$; $\varepsilon = 0.5$ is the cold start.}
\end{frame}

\begin{frame}\ft{What can be gained, and what cannot}
\begin{itemize}\setlength\itemsep{0.3em}
    \item MaxCut with positive weights: GW's $\alpha_{\rm GW} \approx 0.878$ is optimal under the Unique Games Conjecture, and no polynomial-time algorithm beats $16/17$ unless $\mathbf P = \mathbf{NP}$ (\Chap~\chref{C.Inapprox}).
    \item MaxCut with real weights, i.e., general QUBO: the guarantee of SDP rounding and the hardness factor become $11/13$ \citep{charikar2004maximizing}; the dashed lines in the plots below mark it.
    \item Under the UGC, no warm-started variant can improve on the classical guarantee unless quantum computers solve \textbf{NP}-hard problems in polynomial time. If the UGC is false, both classical and quantum improvements may be possible, and then a quantum circuit that starts from the SDP solution is a natural candidate.
    \item What a warm start \emph{does} deliver regardless: the classical ratio for any angles, and empirically better cuts on many instances.
\end{itemize}
\end{frame}

\section{Warm-starting from a rounded solution}

\begin{frame}\ft{Rounded warm start}
\onp{\footnotesize}
\ona{For MaxCut, $\Sigma$ is not positive semidefinite, \eqref{eq:ch10:qp} is not convex, and one warm-starts from a \emph{rounded} solution: a cut $z\in\{\pm1\}^n$ produced by the random hyperplane, encoded as in \eqref{eq:ch10:init}--\eqref{eq:ch10:eps} with $\varepsilon\in(0,\frac12)$, so that the circuit can leave it. The mixer differs from that of the continuous warm start of \Chap~\chref{C.WSContinuous} in the sign of its off-diagonal entries.}
\begin{defn}\label{defn:ch10:ws}
For a cut $z$ and $\varepsilon\in(0,\frac12)$, let $c_i = \varepsilon$ if $z_i = +1$ and $c_i = 1-\varepsilon$ if $z_i = -1$, and $\theta_i = 2\arcsin\sqrt{c_i}$. The \emph{rounded warm start} prepares $\bigotimes_iR_Y(\theta_i)\ket0$, and its layers apply $e^{-i\gamma H_C}$, with $H_C$ minus the cut, and then the \emph{modified mixer}
\begin{equation}\label{eq:ch10:mixer}
    U_M(\beta) = \bigotimes_{i=1}^nR_Y(-\theta_i)\,R_Z(-2\beta)\,R_Y(\theta_i) = \bigotimes_{i=1}^n\big(\cos\beta\,\one+i\sin\beta\,\boldsymbol n_i'\cdot\boldsymbol\sigma\big),
\end{equation}
with $\boldsymbol n_i' = (-\sin\theta_i,0,\cos\theta_i)$: a rotation of each qubit by $2\beta$ about the axis through $\boldsymbol n_i'$, the mirror image of its warm state $\boldsymbol n_i = (\sin\theta_i,0,\cos\theta_i)$ in the $z$ axis.
\end{defn}
\ona{The second form in \eqref{eq:ch10:mixer} follows as in \Chap~\chref{C.WSContinuous}: conjugation by $R_Y(-\theta_i)$ turns $Z$ into $\boldsymbol n_i'\cdot\boldsymbol\sigma$, and $R_Z(-2\beta) = e^{i\beta Z}$. Unlike the continuous warm start, the warm state is not an eigenstate of this mixer, so that the mixer moves it even when $\gamma = 0$.}
\end{frame}

\begin{frame}\ft{The warm start keeps the guarantee}
\onp{\footnotesize}
\begin{propn}\label{propn:ch10:gw}
Let $\varepsilon = \frac14$. The circuit of depth one with $\gamma = 0$ and $\beta = \pi/2$ prepares, up to a phase, the computational basis state of $-z$, which is the cut $z$ with its sides exchanged. Hence, at every depth, the largest expected cut over the angles is at least the cut $C(z)$; and if $z$ is drawn by the random hyperplane and the weights are non-negative, it is at least $\alpha_{\rm GW}\,\mathrm{SDP}\ge\alpha_{\rm GW}\,\mathrm{OPT}$ in expectation over $z$.
\end{propn}
\ona{\emph{Proof.} With $\gamma = 0$ only the mixer acts, qubit by qubit, and $U_M(\pi/2) = \bigotimes_ii\,\boldsymbol n_i'\cdot\boldsymbol\sigma$: a rotation by $\pi$ about $\boldsymbol n_i'$. It maps the Bloch vector $\boldsymbol n_i$ to $2(\boldsymbol n_i\cdot\boldsymbol n_i')\boldsymbol n_i'-\boldsymbol n_i = 2\cos(2\theta_i)\,\boldsymbol n_i'-\boldsymbol n_i$, whose third coordinate is $\cos\theta_i\,(2\cos2\theta_i-1)$ and whose first is $-\sin\theta_i\,(2\cos2\theta_i+1)$. For $c_i = \frac14$, $\theta_i = \pi/3$, and for $c_i = \frac34$, $\theta_i = 2\pi/3$; in both cases $\cos2\theta_i = -\frac12$, the first coordinate vanishes, and the third is $-2\cos\theta_i = \mp1$. So a qubit with $z_i = +1$ ends in $\ket1$, which encodes $z_i = -1$, and one with $z_i = -1$ ends in $\ket0$: the state is that of $-z$, and $C(-z) = C(z)$. A layer with $\beta = \gamma = 0$ is the identity, so every depth can prepare it. For non-negative weights, the random hyperplane gives $\mathbb E\,C(z)\ge\alpha_{\rm GW}\,\mathrm{SDP}$, as shown in \Chap~\chref{C.Motivation}. \qed}
\onp{Proof: $U_M(\pi/2)$ rotates each $\boldsymbol n_i$ by $\pi$ about $\boldsymbol n_i'$, to the height $\cos\theta_i(2\cos2\theta_i-1) = \mp1$ when $\cos2\theta_i = -\frac12$, that is $\varepsilon = \frac14$.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item The guarantee of the rounding survives any circuit that can reproduce its cut; the circuit can only add to it.
    \item Only $\varepsilon = \frac14$ lands exactly on the poles; the initial state is not an eigenstate of the mixer, and the convergence theorem of \Chap~\chref{C.QAOA} does not apply.
\end{itemize}
\end{frame}

\section{Warm-start recursive QAOA}

\begin{frame}\ft{Warm-start recursive QAOA}
\onp{\footnotesize}
\ona{Rounding in the classical pre-processing combines with recursive QAOA (RQAOA) \citep{Bravyi2020}. At each step, run depth-one circuits, form the correlations $\mathcal M_{ij} = \braket{Z_iZ_j}$, and eliminate the variable of the pair with the largest $|\mathcal M_{ij}|$ by $z_i\leftarrow\mathrm{sgn}(\mathcal M_{ij})\,z_j$. The weights of $i$ move to $j$, with the sign of $\mathcal M_{ij}$, and the edge $ij$ becomes a constant. Recurse until $n_{\rm stop}$ variables remain, and solve those exactly. In the warm-started variant (WS-RQAOA), each step draws ten GW cuts of the current graph, runs a rounded warm start with $\varepsilon = \frac14$ from each of the five best distinct ones, and averages their correlations \citep{egger2021warm}.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item One elimination per step: $z_i\leftarrow\mathrm{sgn}(\mathcal M_{ij})\,z_j$ for the largest $|\mathcal M_{ij}|$; the graph loses a node.
    \item WS-RQAOA: the correlations of rounded warm starts from the five best of ten GW cuts.
    \item By \ona{Proposition~\ref{propn:ch10:gw}}\onp{the proposition on the guarantee}, a depth-one warm start can return its cut, with correlations $\pm1$, so that the recursion follows the consensus of the cuts.
\end{itemize}
\end{frame}

\section{Roundings as Brownian motions}\label{sec:ch10:bm}

\begin{frame}\ft{Sticky Brownian motions}
\onp{\footnotesize}
\ona{A randomised rounding turns unit vectors $u_1,\dots,u_n$ into signs $\sigma_1,\dots,\sigma_n$, and the measurement of a warm-started circuit is one too. \citet{abbasi2020sticky} view roundings through a Brownian motion $B$ in $\R^d$: the \emph{sticky Brownian motion} follows $W_{u_i}(t) = u_i^\top B(t)$ until it first reaches $\pm1$, and stays there, so that $\sigma_i = W_{u_i}(\infty)$. More generally, a \emph{speed function} $\xi$, positive on $(-1,1)$ and vanishing at $\pm1$, slows the motion down near the boundary, $\mathrm dW_u = \xi(W_u)\,u^\top\mathrm dB$, and the signs are the limits $\sigma_u = \lim_{t\to\infty}W_u(t)$. Each speed function is a rounding, with its own correlations $\mathbb E[\sigma_u\sigma_v]$, and so its own ratio $\min_t(1-\mathbb E[\sigma_u\sigma_v])/(1-t)$ over $t = u^\top v$.}
\begin{propn}[\citet{eldan2019krivine}]\label{propn:ch10:krivine}
Let $W_u(t) = 2\Phi\big(u^\top B(t)/\sqrt{1-t}\big)-1$ for $0\le t<1$, with $\Phi$ the standard normal distribution function. After the time change $s = -\ln(1-t)$, $W_u$ satisfies $\mathrm dW_u = \xi(W_u)\,\mathrm d\beta_s$, with a standard Brownian motion $\beta$ and
\begin{equation}\label{eq:ch10:krivine}
    \xi(w) = \sqrt{2/\pi}\,e^{-\Phi^{-1}((1-w)/2)^2/2},
\end{equation}
and $W_u(t)\to\mathrm{sign}(u^\top B(1))$ as $t\to1$. Hence this rounding is the random hyperplane with the normal $B(1)$, and $\mathbb E[\sigma_u\sigma_v] = \frac2\pi\arcsin(u^\top v)$.
\end{propn}
\ona{\emph{Proof.} Given the path up to $t$, $u^\top B(1)$ is normal with the mean $u^\top B(t)$ and the variance $1-t$, so $\mathrm{Pr}[u^\top B(1)>0\,|\,B(r),r\le t] = \Phi(u^\top B(t)/\sqrt{1-t})$, and $W_u(t)$ is the conditional expectation of $\mathrm{sign}(u^\top B(1))$: a martingale that converges to it. By It\^o's formula, with $x = u^\top B(t)/\sqrt{1-t}$, the martingale has no drift, and $\mathrm dW_u = 2\Phi'(x)\,u^\top\mathrm dB/\sqrt{1-t}$, where $2\Phi'(x) = \sqrt{2/\pi}\,e^{-x^2/2}$ and $x^2 = \Phi^{-1}((1+W_u)/2)^2 = \Phi^{-1}((1-W_u)/2)^2$. Finally, $\mathrm d\beta_s = u^\top\mathrm dB/\sqrt{1-t}$ has the quadratic variation $\mathrm dt/(1-t) = \mathrm ds$. Two vectors are separated by a random hyperplane with the probability $\arccos(u^\top v)/\pi$, so $\mathbb E[\sigma_u\sigma_v] = 1-\frac2\pi\arccos(u^\top v) = \frac2\pi\arcsin(u^\top v)$. \qed}
\onp{Proof: $W_u(t)$ is the conditional expectation of $\mathrm{sign}(u^\top B(1))$; It\^o's formula and $s = -\ln(1-t)$ give the speed $\xi$.}
\end{frame}

\section{Lessons}

\begin{frame}\ft{What the simulations show}
\onp{\footnotesize}
\begin{itemize}\setlength\itemsep{0.3em}
    \item A warm start from a cut can always return that cut, with $\varepsilon = \frac14$, and so keeps the guarantee of the rounding; at depth one on graphs of $10$ nodes it returns the cut, and improved on it for one pair in $46$.
    \item Deeper circuits stay near the cut: on the worked example the warm start leaves it only at depth five, while standard QAOA climbs steadily from far below.
    \item The regularisation must match the mixer: only $\varepsilon = \frac14$ lands the half turn on the poles, and the continuous mixer loses the cut.
    \item WS-RQAOA repairs most failures of RQAOA; on graphs of $10$ nodes, ten GW roundings are already hard to beat.
\end{itemize}
\ona{\citet{egger2021warm} report larger gains on graphs of $20$ and $30$ nodes, where the rounding leaves more room. An implementation of WS-QAOA is available in Qiskit Optimization.}
\end{frame}

\section{The design space}

\begin{frame}\ft{The design space: what, when, and how to round}
\begin{description}\setlength\itemsep{0.2em}
    \item[What to round?] the \eqref{eq:ch10:qp} relaxation; second-order cone strengthenings; the basic \eqref{eq:ch10:sdp}; entropy-penalised or moment/sum-of-squares hierarchies, increasingly strong and increasingly expensive.
    \item[When to round?] in classical pre-processing (WS-RQAOA), or inside the quantum circuit, in the simplest case by the final measurement (WS-QAOA, \Chap~\chref{C.WSContinuous}).
    \item[How to round?] coordinate-wise uniformly at random ($1/2$ for MaxCut); random hyperplane ($\alpha_{\rm GW}$); iterative ``sticky'' roundings analysable by Brownian motion, which match the SDP guarantees; or a measurement.
\end{description}
\ona{Representing the matrix-valued SDP solution $Y^\star$ directly in a quantum state would need $\Theta(n^2)$ qubits, too many for the near term, but would inherit the strongest guarantees. Under the Unique Games Conjecture no variant can improve on GW unless quantum computers solve \textbf{NP}-hard problems in polynomial time; if the conjecture is false, both classical and quantum improvements may be possible.}
\end{frame}

\begin{frame}\ft{Exercises}
\onp{\footnotesize}
\begin{exe}[The warm state of a cut]\label{exe:ch10:warm}
For the warm state of a cut $z$ with the regularisation $\varepsilon$, show that $\mathbb E[z_iz_j] = (1-2\varepsilon)^2z_iz_j$ for $i\ne j$, and hence that the expected cut is $\frac12(1-(1-2\varepsilon)^2)\sum_{i<j}w_{ij}+(1-2\varepsilon)^2C(z)$. Check the value $2.25$ of the worked example, whose weights add up to $-4$.
\end{exe}
\begin{exe}[The half turn triples the angle]\label{exe:ch10:triple}
Show that the half turn $\beta = \pi/2$ of \eqref{eq:ch10:mixer} moves a qubit at the polar angle $\theta$ to the polar angle $3\theta$, so that it reads the flipped bit with the probability $\sin^2(3\theta/2)$. Deduce that it lands on the pole only for $\varepsilon = \frac14$ in $(0,\frac12)$, and compute the probability for $\varepsilon = 0.15$; compare with \ona{Fig.~\ref{fig:ch10:eps}}\onp{the figure on regularisation}.
\end{exe}
\end{frame}

\begin{frame}\ft{Exercises, continued}
\onp{\footnotesize}
\begin{exe}[The speed of the Krivine diffusion]\label{exe:ch10:speed}
Show that the speed \eqref{eq:ch10:krivine} satisfies $\xi(0) = \sqrt{2/\pi}$ and $\xi(w)\to0$ as $w\to\pm1$, and that $\xi(w)^2/(1-w^2)\to0$, so that the diffusion slows down near the boundary faster than $(1-w^2)^{1/2}$. \end{exe}

Verify the weight $26$ between nodes $5$ and $6$ in \ona{Fig.~\ref{fig:ch10:rqaoa}}\onp{the figure of the recursion}.
\end{exe}
