\chapter{Iteration Complexity of QAOA}\label{C.IterationComplexity}

\ona{The optimizer of a variational algorithm is an iterative method, and this \chap{} asks how many iterations it needs. \Chap~\chref{C.VQE} showed that no polynomial-time method can guarantee the global minimum, nor even a point near a local minimum, and we therefore ask for a point at which the gradient is small. For gradient descent with the parameter-shift rule of \Chap~\chref{C.PSR}, and with the derivative-free estimators described there, we bound the number of iterations after which the expected squared norm of the gradient falls below a given $\epsilon$. The estimates of a quantum computer are noisy, which the theory of stochastic gradient descent accommodates, and they are also \emph{biased}: the noise of the device shifts the mean of an estimate, however many copies are measured. Following \citet{kungurtsev2022iteration}, we show that the bias does not change the rate of convergence, but sets a floor below which the guarantee cannot go, and we show that no method can remove this floor. We make the constants explicit for QAOA of depth $p$, run the derivative-free estimators on the worked example, and show that decreasing steps remove the noise, but not the bias. We state the results with the constants that our proofs give, which differ in places from those of \citet{kungurtsev2022iteration}. The numbers quoted in this \chap{} are computed by \texttt{code/ch08\_figures.py}, on the worked example of \Chap~\chref{C.VQE}.}

\section{Iteration complexity}\label{sec:ch08:complexity}

\begin{frame}\ft{Counting iterations}
\onp{\footnotesize}
\ona{We consider, as in \Chap~\chref{C.PSR}, the cost of a circuit with $d$ parameters, $C(\boldsymbol\theta) = \braket{\psi(\boldsymbol\theta)|O|\psi(\boldsymbol\theta)}$, its infimum $C^*$, and iterations of the form}
\begin{equation}\label{eq:ch08:iter}
    \boldsymbol\theta_{k+1} = \boldsymbol\theta_k - \alpha\,\boldsymbol g_k,\qquad k = 0,1,\dots,K-1,
\end{equation}
\ona{where $\boldsymbol g_k$ is an estimate of the gradient $\nabla C(\boldsymbol\theta_k)$, obtained from circuits, and $\alpha>0$ is the step. The \emph{iteration complexity} of such a method, for a class of costs and a measure of accuracy, is the number of iterations that it needs, in the worst case over the class, to reach the accuracy. What a single iteration costs, in circuits and copies, is the subject of \Chap~\chref{C.PerIteration}.}
\begin{defn}\label{defn:ch08:stationary}
Let $\epsilon>0$. A random point $\hat{\boldsymbol\theta}$ is an \emph{$\epsilon$-stationary point} of $C$ if $\mathbb E\,\|\nabla C(\hat{\boldsymbol\theta})\|^2\le\epsilon$.
\end{defn}
\begin{itemize}\setlength\itemsep{0.2em}
    \item The global minimum cannot be guaranteed: training VQA is $\mathbf{NP}$-hard even for circuits that can be simulated classically, and so is finding a point near a local minimum (\Chap~\chref{C.VQE}).
    \item For a constant number of parameters, the algorithm of \citet{sakos2026global} (\Chap~\chref{C.PSR}) gives the global minimum and a global minimiser, to within $\epsilon$; first-order methods give only first-order $\epsilon$-critical points.
    \item A small gradient is necessary at a minimum, and it is not sufficient: maxima and saddle points are stationary too. The bounds of this \chap{} say nothing about the value of $C$; good starting points are the subject of \Chaps~\chref{C.WSRounded}--\chref{C.WSContinuous}.
    \item The expectation is over the measurements and over any randomness of the method.
\end{itemize}

\ona{The guarantees of this \chap{} are thus weaker in kind than that of \citet{sakos2026global}. For a constant number of parameters, their algorithm, described in \Chap~\chref{C.PSR}, reconstructs the cost from its values on a grid and approximates its global minimum to within $\epsilon$; a global minimiser, to within $\epsilon$ in value, can then be found classically, by evaluating the reconstruction on a sufficiently fine grid, whose size is polynomial when $d$ is constant. The first-order methods of this \chap{} guarantee only first-order $\epsilon$-critical points, the $\epsilon$-stationary points of Definition~\ref{defn:ch08:stationary}, which may lie near saddle points or poor local minima; in exchange, their cost grows polynomially with $d$, and not exponentially.}
\end{frame}

\begin{frame}\ft{Bounds on the derivatives of the cost}
\onp{\scriptsize}
\ona{The analysis of every method in this \chap{} rests on bounds on the derivatives of the cost. By the lemma on frequencies of \Chap~\chref{C.PSR}, applied to each gate in turn, the cost is a Hermitian trigonometric polynomial in the sense of \Chap~\chref{C.VQE}, whose degree in $\theta_j$ is at most the largest frequency of the $j$-th gate; for a Pauli rotation it is $1$.}
\begin{lem}\label{lem:ch08:derivs}
Let $C$ be a Hermitian trigonometric polynomial of degree at most $R_j$ in $\theta_j$, $j = 1,\dots,d$, let $W$ be half the width of its range, and let $\|\boldsymbol R\|^2 = \sum_jR_j^2$. Then for all $\boldsymbol\theta$ and $\boldsymbol h$ in $\R^d$,
\begin{equation}\label{eq:ch08:derivs}
    \|\nabla C(\boldsymbol\theta)\|\le\|\boldsymbol R\|\,W,\qquad |\boldsymbol h^\top\nabla^2C(\boldsymbol\theta)\boldsymbol h|\le\|\boldsymbol R\|^2W\|\boldsymbol h\|^2,\qquad |D^3C(\boldsymbol\theta)[\boldsymbol h,\boldsymbol h,\boldsymbol h]|\le\|\boldsymbol R\|^3W\|\boldsymbol h\|^3.
\end{equation}
For a circuit of $d$ Pauli rotations, $R_j = 1$ and $W\le w/2$, where $w = \lambda_{\max}(O)-\lambda_{\min}(O)$ is the spectral width of $O$; hence $\|\nabla C\|\le\sqrt d\,w/2$ and $\|\nabla^2C\|\le dw/2$.
\end{lem}
\ona{\emph{Proof.} By the general parameter-shift rule of \Chap~\chref{C.PSR}, applied at an arbitrary point, a real trigonometric polynomial $p$ of degree $R$ satisfies $|p'|\le\sum_\mu2|y_\mu|\max|p| = R\max|p|$, which is Bernstein's inequality \citep{zygmund2003trigonometric}; it applies to $p$ minus any constant, which has the same derivative. Applied to $\theta_j\mapsto C(\boldsymbol\theta)$ minus the midpoint of the range, it gives $|\partial_jC|\le R_jW$. The derivative $\partial_jC$ is again a trigonometric polynomial, of degree at most $R_k$ in $\theta_k$ and bounded by $R_jW$, so that $|\partial_k\partial_jC|\le R_kR_jW$, and in the same way $|\partial_l\partial_k\partial_jC|\le R_lR_kR_jW$. Hence $|\boldsymbol h^\top\nabla^2C\boldsymbol h|\le\sum_{j,k}R_jR_k|h_j||h_k|W = (\sum_jR_j|h_j|)^2W\le\|\boldsymbol R\|^2\|\boldsymbol h\|^2W$ by the Cauchy--Schwarz inequality, and similarly for the gradient and the third derivative. The range of $C$ lies between the extreme eigenvalues of $O$. \qed}
\onp{Proof: Bernstein's inequality $|p'|\le R\max|p|$, from the general rule, applied once, twice or three times, and the Cauchy--Schwarz inequality.}
\begin{itemize}\setlength\itemsep{0.15em}
    \item The bounds are attained: $R_Y(\theta_d)\cdots R_Y(\theta_1)$ on one qubit, with $O = Z$, has the cost $\cos(\theta_1+\dots+\theta_d)$, with $\|\nabla C\| = \sqrt d$ and $\|\nabla^2C\| = d$ at suitable points.
    \item In the worked example, $F(\beta,\gamma) = -2+\sin4\beta\sin2\gamma$ has the degrees $4$ and $2$ and $W = 1$: $\|\nabla F\|\le\sqrt{20}\approx4.47$ and $\|\nabla^2F\|\le20$, against the largest values $4$ and $16$. The spectra alone, with the degrees $8$ and $4$ and $W = 2$, give $\|\nabla^2F\|\le160$.
\end{itemize}
\end{frame}

\begin{frame}\ft{The constants of QAOA}
\onp{\scriptsize}
\ona{For QAOA the degrees in Lemma~\ref{lem:ch08:derivs} follow from the spectra of the two Hamiltonians. We use the conventions of the worked example of \Chap~\chref{C.VQE}. For MaxCut on a graph with $n$ vertices, $m$ edges and the maximum cut $c^*$, the cost Hamiltonian has the eigenvalues $-c(x)$, minus the cuts; layer $k$ applies the $m$ gates $e^{-i\gamma_kZ_iZ_j/2}$, each a gate $R_Z(\gamma_k)$ between two CNOTs, and then the $n$ gates $R_X(2\beta_k) = e^{-i\beta_kX_i}$.}
\begin{cor}\label{cor:ch08:qaoa}
The cost $F_p$ of QAOA of depth $p$ satisfies, for all angles $(\boldsymbol\beta,\boldsymbol\gamma)\in\R^{2p}$,
\begin{equation}\label{eq:ch08:qaoa}
    \|\nabla F_p\|\le\sqrt{p\,(c^{*2}+4n^2)}\;\frac{c^*}2,\qquad\|\nabla^2F_p\|\le p\,(c^{*2}+4n^2)\,\frac{c^*}2,
\end{equation}
and the parameter-shift rule, applied to each gate, gives its gradient from $2p(m+n)$ circuits.
\end{cor}
\ona{\emph{Proof.} The gates of a layer commute, and $\gamma_k$ enters through $e^{-i\gamma_kH}$ with $H = \frac12\sum_{ij\in E}Z_iZ_j$, whose eigenvalues $\frac m2-c(x)$ differ by integers and span an interval of length $c^*$; $\beta_k$ enters through $e^{-i\beta_k\sum_iX_i}$, whose generator has the eigenvalues $-n,\dots,n$. By the lemma on frequencies of \Chap~\chref{C.PSR}, $F_p$ has degree at most $c^*$ in each $\gamma_k$ and $2n$ in each $\beta_k$, so that $\|\boldsymbol R\|^2 = p\,(c^{*2}+4n^2)$; the range of $F_p$ lies in $[-c^*,0]$, so that $W\le c^*/2$. Lemma~\ref{lem:ch08:derivs} gives \eqref{eq:ch08:qaoa}. Each of the $p(m+n)$ gates requires two circuits. \qed}
\onp{Proof: the generators of $\gamma_k$ and $\beta_k$ have spectra of widths $c^*$ and $2n$; the lemma on derivatives with $W = c^*/2$.}
\begin{itemize}\setlength\itemsep{0.15em}
    \item The constants are polynomial in $p$, $n$ and $m$. With $C(\boldsymbol\theta_0)-C^*\le c^*$, the iteration counts of the following sections are polynomial as well.
    \item For the worked example, $p = 1$ and $c^* = n = m = 4$: $\|\nabla^2F\|\le160$, the bound from the spectra quoted above.
\end{itemize}
\end{frame}

\begin{frame}\ft{The descent lemma}
\onp{\footnotesize}
\begin{lem}\label{lem:ch08:descent}
If $\|\nabla^2C(\boldsymbol\theta)\|\le L$ for all $\boldsymbol\theta$, then for all $\boldsymbol\theta$ and $\boldsymbol\theta'$,
\begin{equation}\label{eq:ch08:descent}
    C(\boldsymbol\theta')\le C(\boldsymbol\theta) + \nabla C(\boldsymbol\theta)^\top(\boldsymbol\theta'-\boldsymbol\theta) + \frac L2\|\boldsymbol\theta'-\boldsymbol\theta\|^2.
\end{equation}
In particular, the step $\boldsymbol\theta' = \boldsymbol\theta-\frac1L\nabla C(\boldsymbol\theta)$ decreases the cost by at least $\|\nabla C(\boldsymbol\theta)\|^2/(2L)$.
\end{lem}
\ona{\emph{Proof.} With $\boldsymbol h = \boldsymbol\theta'-\boldsymbol\theta$, Taylor's theorem with the integral form of the remainder gives $C(\boldsymbol\theta')-C(\boldsymbol\theta)-\nabla C(\boldsymbol\theta)^\top\boldsymbol h = \int_0^1(1-t)\,\boldsymbol h^\top\nabla^2C(\boldsymbol\theta+t\boldsymbol h)\boldsymbol h\,dt\le\frac L2\|\boldsymbol h\|^2$. The right-hand side of \eqref{eq:ch08:descent} is least at $\boldsymbol\theta' = \boldsymbol\theta-\nabla C(\boldsymbol\theta)/L$, where it equals $C(\boldsymbol\theta)-\|\nabla C(\boldsymbol\theta)\|^2/(2L)$. \qed}
\onp{Proof: Taylor's theorem with the integral form of the remainder; the right-hand side is least at $\boldsymbol\theta-\nabla C/L$.}

\ona{We call $L$ the \emph{smoothness constant} of $C$; it is a Lipschitz constant of $\nabla C$, and by Lemma~\ref{lem:ch08:derivs} it may be taken to be $dw/2$ for every circuit of $d$ Pauli rotations. The lemma says that the cost lies below a parabola that touches it at $\boldsymbol\theta$, and minimising the parabola in place of the cost gives a step of gradient descent with $\alpha = 1/L$. Figure~\ref{fig:ch08:descent} shows this for the energy of the worked example as a function of the angle $\phi$ of one gate, as in \Chap~\chref{C.PSR}, $f(\phi) = -2.039+0.195\sin(\phi-0.1)$, for which $L = 0.195$. From $\phi_0 = -0.1$, where $f = -2.077$ and $f' = 0.191$, the step reaches the vertex $\phi_1 = -1.080$ of the parabola. The lemma guarantees a decrease of $f'(\phi_0)^2/(2L) = 0.094$, and the actual decrease is $0.141$. The lemma is then applied afresh at $\phi_1$, with a new parabola whose vertex $\phi_2 = -1.461$ is nearly the minimiser $-1.471$; this step gains $0.0147$, against $0.0141$ guaranteed, since $f$ is nearly a parabola of curvature $L$ near its minimum. A third step leaves an error of $2\cdot10^{-7}$ (Exercise~\ref{exe:ch08:gate1}).}

\section{Gradient descent}\label{sec:ch08:gd}

\begin{frame}\ft{Gradient descent}
\onp{\footnotesize}
\begin{thm}\label{thm:ch08:gd}
Let $\|\nabla^2C\|\le L$. Gradient descent with exact gradients and the step $1/L$, $\boldsymbol\theta_{k+1} = \boldsymbol\theta_k-\frac1L\nabla C(\boldsymbol\theta_k)$, satisfies
\begin{equation}\label{eq:ch08:gd}
    \min_{k<K}\|\nabla C(\boldsymbol\theta_k)\|^2\le\frac1K\sum_{k=0}^{K-1}\|\nabla C(\boldsymbol\theta_k)\|^2\le\frac{2L\,(C(\boldsymbol\theta_0)-C^*)}K.
\end{equation}
\end{thm}
\ona{\emph{Proof.} By Lemma~\ref{lem:ch08:descent}, $\|\nabla C(\boldsymbol\theta_k)\|^2\le2L\,(C(\boldsymbol\theta_k)-C(\boldsymbol\theta_{k+1}))$. Summed over $k<K$, the right-hand side telescopes to $2L\,(C(\boldsymbol\theta_0)-C(\boldsymbol\theta_K))\le2L\,(C(\boldsymbol\theta_0)-C^*)$. \qed}
\onp{Proof: each step decreases $C$ by at least $\|\nabla C(\boldsymbol\theta_k)\|^2/(2L)$, and the decreases add up to at most $C(\boldsymbol\theta_0)-C^*$.}

\ona{The right-hand side of \eqref{eq:ch08:gd} involves three quantities. The first, $L$, is the smoothness constant of Lemma~\ref{lem:ch08:descent}: a bound on the curvature of $C$ in every direction or, equivalently, a Lipschitz constant of the gradient, $\|\nabla C(\boldsymbol\theta)-\nabla C(\boldsymbol\theta')\|\le L\,\|\boldsymbol\theta-\boldsymbol\theta'\|$. It sets the step, and with it the decrease that a step can be guaranteed to achieve. The second, $C(\boldsymbol\theta_0)-C^*$, is the height of the starting point above the infimum. The level set $\{\boldsymbol\theta:C(\boldsymbol\theta) = C(\boldsymbol\theta_0)\}$ bounds the sublevel set $\{\boldsymbol\theta:C(\boldsymbol\theta)\le C(\boldsymbol\theta_0)\}$, which the iterates never leave, since each step decreases $C$; the height is a budget that the decreases share. The third, $K$, is the number of iterations among which the budget is divided. Figure~\ref{fig:ch08:gdmap} shows the three on the worked example.}

\ona{Hence $\lceil2L(C(\boldsymbol\theta_0)-C^*)/\epsilon\rceil$ iterations give a point with $\|\nabla C\|^2\le\epsilon$. For a circuit of $d$ Pauli rotations, $L\le dw/2$ and $C(\boldsymbol\theta_0)-C^*\le w$, so that $dw^2/\epsilon$ iterations suffice, whatever the circuit; the number grows linearly with the number of parameters, through the smoothness constant. For QAOA of depth $p$, Corollary~\ref{cor:ch08:qaoa} and $C(\boldsymbol\theta_0)-C^*\le c^*$ give $p\,(c^{*2}+4n^2)\,c^{*2}/\epsilon$. For general smooth functions the order cannot be improved: \citet{carmon2020lower} show that, in sufficiently high dimension, every method that uses gradients requires a number of iterations of the order of $L(C(\boldsymbol\theta_0)-C^*)/\epsilon$ in the worst case. Their functions are not costs of circuits, and the lower bound does not transfer to them directly. In the worked example, gradient descent with the step $1/20$ from $(0.1,-0.1)$ reaches the minimum $-3$ to six digits within $200$ iterations.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item On the right: $L$ bounds the curvature, a Lipschitz constant of $\nabla C$; $C(\boldsymbol\theta_0)-C^*$ is the height of the start above the infimum, a budget for the decreases; the iterates stay in the sublevel set $\{C\le C(\boldsymbol\theta_0)\}$; $K$ counts the iterations.
    \item $2L(C(\boldsymbol\theta_0)-C^*)/\epsilon$ iterations; at most $dw^2/\epsilon$ for $d$ Pauli rotations.
    \item The order is the best possible for general smooth functions \citep{carmon2020lower}.
    \item The bound holds for the least gradient among the iterates, not for the last iterate, and says nothing about the value.
\end{itemize}
\end{frame}

\begin{frame}\ft{Linear convergence near a minimiser}
\onp{\footnotesize}
\ona{The rate $1/K$ of Theorem~\ref{thm:ch08:gd} is slow. Near a minimiser gradient descent may converge much faster, under a condition that is weaker than convexity.}
\begin{propn}[\citet{polyak1963gradient}]\label{propn:ch08:pl}
Let $\|\nabla^2C\|\le L$, and suppose that on a set that contains the iterates, $C$ satisfies the \emph{Polyak--\Loj{} inequality}
\begin{equation}\label{eq:ch08:pl}
    \tfrac12\|\nabla C(\boldsymbol\theta)\|^2\ge\mu\,(C(\boldsymbol\theta)-C^*)
\end{equation}
for some $\mu>0$. Then gradient descent with the step $1/L$ satisfies $C(\boldsymbol\theta_k)-C^*\le(1-\mu/L)^k(C(\boldsymbol\theta_0)-C^*)$.
\end{propn}
\ona{\emph{Proof.} By Lemma~\ref{lem:ch08:descent} and \eqref{eq:ch08:pl}, $C(\boldsymbol\theta_{k+1})-C^*\le C(\boldsymbol\theta_k)-C^*-\|\nabla C(\boldsymbol\theta_k)\|^2/(2L)\le(1-\mu/L)(C(\boldsymbol\theta_k)-C^*)$. \qed}

\ona{A periodic cost satisfies \eqref{eq:ch08:pl} on no set that contains a maximiser, where the gradient vanishes and $C>C^*$. Near a minimiser it may. For a sinusoid $f(\phi) = a-A\cos(\phi-\phi^*)$, one has $\frac12f'(\phi)^2 = \frac A2(1+\cos(\phi-\phi^*))(f(\phi)-f^*)$, so that \eqref{eq:ch08:pl} holds with $\mu = A(1+\cos e_0)/2$ on the set $|\phi-\phi^*|\le e_0<\pi$, a sublevel set that gradient descent does not leave. This is the linear convergence quoted for gradient descent in \Chap~\chref{C.QAOA}; the inequality holds for strongly convex functions, and \citet{karimi2016linear} relate it to other conditions.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item Strong convexity implies the inequality; the inequality does not imply convexity.
    \item A periodic cost satisfies it on no set containing a maximiser; a sinusoid satisfies it on every set $|\phi-\phi^*|\le e_0<\pi$, with $\mu = A(1+\cos e_0)/2$.
    \item With $\alpha = 1/A$ the sinusoid converges even cubically\ona{ (Exercise~\ref{exe:ch08:gate1})}.
\end{itemize}
\end{frame}

\section{Noise and bias}\label{sec:ch08:bias}

\begin{frame}\ft{Bias in the device}
\onp{\footnotesize}
\ona{An estimate $\hat C(\boldsymbol\theta)$ from $M$ copies has a mean}
\begin{equation}\label{eq:ch08:device}
    \tilde C(\boldsymbol\theta) = \mathbb E\,\hat C(\boldsymbol\theta),
\end{equation}
\ona{the cost that the device computes, which does not depend on $M$, and a variance that decreases as $1/M$. On an ideal device, $\tilde C = C$. On a real one, $\tilde C$ differs from $C$, and we call $\tilde C-C$ the \emph{bias} of the device. \Chap~\chref{C.PSR} showed that noise that does not depend on the angles leaves the parameter-shift rule exact for $\tilde C$; it is $C$, however, that the optimizer is meant to minimise. Following \citet{kungurtsev2022iteration}, we assume that}
\begin{equation}\label{eq:ch08:biasvar}
    |\tilde C(\boldsymbol\theta)-C(\boldsymbol\theta)|\le b,\qquad\mathrm{Var}\,\hat C(\boldsymbol\theta)\le\sigma^2/M\qquad\text{for all }\boldsymbol\theta,
\end{equation}
\ona{with $\sigma^2$ a bound on the variance of a single measured value, as in \Chap~\chref{C.PSR}, and that the estimates of different circuits are independent. The bias has several sources:}
\begin{itemize}\setlength\itemsep{0.2em}
    \item Depolarisation before the measurement: $\tilde C = (1-p)C+p\Tr(O)/2^n$. The bias is large, but $\tilde C$ has the stationary points of $C$ (\Chap~\chref{C.PSR}).
    \item Errors of the read-out, which map $\braket{Z_i}$ affinely but mix terms into $\braket{Z_iZ_j}$.
    \item Coherent errors: a gate applied with a wrong angle or axis. A fixed error moves the minimisers.
    \item Errors that depend on the angle, as when the duration of a pulse grows with it: the shift rule is then biased even for $\tilde C$.
\end{itemize}
\ona{More copies reduce the variance of an estimate, not its bias.}

\ona{The bias affects the global method of Section~\ref{sec:ch08:complexity} as well as the gradient methods. By the lemma on the grid in \Chap~\chref{C.PSR}, values of $\tilde C$ reconstruct $C$ to within $\sqrt{|\mathcal G|}\,b$, where $|\mathcal G|$ is the number of points of the grid, and to within $b$ if $\tilde C$ is a trigonometric polynomial of the same degrees as $C$; for the gradient methods, the bias sets a floor below which $\epsilon$ cannot be taken, as the next two sections show.}
\end{frame}

\begin{frame}\ft{Errors of the gates}
\onp{\footnotesize}
\begin{lem}\label{lem:ch08:gate}
If a device applies unitaries $\tilde U_j$ in place of $U_j$ and measures $O$ exactly, then $|\tilde C(\boldsymbol\theta)-C(\boldsymbol\theta)|\le w\sum_{j=1}^d\|\tilde U_j-U_j\|$, where $w$ is the spectral width of $O$.
\end{lem}
\ona{\emph{Proof.} Replacing $O$ by $O$ minus the midpoint of its spectrum changes $C$ and $\tilde C$ by the same constant and gives $\|O\| = w/2$. With $\ket\psi$ and $\ket{\tilde\psi}$ the two states, $\tilde C-C = \braket{\tilde\psi-\psi|O|\tilde\psi}+\braket{\psi|O|\tilde\psi-\psi}$, so that $|\tilde C-C|\le w\,\|\tilde\psi-\psi\|$. Finally, $\tilde U_d\cdots\tilde U_1-U_d\cdots U_1 = \sum_j\tilde U_d\cdots\tilde U_{j+1}(\tilde U_j-U_j)U_{j-1}\cdots U_1$, a sum of $d$ terms of norm at most $\|\tilde U_j-U_j\|$. \qed}
\onp{Proof: $|\tilde C-C|\le w\|\tilde\psi-\psi\|$ with $\|O\| = w/2$, and a telescoping sum over the gates.}

\ona{A rotation applied with its angle in error by $\delta$, $R_P(\theta+\delta)$ in place of $R_P(\theta)$, contributes $\|R_P(\delta)-\one\| = 2\sin(|\delta|/4)$. In the worked example, with $w = 4$, if the four gates $R_X(2\beta)$ of the mixer are each applied as $R_X(2\beta+\delta)$, the lemma gives $b\le4\cdot4\cdot2\sin(\delta/4)$, which is $0.40$ for $\delta = 0.05$; the largest bias found at $20\,000$ random angles of the eight gates is $0.094$, and at the angles of QAOA it is $2\sin\delta = 0.10$. Such a device computes $\tilde F(\beta,\gamma) = F(\beta+\delta/2,\gamma)$, whose minimisers are displaced by $\delta/2$; we return to it below.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item An angle in error by $\delta$: $\|R_P(\delta)-\one\| = 2\sin(|\delta|/4)$.
    \item A mixer over-rotated by $\delta = 0.05$ in the worked example: $b\le0.40$ by the lemma, $0.094$ at $20\,000$ random angles of the eight gates. The device computes $F(\beta+\delta/2,\gamma)$, with displaced minimisers.
    \item The bound is additive over the gates: deeper circuits, larger bias.
\end{itemize}

\begin{frame}\ft{Errors of the read-out}
\onp{\footnotesize}
\onp{Each qubit read out independently, reporting $1$ for $0$ with probability $q_{0i}$ and $0$ for $1$ with probability $q_{1i}$: with $a_i = 1-q_{0i}-q_{1i}$ and $c_i = q_{1i}-q_{0i}$, the reported $Z_iZ_j$ has the mean}
\ona{Read-out errors are of a different kind. Suppose that each qubit $i$ is read out independently, reporting $1$ for $0$ with probability $q_{0i}$ and $0$ for $1$ with probability $q_{1i}$. The reported value of $Z_i$ then has the conditional mean $a_iz_i+c_i$, with $a_i = 1-q_{0i}-q_{1i}$ and $c_i = q_{1i}-q_{0i}$, where $z_i = \pm1$ is the true value, and that of $Z_iZ_j$ has the mean}
\begin{equation}\label{eq:ch08:readout}
    a_ia_j\braket{Z_iZ_j}+a_ic_j\braket{Z_i}+c_ia_j\braket{Z_j}+c_ic_j.
\end{equation}
\ona{In QAOA with the mixer $\sum_iX_i$, the operator $X^{\otimes n}$ commutes with every gate and leaves $\ket+^{\otimes n}$ unchanged, while it anticommutes with $Z_i$; hence $\braket{Z_i} = 0$ for all angles. The cost of MaxCut that the device computes is then that of the instance with the weights $a_ia_jw_{ij}$, up to a constant: a reweighting, which leaves the stationary points unchanged when the $a_i$ are equal. On the $4$-cycle all edges are alike, and with the rates of \texttt{lima}, \texttt{santiago}, \texttt{bogota} and \texttt{manila} (Table~\ref{tab:ch08:ibm}) on the four qubits, the device computes $-2+0.812\,(F+2)$ plus a constant; the minimisers do not move (Exercise~\ref{exe:ch08:readout}). Read-out errors that are correlated between qubits, and noise during the circuit that is not invariant under conjugation by $X^{\otimes n}$, such as the decay of $\ket1$ to $\ket0$, break the symmetry.}
\onp{\begin{itemize}\setlength\itemsep{0.2em}
    \item In QAOA with the mixer $\sum_iX_i$: $X^{\otimes n}$ commutes with the circuit and fixes $\ket+^{\otimes n}$, so $\braket{Z_i} = 0$.
    \item Independent read-out errors then reweight the edges, $w_{ij}\mapsto a_ia_jw_{ij}$: on the $4$-cycle the minimisers do not move.
    \item Correlated read-out errors, and the decay of $\ket1$ to $\ket0$ during the circuit, break the symmetry.
\end{itemize}}

\begin{frame}\ft{Stochastic gradient descent with biased estimates}
\onp{\scriptsize}
\ona{We write the estimate of the gradient at $\boldsymbol\theta_k$ as}
\begin{equation}\label{eq:ch08:decomp}
    \boldsymbol g_k = \nabla C(\boldsymbol\theta_k)+\boldsymbol\beta_k+\boldsymbol\rho_k,\qquad\boldsymbol\beta_k = \mathbb E[\boldsymbol g_k\,|\,\boldsymbol\theta_k]-\nabla C(\boldsymbol\theta_k),
\end{equation}
\ona{its bias $\boldsymbol\beta_k$ and its random error $\boldsymbol\rho_k$, and assume that for constants $\zeta$, $A$ and $V$,}
\begin{equation}\label{eq:ch08:assume}
    \|\boldsymbol\beta_k\|\le\zeta,\qquad\mathbb E\big[\|\boldsymbol\rho_k\|^2\,\big|\,\boldsymbol\theta_k\big]\le A\,\|\nabla C(\boldsymbol\theta_k)+\boldsymbol\beta_k\|^2+V.
\end{equation}
\ona{Shot noise contributes to $V$, which decreases with $M$; the constant $A$ allows an error that grows with the gradient, as for the derivative-free estimators; the bias $\zeta$ comes from the device. In Figure~\ref{fig:ch08:noise}, $\boldsymbol\beta_k$ is the red arrow, of norm at most $\zeta = 8\sin\delta$, and $\boldsymbol\rho_k$ is the scatter of the blue points about their mean, with $A = 0$ and $V = 0.0098$ at $\boldsymbol\theta_0$; for simultaneous perturbation with exact values, $A = 1$ and $V = 0$.}
\begin{thm}[\citet{ajalloeian2020convergence,kungurtsev2022iteration}]\label{thm:ch08:sgd}
Let $\|\nabla^2C\|\le L$, and let the estimates satisfy \eqref{eq:ch08:assume}. With a constant step $\alpha\le1/((A+1)L)$, the iterates \eqref{eq:ch08:iter} satisfy
\begin{equation}\label{eq:ch08:sgd}
    \frac1K\sum_{k=0}^{K-1}\mathbb E\,\|\nabla C(\boldsymbol\theta_k)\|^2\le\frac{2\,(C(\boldsymbol\theta_0)-C^*)}{\alpha K}+\alpha LV+\zeta^2.
\end{equation}
\end{thm}
\ona{\emph{Proof.} By Lemma~\ref{lem:ch08:descent}, $C(\boldsymbol\theta_{k+1})\le C(\boldsymbol\theta_k)-\alpha\nabla C(\boldsymbol\theta_k)^\top\boldsymbol g_k+\frac{L\alpha^2}2\|\boldsymbol g_k\|^2$. Write $\boldsymbol\nabla$ for $\nabla C(\boldsymbol\theta_k)$ and $\boldsymbol\beta$ for $\boldsymbol\beta_k$. Conditionally on $\boldsymbol\theta_k$,}
\ona{\begin{gather*}
    \mathbb E\,\boldsymbol\nabla^\top\boldsymbol g_k = \boldsymbol\nabla^\top(\boldsymbol\nabla+\boldsymbol\beta) = \tfrac12\big(\|\boldsymbol\nabla\|^2+\|\boldsymbol\nabla+\boldsymbol\beta\|^2-\|\boldsymbol\beta\|^2\big),\\
    \mathbb E\,\|\boldsymbol g_k\|^2 = \|\boldsymbol\nabla+\boldsymbol\beta\|^2+\mathbb E\,\|\boldsymbol\rho_k\|^2\le(A+1)\|\boldsymbol\nabla+\boldsymbol\beta\|^2+V.
\end{gather*}}
\ona{Hence $\mathbb E[C(\boldsymbol\theta_{k+1})\,|\,\boldsymbol\theta_k]\le C(\boldsymbol\theta_k)-\frac\alpha2\|\boldsymbol\nabla\|^2-\frac\alpha2(1-(A+1)L\alpha)\|\boldsymbol\nabla+\boldsymbol\beta\|^2+\frac\alpha2\zeta^2+\frac{L\alpha^2}2V$, and the third term is not positive by the choice of $\alpha$. Taking expectations, summing over $k<K$ and using $\mathbb E\,C(\boldsymbol\theta_K)\ge C^*$ gives $\frac\alpha2\sum_{k<K}\mathbb E\|\nabla C(\boldsymbol\theta_k)\|^2\le C(\boldsymbol\theta_0)-C^*+\frac{K\alpha}2\zeta^2+\frac{KL\alpha^2}2V$; divide by $K\alpha/2$. \qed}
\onp{Proof: the descent lemma, $\mathbb E\,\boldsymbol\nabla^\top\boldsymbol g_k = \frac12(\|\boldsymbol\nabla\|^2+\|\boldsymbol\nabla+\boldsymbol\beta\|^2-\|\boldsymbol\beta\|^2)$, and a telescoping sum.}
\begin{itemize}\setlength\itemsep{0.15em}
    \item Three terms: the progress of the method, which decreases as $1/K$; the noise, $\alpha LV$, which a smaller step reduces; and the bias, $\zeta^2$, which nothing in the method reduces.
    \item The left-hand side is $\mathbb E\,\|\nabla C(\boldsymbol\theta_R)\|^2$ for an iterate $\boldsymbol\theta_R$ chosen uniformly at random, which is how a guarantee for a single output is obtained \citep{ghadimi2013stochastic}.
\end{itemize}
\end{frame}

\begin{frame}\ft{Unbiased estimates}
\onp{\footnotesize}
\begin{cor}[\citet{ghadimi2013stochastic}]\label{cor:ch08:unbiased}
If $\zeta = 0$ and $A = 0$, the step $\alpha = \min\{1/L,\ (2(C(\boldsymbol\theta_0)-C^*)/(LVK))^{1/2}\}$ gives
\begin{equation}\label{eq:ch08:rate}
    \frac1K\sum_{k=0}^{K-1}\mathbb E\,\|\nabla C(\boldsymbol\theta_k)\|^2\le\frac{4L\,(C(\boldsymbol\theta_0)-C^*)}K+2\Big(\frac{2L\,(C(\boldsymbol\theta_0)-C^*)\,V}K\Big)^{1/2},
\end{equation}
and $K\ge\max\{8L(C(\boldsymbol\theta_0)-C^*)/\epsilon,\ 32L(C(\boldsymbol\theta_0)-C^*)V/\epsilon^2\}$ iterations give an $\epsilon$-stationary point.
\end{cor}
\ona{\emph{Proof.} If the minimum is the square root, the two terms of \eqref{eq:ch08:sgd} both equal $(2L(C(\boldsymbol\theta_0)-C^*)V/K)^{1/2}$. Otherwise $\alpha = 1/L$ and $V<2L(C(\boldsymbol\theta_0)-C^*)/K$, and the bound is at most $4L(C(\boldsymbol\theta_0)-C^*)/K$. The choice of $K$ makes each term of \eqref{eq:ch08:rate} at most $\epsilon/2$. \qed}

\ona{With noise, the iteration complexity is of the order of $1/\epsilon^2$, against $1/\epsilon$ without. \citet{arjevani2023lower} show that the order $LV(C(\boldsymbol\theta_0)-C^*)/\epsilon^2$ cannot be improved, in the worst case over smooth functions in high dimension, by any method that uses unbiased estimates of the gradient with variance $V$. The step depends on $K$, $L$, $V$ and $C(\boldsymbol\theta_0)-C^*$; with a step that decreases as $k^{-1/2}$, the same rate holds up to a logarithmic factor without knowledge of $K$.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item Of the order of $L/\epsilon+LV/\epsilon^2$ iterations; with noise the second term dominates for small $\epsilon$.
    \item No method using unbiased estimates of variance $V$ does better in the worst case \citep{arjevani2023lower}.
\end{itemize}
\end{frame}

\begin{frame}\ft{The parameter-shift rule}
\onp{\footnotesize}
\begin{cor}\label{cor:ch08:psr}
Let $C$ be the cost of $d$ Pauli rotations, estimated by a device that satisfies \eqref{eq:ch08:biasvar}. The parameter-shift rule $\partial_jC(\boldsymbol\theta) = \frac12[C(\boldsymbol\theta+\frac\pi2\boldsymbol e_j)-C(\boldsymbol\theta-\frac\pi2\boldsymbol e_j)]$, with $M$ copies for each of its $2d$ circuits, gives estimates that satisfy \eqref{eq:ch08:assume} with $\zeta = \sqrt d\,b$, $A = 0$ and $V = d\sigma^2/(2M)$. Hence, for $\alpha\le1/L$,
\begin{equation}\label{eq:ch08:psrbound}
    \frac1K\sum_{k=0}^{K-1}\mathbb E\,\|\nabla C(\boldsymbol\theta_k)\|^2\le\frac{2\,(C(\boldsymbol\theta_0)-C^*)}{\alpha K}+\frac{\alpha Ld\sigma^2}{2M}+db^2.
\end{equation}
\end{cor}
\ona{\emph{Proof.} The $j$-th component of the estimate is $\frac12[\hat C(\boldsymbol\theta+\frac\pi2\boldsymbol e_j)-\hat C(\boldsymbol\theta-\frac\pi2\boldsymbol e_j)]$. By the parameter-shift rule its mean is $\partial_jC(\boldsymbol\theta)$ plus half the difference of the biases at the two points, which is at most $b$ in absolute value; hence $\|\boldsymbol\beta_k\|\le\sqrt d\,b$. Its variance is at most $\frac14(\sigma^2/M+\sigma^2/M)$, and the variances of the $d$ components add. \qed}

\ona{This is the bound of \citet{kungurtsev2022iteration} for the parameter-shift rule, with the bias $\sqrt d\,b$ of the gradient in place of the bias $b$ of a value. Each iteration requires $2d$ circuits and $2dM$ copies. With $N$ copies in all, $K = N/(2dM)$, and the step of Corollary~\ref{cor:ch08:unbiased}, the noise term of \eqref{eq:ch08:rate} becomes $2d\,(2L(C(\boldsymbol\theta_0)-C^*)\sigma^2/N)^{1/2}$, which depends on $N$ alone: as far as the bound is concerned, and once $K$ is large enough for the first term, many iterations with few copies are as good as few iterations with many, down to $M = 1$ \citep{sweke2020stochastic}. What favours many copies is the time taken to submit a circuit (\Chap~\chref{C.PerIteration}). In QAOA of depth $p$, $\gamma_k$ controls $m$ gates and $\beta_k$ controls $n$ gates with twice the angle; by the chain rule of \Chap~\chref{C.PSR}, the derivative with respect to $\gamma_k$ is the sum of those with respect to its gates, and that with respect to $\beta_k$ twice the sum. The biases and the variances of the gates add, and the estimate from the $2p(m+n)$ circuits of Corollary~\ref{cor:ch08:qaoa} has $\zeta\le\sqrt{p\,(m^2+4n^2)}\,b$ and $V\le p\,(m+4n)\,\sigma^2/(2M)$: in the worked example, $\zeta\le\sqrt{80}\,b$ and $V\le10\sigma^2/M$.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item The floor $db^2$ does not decrease with $K$, $M$ or $\alpha$.
    \item With $N$ copies in all, the noise term depends on $N$ alone: few copies per circuit suffice \citep{sweke2020stochastic}; the latency of a circuit favours many.
    \item QAOA of depth $p$, gate by gate: $\zeta\le\sqrt{p(m^2+4n^2)}\,b$, $V\le p(m+4n)\sigma^2/(2M)$; in the worked example $\sqrt{80}\,b$ and $10\sigma^2/M$.
\end{itemize}
\end{frame}

\begin{frame}\ft{The floor cannot be removed}
\onp{\scriptsize}
\begin{propn}\label{propn:ch08:floor}
Let $d\ge1$, $a>0$ and $0<\delta<\pi$, and let $C_1(\boldsymbol\theta) = -a\cos(\theta_1+\dots+\theta_d)$ and $C_2(\boldsymbol\theta) = -a\cos(\theta_1+\dots+\theta_d-\delta)$, the costs of $R_Y(\theta_d)\cdots R_Y(\theta_1)$ on one qubit with $O = -aZ$, without and with a fixed rotation $R_Y(-\delta)$. Then $|C_1-C_2|\le b := 2a\sin(\delta/2)$, and every method that uses only the estimates of a device with bias at most $b$, with any number of iterations and copies, produces an output $\hat{\boldsymbol\theta}$ such that, for $i = 1$ or $i = 2$,
\begin{equation}\label{eq:ch08:floor}
    \mathbb E\,\|\nabla C_i(\hat{\boldsymbol\theta})\|^2\ge\tfrac d4\,b^2\cos^2(\delta/2).
\end{equation}
\end{propn}
\ona{\emph{Proof.} A device whose estimates have the mean $C_1$ is a device for $C_1$ without bias and a device for $C_2$ with bias $C_1-C_2$, which is at most $b$. The method receives the same estimates in both cases, and its output has the same distribution. With $s = \sum_j\hat\theta_j$, $\|\nabla C_1(\hat{\boldsymbol\theta})\| = a\sqrt d\,|\sin s|$ and $\|\nabla C_2(\hat{\boldsymbol\theta})\| = a\sqrt d\,|\sin(s-\delta)|$, and $|\sin s|+|\sin(s-\delta)|\ge\sin\delta$, since $\sin\delta = \sin s\cos(s-\delta)-\cos s\sin(s-\delta)$. Hence $\|\nabla C_1\|^2+\|\nabla C_2\|^2\ge\frac12(\|\nabla C_1\|+\|\nabla C_2\|)^2\ge\frac12da^2\sin^2\delta$, the larger of the two expectations is at least $\frac14da^2\sin^2\delta$, and $a\sin\delta = b\cos(\delta/2)$. \qed}
\onp{Proof: the method cannot tell $C_1$ from $C_2$, and at every point one of the two gradients has norm at least $\frac12a\sqrt d\sin\delta$.}

\ona{As $\delta\to0$, no method can guarantee less than about $db^2/4$, however many copies and iterations it uses, and the floor $db^2$ of Corollary~\ref{cor:ch08:psr} is attained up to a factor of about $4$. The bias in the proposition is a coherent error of a single gate. Figure~\ref{fig:ch08:twocost} shows the case $d = 1$: in (a), the two costs and the band of width $2b$ about $C_1$ that contains $C_2$; in (b), the two slopes, the larger of which never falls below $\frac12\sin\delta$.}

\section{Derivative-free estimators under bias}\label{sec:ch08:dfo}

\begin{frame}\ft{Simultaneous perturbation under bias}
\onp{\scriptsize}
\ona{Simultaneous perturbation (\Chap~\chref{C.PSR}) estimates the gradient from two circuits, at $\boldsymbol\theta\pm c\boldsymbol\Delta$ with $\boldsymbol\Delta$ uniform on $\{\pm1\}^d$. Since $1/\Delta_i = \Delta_i$, the estimate is $\boldsymbol g = [\hat C(\boldsymbol\theta+c\boldsymbol\Delta)-\hat C(\boldsymbol\theta-c\boldsymbol\Delta)]\,\boldsymbol\Delta/(2c)$.}
\begin{propn}\label{propn:ch08:spsa}
Let $\|\nabla^2C\|\le L$ and $|D^3C[\boldsymbol h,\boldsymbol h,\boldsymbol h]|\le L_3\|\boldsymbol h\|^3$, and let the device satisfy \eqref{eq:ch08:biasvar}. Then, with $\kappa = L_3d^{3/2}/6$,
\begin{equation}\label{eq:ch08:spsa}
    \|\mathbb E\,\boldsymbol g-\nabla C(\boldsymbol\theta)\|\le\kappa c^2+\frac bc,\qquad\mathbb E\,\|\boldsymbol g\|^2\le4d\,\|\nabla C(\boldsymbol\theta)\|^2+L^2c^2d^3+\frac{d\,(4b^2+\sigma^2/M)}{2c^2},
\end{equation}
and for $\alpha\le1/((8d+1)L)$ the iterates of simultaneous perturbation satisfy
\begin{equation}\label{eq:ch08:spsabound}
    \frac1K\sum_{k=0}^{K-1}\mathbb E\,\|\nabla C(\boldsymbol\theta_k)\|^2\le\frac{2\,(C(\boldsymbol\theta_0)-C^*)}{\alpha K}+\alpha L\Big[L^2c^2d^3+\frac{d\,(4b^2+\sigma^2/M)}{2c^2}\Big]+2\Big(\kappa c^2+\frac bc\Big)^2.
\end{equation}
\end{propn}
\ona{\emph{Proof.} By Taylor's theorem, the terms of second order cancel in the difference, and $[C(\boldsymbol\theta+c\boldsymbol\Delta)-C(\boldsymbol\theta-c\boldsymbol\Delta)]/(2c) = \nabla C(\boldsymbol\theta)^\top\boldsymbol\Delta+e$ with $|e|\le\frac{c^2}6L_3\|\boldsymbol\Delta\|^3 = \kappa c^2$. Since $\mathbb E\,\boldsymbol\Delta\boldsymbol\Delta^\top = \one$, the first term contributes $\nabla C(\boldsymbol\theta)$ to $\mathbb E\,\boldsymbol g$. For a unit vector $\boldsymbol v$, $\mathbb E\,|\boldsymbol\Delta^\top\boldsymbol v|\le(\mathbb E(\boldsymbol\Delta^\top\boldsymbol v)^2)^{1/2} = 1$; hence the remainder contributes at most $\kappa c^2$ to $\|\mathbb E\,\boldsymbol g-\nabla C\|$, and the bias of the values, which changes the difference by at most $2b$, at most $b/c$. For the second moment, $\|\boldsymbol g\|^2 = d\,[\hat C(\boldsymbol\theta+c\boldsymbol\Delta)-\hat C(\boldsymbol\theta-c\boldsymbol\Delta)]^2/(4c^2)$. By Lemma~\ref{lem:ch08:descent}, applied to $C$ and to $-C$, $|C(\boldsymbol\theta+c\boldsymbol\Delta)-C(\boldsymbol\theta-c\boldsymbol\Delta)-2c\nabla C^\top\boldsymbol\Delta|\le Lc^2d$, so that the square of the difference of the means has the expectation at most $2[8c^2\|\nabla C\|^2+2L^2c^4d^2]+8b^2$; the random parts add $2\sigma^2/M$. Multiplying by $d/(4c^2)$ gives the second inequality. Finally, with $\zeta = \kappa c^2+b/c$, the inequality $\|\nabla C\|^2\le2\|\nabla C+\boldsymbol\beta\|^2+2\zeta^2$ gives \eqref{eq:ch08:assume} with $A = 8d$ and $V$ the bracket in \eqref{eq:ch08:spsabound} plus $8d\zeta^2$, and $\alpha L\cdot8d\zeta^2\le\zeta^2$ by the choice of $\alpha$; Theorem~\ref{thm:ch08:sgd} gives \eqref{eq:ch08:spsabound}. \qed}
\onp{Proof: Taylor's theorem to third order, $\mathbb E\,|\boldsymbol\Delta^\top\boldsymbol v|\le1$ for unit $\boldsymbol v$, the descent lemma for the second moment, and the theorem on biased estimates with $A = 8d$.}

\ona{The bias of the values enters as $b/c$, which calls for a large perturbation, and the error of the Taylor expansion as $\kappa c^2$, which calls for a small one. The best choice, $c^3 = b/(2\kappa)$, gives the floor $2(\kappa c^2+b/c)^2 = 9\cdot2^{-1/3}\kappa^{2/3}b^{4/3}$, which decreases with $b$ more slowly than the floor $db^2$ of the parameter-shift rule. The step is smaller by a factor of about $8d$, and the number of iterations larger by a factor of the order of $d$; each iteration, however, requires $2$ circuits instead of $2d$. This is the trade-off of the estimators compared in \Chap~\chref{C.PSR}, now over many iterations. The theorem of \citet{kungurtsev2022iteration} for simultaneous perturbation has the same three terms.}
\end{frame}

\begin{frame}\ft{A bias that varies smoothly}
\onp{\footnotesize}
\ona{The term $b/c$ of Proposition~\ref{propn:ch08:spsa} allows a bias that changes by $2b$ between two points at the distance $2c\sqrt d$, however small $c$ is. The bias of a device need not behave so badly. If the noise does not depend on the angles, the proposition on noise of \Chap~\chref{C.PSR} shows that the cost $\tilde C$ that the device computes has the frequencies of $C$, and so does the bias $\tilde C-C$; the over-rotated mixer of the worked example is an instance, with $\tilde F(\beta,\gamma) = F(\beta+\delta/2,\gamma)$.}
\begin{propn}\label{propn:ch08:smooth}
Let $C$ and $\tilde C$ be Hermitian trigonometric polynomials of degrees at most $R_j$ in $\theta_j$, with $|\tilde C-C|\le b$, and let $\tilde W$ be half the width of the range of $\tilde C$. Then $\|\nabla\tilde C-\nabla C\|\le\|\boldsymbol R\|\,b$, and the estimate of simultaneous perturbation from the device has the bias
\begin{equation}\label{eq:ch08:smooth}
    \|\mathbb E\,\boldsymbol g-\nabla C(\boldsymbol\theta)\|\le\|\boldsymbol R\|\,b+\tilde\kappa c^2,\qquad\tilde\kappa = \|\boldsymbol R\|^3\tilde W d^{3/2}/6.
\end{equation}
\end{propn}
\ona{\emph{Proof.} The range of $\tilde C-C$ lies in $[-b,b]$, and Lemma~\ref{lem:ch08:derivs} applied to it gives the first claim. The mean of the estimate is $\mathbb E[\tilde C(\boldsymbol\theta+c\boldsymbol\Delta)-\tilde C(\boldsymbol\theta-c\boldsymbol\Delta)]\boldsymbol\Delta/(2c)$, which, as in the proof of Proposition~\ref{propn:ch08:spsa} with $\tilde C$ in place of $C$, differs from $\nabla\tilde C(\boldsymbol\theta)$ by at most $\tilde\kappa c^2$, since $|D^3\tilde C[\boldsymbol h,\boldsymbol h,\boldsymbol h]|\le\|\boldsymbol R\|^3\tilde W\|\boldsymbol h\|^3$ by Lemma~\ref{lem:ch08:derivs}. \qed}
\onp{Proof: the lemma on derivatives applied to $\tilde C-C$, whose range lies in $[-b,b]$, and the Taylor argument of the previous proposition for $\tilde C$.}
\begin{itemize}\setlength\itemsep{0.15em}
    \item The term $b/c$ becomes $\|\boldsymbol R\|\,b$, which does not grow as $c\to0$. In the same way, the term $2db^2/c^2$ of the second moment \eqref{eq:ch08:spsa} becomes $2d^2\|\boldsymbol R\|^2b^2$, and only the shot noise, $d\sigma^2/(2Mc^2)$, still grows.
    \item The floor $2(\|\boldsymbol R\|b+\tilde\kappa c^2)^2$ tends to $2\|\boldsymbol R\|^2b^2$, which is $2db^2$ for $d$ Pauli rotations: the order of the floor $db^2$ of the parameter-shift rule.
    \item In the worked example with the over-rotated mixer, $b = 2\sin\delta = 0.10$ on the angles of QAOA: $b/c = 1.0$ for $c = 0.1$, against $\|\nabla\tilde F-\nabla F\|\le8\sin\delta = 0.40$ and $\|\boldsymbol R\|\,b = \sqrt{20}\cdot0.10 = 0.45$.
\end{itemize}
\end{frame}

\begin{frame}\ft{Gaussian smoothing under bias}
\onp{\scriptsize}
\ona{The two-point estimator of \Chap~\chref{C.PSR}, $\boldsymbol g = [\hat C(\boldsymbol\theta+c\boldsymbol u)-\hat C(\boldsymbol\theta)]\,\boldsymbol u/c$ with $\boldsymbol u\sim\mathcal N(0,\one_d)$, is unbiased for the gradient of the smoothed cost $C_c$ on an ideal device (\Chap~\chref{C.PSR}). We use three bounds of \citet{Nesterov2015}, for $\|\nabla^2C\|\le L$: $|C_c-C|\le\frac{c^2}2Ld$, $\|\nabla C_c-\nabla C\|\le\frac c2L(d+3)^{3/2}$, and $\mathbb E[(C(\boldsymbol\theta+c\boldsymbol u)-C(\boldsymbol\theta))^2\|\boldsymbol u\|^2]/c^2\le\frac{c^2}2L^2(d+6)^3+2(d+4)\|\nabla C(\boldsymbol\theta)\|^2$.}
\onp{Smoothing bounds \citep{Nesterov2015}: $|C_c-C|\le\frac{c^2}2Ld$, $\|\nabla C_c-\nabla C\|\le\frac c2L(d+3)^{3/2}$.}
\begin{propn}\label{propn:ch08:twopoint}
Let $\|\nabla^2C\|\le L$ and let the device satisfy \eqref{eq:ch08:biasvar}. Then $\|\mathbb E\,\boldsymbol g-\nabla C_c(\boldsymbol\theta)\|\le2\sqrt{2/\pi}\,b/c$ and $\mathbb E\,\|\boldsymbol g\|^2\le4(d+4)\|\nabla C(\boldsymbol\theta)\|^2+c^2L^2(d+6)^3+2d\,(4b^2+\sigma^2/M)/c^2$. For $\alpha\le1/((16d+65)L)$, the iterates satisfy
\begin{equation}\label{eq:ch08:2pbound}
    \frac1K\sum_{k=0}^{K-1}\mathbb E\,\|\nabla C(\boldsymbol\theta_k)\|^2\le\frac{4\,(C_c(\boldsymbol\theta_0)-C^*)}{\alpha K}+2\alpha LV_c+\frac{32}\pi\,\frac{b^2}{c^2}+\frac{c^2}2L^2(d+3)^3,
\end{equation}
where $V_c = c^2L^2[(d+6)^3+2(d+4)(d+3)^3]+2d\,(4b^2+\sigma^2/M)/c^2$.
\end{propn}
\ona{\emph{Proof.} By the lemma on Gaussian smoothing of \Chap~\chref{C.PSR}, $\mathbb E\,\boldsymbol g-\nabla C_c(\boldsymbol\theta)$ is the mean of $[(\tilde C-C)(\boldsymbol\theta+c\boldsymbol u)-(\tilde C-C)(\boldsymbol\theta)]\,\boldsymbol u/c$, and $\mathbb E\,|\boldsymbol u^\top\boldsymbol v| = \sqrt{2/\pi}$ for a unit vector $\boldsymbol v$. The second moment follows as for simultaneous perturbation, with the third bound of \citet{Nesterov2015} in place of Taylor's theorem and $\mathbb E\|\boldsymbol u\|^2 = d$. Theorem~\ref{thm:ch08:sgd} applies to $C_c$, which has the smoothness constant $L$ and the infimum at least $C^*$, with $\zeta = 2\sqrt{2/\pi}\,b/c$, $A = 16(d+4)$ and $V = V_c+16(d+4)\zeta^2$, by $\|\nabla C\|^2\le2\|\nabla C_c\|^2+\frac{c^2}2L^2(d+3)^3\le4\|\nabla C_c+\boldsymbol\beta\|^2+4\zeta^2+\frac{c^2}2L^2(d+3)^3$. It bounds the average of $\mathbb E\|\nabla C_c(\boldsymbol\theta_k)\|^2$ by $2(C_c(\boldsymbol\theta_0)-C^*)/(\alpha K)+\alpha LV_c+2\zeta^2$, and the first of these inequalities, applied once more, gives \eqref{eq:ch08:2pbound}. \qed}

\begin{frame}\ft{The experiments of Kungurtsev et al.}
\onp{\footnotesize}
\ona{\citet{kungurtsev2022iteration} compare the three estimators on depth-one QAOA for an instance of MaxCut on four vertices, simulated with depolarising noise of strength $10^{-3}$ after each gate on one qubit and $10^{-2}$ after each CNOT, and with the read-out errors of Table~\ref{tab:ch08:ibm}, extended to four qubits by allowing a single flipped bit in each outcome. \begin{itemize}\setlength\itemsep{0.25em}
    \item With $c = 0.1$ and $\alpha = 0.1$, simultaneous perturbation converged in as few iterations as the parameter-shift rule, to a region of larger fluctuations; with $\alpha = 10^{-3}$ it converged in fewer.
    \item With smaller $c$ and $\alpha = 0.1$, simultaneous perturbation became erratic; the two-point estimator decreased slowly with $c = 0.01$ and made no progress with $c = 10^{-3}$.
    \item In the other regimes, the two-point estimator was the slowest or the noisiest of the three.

\end{itemize}
\ona{Measured in iterations, simultaneous perturbation is thus competitive with the parameter-shift rule at a fraction of the circuits, provided that $c$ is not too small, in agreement with the floors of Table~\ref{tab:ch08:floors}, which favour a large $c$ when the bias is large.}

\section{Decreasing steps}\label{sec:ch08:decr}

\begin{frame}\ft{Decreasing steps}
\onp{\scriptsize}
\ona{With a constant step, the bound of Theorem~\ref{thm:ch08:sgd} keeps the term $\alpha LV$, and the iterates keep moving under the noise. Stochastic approximation in the sense of \citet{robbins1951stochastic} uses steps $\alpha_k$ that decrease, slowly enough that the iterates can travel any distance, $\sum_k\alpha_k = \infty$, and fast enough that the noise is averaged out, $\sum_k\alpha_k^2<\infty$. The proof rests on a theorem of \citet{robbins1971convergence}: if non-negative random variables $X_k$, $Y_k$ and $Z_k$, determined by the first $k$ iterations, satisfy $\mathbb E[X_{k+1}\,|\,\text{the first }k\text{ iterations}]\le X_k-Y_k+Z_k$ and $\sum_kZ_k<\infty$, then $X_k$ converges and $\sum_kY_k<\infty$, almost surely.}
\onp{Robbins--Siegmund \citep{robbins1971convergence}: if $X_k,Y_k,Z_k\ge0$, $\mathbb E[X_{k+1}\,|\,\text{past}]\le X_k-Y_k+Z_k$ and $\sum Z_k<\infty$, then $X_k$ converges and $\sum Y_k<\infty$, almost surely.}
\begin{thm}[\citet{robbins1971convergence,bertsekas2000gradient}]\label{thm:ch08:decr}
Let $\|\nabla^2C\|\le L$, and let the estimate at iteration $k$ satisfy \eqref{eq:ch08:assume} with the constants $\zeta_k$, $A$ and $V_k$. If $\alpha_k\le1/((A+1)L)$ for all $k$ from some $k_0$ on, $\sum_k\alpha_k = \infty$, and
\begin{equation}\label{eq:ch08:decr}
    \sum_k\big(\alpha_k\zeta_k^2+\alpha_k^2V_k\big)<\infty,
\end{equation}
then $C(\boldsymbol\theta_k)$ converges and $\liminf_k\|\nabla C(\boldsymbol\theta_k)\| = 0$, almost surely.
\end{thm}
\ona{\emph{Proof.} The estimate at iteration $k$ depends on the past only through $\boldsymbol\theta_k$, so that conditioning on $\boldsymbol\theta_k$ is conditioning on the first $k$ iterations. The proof of Theorem~\ref{thm:ch08:sgd}, with the step $\alpha_k$, gives for $k\ge k_0$ that $\mathbb E[C(\boldsymbol\theta_{k+1})-C^*\,|\,\boldsymbol\theta_k]\le C(\boldsymbol\theta_k)-C^*-\frac{\alpha_k}2\|\nabla C(\boldsymbol\theta_k)\|^2+\frac12(\alpha_k\zeta_k^2+L\alpha_k^2V_k)$. The theorem of Robbins and Siegmund, with $X_k = C(\boldsymbol\theta_k)-C^*$, $Y_k = \frac{\alpha_k}2\|\nabla C(\boldsymbol\theta_k)\|^2$ and $Z_k = \frac12(\alpha_k\zeta_k^2+L\alpha_k^2V_k)$, shows that $C(\boldsymbol\theta_k)$ converges and $\sum_k\alpha_k\|\nabla C(\boldsymbol\theta_k)\|^2<\infty$, almost surely. If $\|\nabla C(\boldsymbol\theta_k)\|^2\ge\eta>0$ for all large $k$, the sum is infinite, since $\sum_k\alpha_k = \infty$. \qed}
\onp{Proof: the inequality in the proof of the theorem on biased estimates, with $X_k = C(\boldsymbol\theta_k)-C^*$ and $Y_k = \frac{\alpha_k}2\|\nabla C(\boldsymbol\theta_k)\|^2$.}
\begin{itemize}\setlength\itemsep{0.15em}
    \item \citet{bertsekas2000gradient} show more: $\nabla C(\boldsymbol\theta_k)\to0$ almost surely. Every limit point is stationary; which one is not said.
    \item The steps $\alpha_k = a/(k+1+A)^s$ with $\frac12<s\le1$ satisfy $\sum_k\alpha_k = \infty$ and $\sum_k\alpha_k^2<\infty$, which suffice when $V_k\le V$ and $\zeta_k = 0$.
    \item A constant bias $\zeta>0$ makes $\sum_k\alpha_k\zeta^2$ infinite: decreasing steps average the noise out, not the bias.
\end{itemize}
\end{frame}

\begin{frame}\ft{Simultaneous perturbation with decreasing gains}
\onp{\footnotesize}
\ona{For simultaneous perturbation the perturbation $c_k$ must decrease as well, for the error of the Taylor expansion to vanish; but the shot noise grows as $1/c_k^2$. The theorem balances the two.}
\begin{cor}[\citet{spall1992multivariate}]\label{cor:ch08:spsagains}
On a device without bias, simultaneous perturbation with the gains $\alpha_k = a/(k+1+A)^s$ and $c_k = c/(k+1)^t$, where $s\le1$, $s+4t>1$ and $s-t>\frac12$, satisfies the conclusions of \ona{Theorem~\ref{thm:ch08:decr}}\onp{the theorem on decreasing steps}.
\end{cor}
\ona{\emph{Proof.} By the proof of Proposition~\ref{propn:ch08:spsa} with $b = 0$, the estimate at iteration $k$ satisfies \eqref{eq:ch08:assume} with $\zeta_k = \kappa c_k^2$, $A = 8d$ and $V_k = L^2c_k^2d^3+d\sigma^2/(2Mc_k^2)+8d\kappa^2c_k^4$. Hence $\alpha_k\zeta_k^2$ is of the order of $k^{-(s+4t)}$, and $\alpha_k^2V_k$ of the order of $k^{-2(s-t)}$, its largest term being that with $c_k^{-2}$; both are summable under the conditions. As $s\le1$, $\sum_k\alpha_k = \infty$, and as $\alpha_k\to0$, $\alpha_k\le1/((8d+1)L)$ from some $k_0$ on. \qed}
\onp{Proof: $\zeta_k = \kappa c_k^2$ and $V_k\approx d\sigma^2/(2Mc_k^2)$, so that the terms of \eqref{eq:ch08:decr} are of the orders $k^{-(s+4t)}$ and $k^{-2(s-t)}$.}
\begin{itemize}\setlength\itemsep{0.2em}
    \item The exponents $s = 0.602$ and $t = 0.101$ recommended in practice \citep{spall1998implementation} satisfy both conditions, barely: $s+4t = 1.006$ and $s-t = 0.501$. The asymptotically optimal choice is $s = 1$, $t = \frac16$.
    \item This is the almost sure convergence of SPSA under decreasing gains quoted in \Chap~\chref{C.QAOA}.
    \item With a bias $b>0$, $\zeta_k\ge b/c_k$ grows, and the corollary fails. If the bias varies smoothly, \ona{Proposition~\ref{propn:ch08:smooth}}\onp{the smooth-bias proposition} gives $\zeta_k\le\|\boldsymbol R\|\,b+\tilde\kappa c_k^2$, which does not vanish: the iterates approach the stationary points of $\tilde C$, not those of $C$.
\end{itemize}
\end{frame}

\section{Conclusions}

\begin{frame}[allowframebreaks]\ft{A short history}
\onp{\scriptsize}
\ona{The following list gives the main steps in the analysis of iterative methods with noisy and biased estimates.}
\begin{description}\setlength\itemsep{0.1em}
    \item[1951, 1952] Stochastic approximation: the root of a function from noisy values \citep{robbins1951stochastic}, and its maximum from noisy differences \citep{kiefer1952stochastic}.
    \item[1963] Linear convergence of gradient methods under the inequality now named after Polyak and \Loj{} \citep{polyak1963gradient}.
    \item[1971] A convergence theorem for non-negative almost supermartingales \citep{robbins1971convergence}.
    \item[1992, 1998] Simultaneous perturbation \citep{spall1992multivariate}, and the gains used in practice \citep{spall1998implementation}.
    \item[2000] Convergence of the gradient in gradient methods with errors \citep{bertsekas2000gradient}.
    \item[2013] The complexity of stochastic first- and zeroth-order methods for non-convex problems \citep{ghadimi2013stochastic}.
    \item[2015] Two-point estimators \citep{duchi2015optimal}; simultaneous perturbation under bounded disturbances \citep{granichin2014simultaneous}.
    \item[2016, 2017] The Polyak--\Loj{} inequality revisited \citep{karimi2016linear}; Gaussian smoothing \citep{Nesterov2015}; the asymptotic bias of stochastic gradient search \citep{tadic2017asymptotic}.
    \item[2019, 2020] Non-asymptotic analyses of biased stochastic approximation \citep{pmlr-v99-karimi19a} and of stochastic gradient descent with biased gradients \citep{ajalloeian2020convergence}; stochastic gradient descent for variational algorithms with few copies \citep{sweke2020stochastic}; lower bounds for finding stationary points \citep{carmon2020lower}.
    \item[2023] Lower bounds for stochastic methods \citep{arjevani2023lower}.
    \item[2024] The iteration complexity of variational algorithms with biased estimates \citep{kungurtsev2022iteration}.
\end{description}
\end{frame}

\begin{frame}\ft{Conclusions}
\onp{\footnotesize}
\begin{itemize}\setlength\itemsep{0.25em}
    \item The cost of $d$ Pauli rotations has $\|\nabla C\|\le\sqrt d\,w/2$ and $\|\nabla^2C\|\le dw/2$, both attained; for QAOA of depth $p$ the constants are polynomial in $p$, $n$ and $m$, and conservative.
    \item Gradient descent finds an $\epsilon$-stationary point in $2L(C(\boldsymbol\theta_0)-C^*)/\epsilon\le dw^2/\epsilon$ iterations, and converges linearly where the Polyak--\Loj{} inequality holds; with unbiased estimates of variance $V$, of the order of $LV/\epsilon^2$ iterations suffice. Neither order can be improved for general smooth functions.
    \item The bias of a device does not change the rate, but adds a floor $\zeta^2$: $db^2$ for the parameter-shift rule. No method can guarantee less than about $db^2/4$.
    \item The derivative-free estimators use two circuits per iteration instead of $2d$, with floors of the orders of $b^{4/3}$ and $b$ in the worst case, and that of the shift rule when the bias varies smoothly, as on the worked example.
    \item Decreasing steps give convergence to stationary points almost surely on an ideal device; they average out the noise, not the bias.
    \item On the worked example, the bound for the shift rule is within a factor of about two of the simulated averages, and an over-rotated mixer produces the predicted floor.
    \item What an iteration costs is the subject of \Chap~\chref{C.PerIteration}; where to start, so that a stationary point is a good one, that of \Chaps~\chref{C.WSRounded}--\chref{C.WSContinuous}.
\end{itemize}
\end{frame}

\begin{frame}\ft{Exercises}
\onp{\footnotesize}
\begin{exe}[Gradient descent on one gate]\label{exe:ch08:gate1}
For $f(\phi) = a-A\cos(\phi-\phi^*)$, show that gradient descent with $\alpha = 1/A$ maps the error $e = \phi-\phi^*$ to $e-\sin e$, and hence converges cubically. Compute the iterates from $\phi_0 = -0.1$ for the gate of \ona{Fig.~\ref{fig:ch08:descent}}\onp{the figure on the descent lemma}, and compare the decrease at each step with that guaranteed by \ona{Lemma~\ref{lem:ch08:descent}}\onp{the descent lemma}.
\end{exe}
\begin{exe}[The Polyak--\Loj{} inequality]\label{exe:ch08:pl}
Show that a function with $\nabla^2C\succeq\mu\one$ satisfies \eqref{eq:ch08:pl}. Show that $x^2+3\sin^2x$ is not convex, and find numerically the largest $\mu$ with which it satisfies \eqref{eq:ch08:pl} on $\R$; \citet{karimi2016linear} give $\mu = 1/32$. Show that no non-constant periodic function satisfies \eqref{eq:ch08:pl} on $\R^d$.
\end{exe}
\end{frame}

\begin{frame}\ft{Exercises, continued}
\onp{\footnotesize}
\begin{exe}[A fixed number of copies]\label{exe:ch08:budget}
For the parameter-shift rule with $N$ copies in all, $K$ iterations and $M = N/(2dK)$ copies per circuit, show that the step of \ona{Corollary~\ref{cor:ch08:unbiased}}\onp{the corollary on unbiased estimates} gives the bound $4L(C(\boldsymbol\theta_0)-C^*)/K+2d(2L(C(\boldsymbol\theta_0)-C^*)\sigma^2/N)^{1/2}+db^2$. Show that $K\ge8L(C(\boldsymbol\theta_0)-C^*)/\epsilon$ and $N\ge32d^2L(C(\boldsymbol\theta_0)-C^*)\sigma^2/\epsilon^2$ make it at most $\epsilon+db^2$, and compute the number of circuits and the copies per circuit for $d = 2$, $L = 20$, $C(\boldsymbol\theta_0)-C^* = 0.92$, $\sigma^2 = 2$ and $\epsilon = 0.01$.
\end{exe}
\begin{exe}[Read-out errors in QAOA]\label{exe:ch08:readout}
Show that $X^{\otimes n}$ commutes with the gates of QAOA with the mixer $\sum_iX_i$ and leaves $\ket+^{\otimes n}$ unchanged, and deduce that $\braket{Z_i} = 0$. Derive the cost of MaxCut that a device with independent read-out errors computes, and show that for the $4$-cycle, with the rates of \texttt{lima}, \texttt{santiago}, \texttt{bogota} and \texttt{manila} on the four qubits, it is $-2+0.812\,(F+2)$ plus a constant.
\end{exe}
\end{frame}

\begin{frame}\ft{Exercises, continued}

\begin{exe}[An over-rotated mixer]\label{exe:ch08:mixer}
In the worked example, let every $R_X(2\beta)$ be applied as $R_X(2\beta+\delta)$. Show that the device computes $F(\beta+\delta/2,\gamma)$, that the bias of the estimate of the gradient by the parameter-shift rule has norm at most $8\sin\delta$, and that $\|\nabla F\|^2 = 16\sin^22\delta$ at the minimisers of the device's cost. Compare with the bounds $b\le0.40$ of \ona{Lemma~\ref{lem:ch08:gate}}\onp{the lemma on gate errors} and $\zeta\le\sqrt{80}\,b$.
\end{exe}
\begin{exe}[The floors of the derivative-free estimators]\label{exe:ch08:floors}
Derive the best perturbations and the floors in \ona{Table~\ref{tab:ch08:floors}}\onp{the table of floors}. For $d = 8$, $L = 8$, $L_3 = 8^{3/2}$ and $b = 10^{-3}$, compute the best $c$ for simultaneous perturbation and for the two-point estimator, and find the bias above which the floor of simultaneous perturbation exceeds $\max\|\nabla C\|^2 = 8$.
\end{exe}
